  function mergeParams(params) {
    var merged = Object.assign({}, DEFAULT_PARAMS);
    var p = params || {};
    var k;
    for (k in merged) {
      if (!Object.prototype.hasOwnProperty.call(merged, k)) continue;
      if (k === 'key') {
        merged.key = isBlank(p.key) ? '' : String(p.key);
        continue;
      }
      if (!isBlank(p[k])) merged[k] = String(p[k]);
    }
    return merged;
  }

  function sha512Hex(msg) {
    var enc = new TextEncoder();
    return crypto.subtle.digest('SHA-512', enc.encode(msg)).then(function (buf) {
      return Array.from(new Uint8Array(buf)).map(function (b) { return b.toString(16).padStart(2, '0'); }).join('');
    });
  }

  function generateHash() {
    setErr('');
    var p = wizardProgress();
    if (!p.paramsSaved) { setErr('Save payment parameters first.'); return; }
    if (!p.pipeShown || hashStage < 1) { setErr('Click Show pre-hash string before computing the hash.'); return; }
    if (isBlank(salt)) { setErr('Salt is required.'); return; }
    var params = mergeParams(lab.params);
    var pipe = buildForwardHashString(params, salt);
    setBusy(true);
    sha512Hex(pipe).then(function (h) {
      var labData = labEnsure();
      labData.salt = salt;
      labData.pipeString = pipe;
      labData.hashHex = h;
      labData.progress.hashGenerated = true;
      labWrite(labData);
      setLab(labData);
      setPipeString(pipe);
      setHashHex(h);
      setHashStage(2);
      setMsg('Hash generated. You can now continue to test checkout.');
      setBusy(false);
      setErr('');
    }).catch(function () {
      setErr('SHA-512 unavailable in this browser context.');
      setBusy(false);
    });
  }

  function syncPayForm(data) {
    var form = payFormRef.current;
    if (!form) return false;
    var p = mergeParams(data.params);
    var fields = {
      key: p.key, txnid: p.txnid, amount: p.amount, productinfo: p.productinfo,
      firstname: p.firstname, email: p.email, phone: p.phone, surl: p.surl, furl: p.furl,
      udf1: p.udf1 || '', udf2: p.udf2 || '', udf3: p.udf3 || '', udf4: p.udf4 || '', udf5: p.udf5 || '',
      hash: data.hashHex
    };
    var name;
    for (name in fields) {
      if (!Object.prototype.hasOwnProperty.call(fields, name)) continue;
      var existing = form.elements[name];
      if (!existing) {
        existing = document.createElement('input');
        existing.type = 'hidden';
        existing.name = name;
        form.appendChild(existing);
      }
      existing.value = fields[name];
    }
    return true;
  }

  function verifyReverseHash() {
    if (isBlank(salt)) { setErr('Enter test salt (same as hash step).'); return; }
    if (isBlank(responseHash)) { setErr('Paste the hash from PayU response.'); return; }
    setBusy(true);
    setErr('');
    setMatch(null);
    setComputedRev('');
    sha512Hex(buildReverseHashString(salt, revResp)).then(function (h) {
      setComputedRev(h);
      var ok = h.toLowerCase() === empty(responseHash).toLowerCase();
      setMatch(ok);
      if (ok) {
        var labData = labEnsure();
        labData.progress.verifyDone = true;
        labWrite(labData);
        setLab(labData);
        setMsg('Reverse hash verified. On your server, recompute and compare case-sensitively before updating orders.');
      } else {
        setMsg('');
        setErr('Hash mismatch — check field order, salt, and status.');
      }
      setBusy(false);
    }).catch(function () { setErr('SHA-512 unavailable.'); setBusy(false); });
  }

          <button type="button" onClick={labClick(function () { submitPayuCheckout('_blank'); })} disabled={!readyPay} style={btnPrimaryStyle}>
            {amountLabel !== '—' ? 'Pay ₹' + amountLabel + ' on PayU Test Checkout' : 'Open PayU test checkout (new tab)'}
          </button>
          <button type="button" onClick={labClick(function () { submitPayuCheckout('_top'); })} disabled={!readyPay} style={btnSecondaryStyle}>
            Open in this tab
          </button>
          <button type="button" onClick={labClick(function () { setShowCurl(!showCurl); })} disabled={!readyPay} style={btnSecondaryStyle}>
            {showCurl ? 'Hide curl' : 'Show curl'}
          </button>
          {readyPay ? (
            <button type="button" onClick={labClick(function () { copyText(buildCurl(lab), setCopyMsg); })} style={btnSecondaryStyle}>Copy curl</button>
          ) : null}
        </div>
        {showCurl && readyPay ? <pre style={codeBlockPreStyle}>{buildCurl(lab)}</pre> : null}
        <p style={mutedTextStyle} role="note">
          {'Test card: 5123456789012346 · OTP 123456 · UPI: anything' + '@' + 'payu'}
        </p>
        {readyPay ? (
          <pre style={codeBlockPreStyle}>
            {'POST ' + TEST_PAYMENT + '\nkey=' + displayVal(savedParams.key) + '&txnid=' + displayVal(savedParams.txnid) + '&amount=' + displayVal(savedParams.amount) + '&hash=' + (hashHex || '<compute first>')}
          </pre>
        ) : (
          <p style={mutedTextStyle}>Save parameters and generate hash to preview the POST body.</p>
        )}
        {wizardProgress().checkoutOpened ? (
          <p style={okTextStyle}>Checkout opened. Continue to review the return payload.</p>
        ) : (
          <p style={hintTextStyle}>Continue is locked until you open PayU test checkout.</p>
        )}
      </div>
    );
  }

  function renderReturn() {
    return (
      <div>
        <p style={bodyTextStyle}>
          PayU POSTs to <strong>surl</strong> or <strong>furl</strong> with fields like these on success. Example values are shown below — <strong>key</strong> and <strong>hash</strong> stay blank for you to fill after payment.
        </p>
        <div style={btnRowSpacedStyle}>
          <button type="button" onClick={labClick(loadReturnFromSaved)} style={btnSecondaryStyle}>Load from saved parameters</button>
          <button type="button" onClick={labClick(loadVerifyFromReturn)} style={btnSecondaryStyle}>Copy to verify step</button>
        </div>
        <div style={fieldStackStyle}>
          {RETURN_FIELDS.map(function (row) {
            return (
              <div key={row.k}>
                <FieldLabel name={row.l} required={row.k !== 'hash'} />
                <input
                  type="text"
                  value={returnPreview[row.k] || ''}
                  onChange={updateReturnField(row.k)}
                  style={inputStyle}
                  placeholder={placeholderFor(row.k)}
                />
              </div>
            );
          })}
        </div>
        <div style={btnRowStyle}>
          <button type="button" onClick={labClick(markReturnReviewed)} style={btnPrimaryStyle}>Mark return step complete</button>
        </div>
        {wizardProgress().returnReviewed ? (
          <p style={okTextStyle}>Return step complete. Continue to verify the reverse hash.</p>
        ) : (
          <p style={hintTextStyle}>Continue is locked until you mark this step complete.</p>
        )}
        <p style={bodyTextStyle}>
          On failure, set <strong>status</strong> to <code>failure</code> — PayU POSTs to <strong>furl</strong>. Always verify the hash before updating orders.
        </p>
      </div>
    );
  }

  function renderVerify() {
    return (
      <div>
        <p style={bodyTextStyle}>
          Recompute the reverse hash on your server and compare to the <strong>hash</strong> from the callback. Example txn/amount fields are pre-filled — enter your <strong>salt</strong>, paste the response <strong>hash</strong>, and click Verify.
        </p>
        <div style={btnRowSpacedStyle}>
          <button type="button" onClick={labClick(loadVerifyFromReturn)} style={btnSecondaryStyle}>Load from return preview</button>
        </div>
        <code style={codeBlockStyle}>
          sha512(SALT|status||||||udf5|udf4|udf3|udf2|udf1|email|firstname|productinfo|amount|txnid|key)
        </code>
        <div style={fieldStackStyle}>
          <div>
            <FieldLabel name="salt" required={true} />
            <input
              type="password"
              value={salt}
              onChange={function (e) { setSalt(e.target.value); }}
              style={inputStyle}
              autoComplete="off"
              placeholder={placeholderFor('salt')}
            />
          </div>
          {REV_FIELDS.map(function (row) {
            var req = row.k.indexOf('udf') !== 0;
            return (
              <div key={row.k}>
                <FieldLabel name={row.l} required={req} />
                <input
                  type="text"
                  value={revResp[row.k] || ''}
                  onChange={updateRevField(row.k)}
                  style={inputStyle}
                  placeholder={placeholderFor(row.k)}
                />
              </div>
            );
          })}
          <div>
            <FieldLabel name="hash (from callback)" required={true} />
            <input
              type="text"
              value={responseHash}
              onChange={function (e) { setResponseHash(e.target.value); }}
              style={inputStyle}
              placeholder={placeholderFor('hash')}
            />
          </div>
        </div>
        <div style={btnRowStyle}>
          <button type="button" onClick={labClick(verifyReverseHash)} disabled={busy} style={btnPrimaryStyle}>
            {busy ? 'Verifying…' : 'Verify reverse hash'}
          </button>
          <a href={HASH_TOOL} target="_blank" rel="noopener noreferrer" style={linkBtnStyle}>Open Hash Verification Tool</a>
        </div>
        {match === true ? <p style={okTextStyle}>Hash matches — response verified for this lab.</p> : null}
        {match === false ? <p style={errTextStyle}>Hash mismatch — check salt, status, and field order.</p> : null}
        {computedRev ? <code style={codeBlockStyle}>{computedRev}</code> : null}
        {!wizardProgress().verifyDone ? (
          <p style={hintTextStyle}>Continue is locked until reverse hash verification succeeds.</p>
        ) : null}
      </div>
    );
  }

  function renderTest() {
    return (
      <div>
        <p style={bodyTextStyle}>
          Before go-live, confirm each item below in your sandbox environment. Check all boxes to continue.
        </p>
        <div style={{ display: 'grid', gap: '8px' }}>
          {TEST_CHECKS.map(function (label, idx) {
            return (
              <label key={label} style={cardCheckStyle}>
                <input
                  type="checkbox"
                  checked={!!testDone[idx]}
                  onChange={function (e) {
                    var n = Object.assign({}, testDone);
                    n[idx] = e.target.checked;
                    setTestDone(n);
                  }}
                />
                <span>{label}</span>
              </label>
            );
          })}
        </div>
      </div>
    );
  }

  function renderGolive() {
    return (
      <div>
        <p style={bodyTextStyle}>
          When sandbox testing is complete, use this checklist before routing production traffic. Production endpoint: <code style={{ fontSize: '12px' }}>{PROD_PAYMENT}</code>
        </p>
        <div style={{ display: 'grid', gap: '8px', marginBottom: '16px' }}>
          {GO_LIVE.map(function (label, idx) {
            return (
              <label key={label} style={checklistLabelStyle}>
                <input
                  type="checkbox"
                  checked={!!goDone[idx]}
                  onChange={function (e) {
                    var n = Object.assign({}, goDone);
                    n[idx] = e.target.checked;
                    setGoDone(n);
                  }}
                  style={{ marginTop: '3px' }}
                />
                <span>{label}</span>
              </label>
            );
          })}
        </div>
        <button type="button" onClick={labClick(restartLab)} style={btnPrimaryStyle}>Restart wizard</button>
      </div>
    );
  }

  return (
    <SandboxAccordion
      title="Try out · Hosted Checkout integration wizard"
      intro="Nine-step guided lab. Example payment fields are pre-filled; you enter test key and salt, generate hash, open checkout, and verify. Each step unlocks only after you complete the required action."
      ariaLabel="PayU Hosted Checkout integration wizard"
    >
      <div className={shellClass} role="region" aria-label="PayU Hosted Checkout integration wizard">
        <div className="mb-1 text-xs font-semibold uppercase tracking-wider text-teal-700 dark:text-teal-400">
          PayU Hosted Checkout · Integration lab
        </div>
        <h2 style={{ margin: '0 0 6px', fontSize: '1.35rem', fontWeight: 700, color: '#111827' }}>{current.title}</h2>
        <p style={{ margin: '0 0 14px', fontSize: '13px', color: '#6b7280' }}>
          Step {step + 1} of {STEPS.length} · {current.subtitle}
        </p>

        <div style={progressTrackStyle}>
          <div style={{ height: '100%', width: pct + '%', background: PAYU_GREEN, transition: 'width 0.25s ease' }} />
        </div>

        {step >= 2 && step <= 6 ? <LabInteractiveBanner /> : null}

        <LabStatus items={[
          { label: 'Parameters saved', ok: prog.paramsSaved },
          { label: 'Hash generated', ok: prog.hashGenerated },
          { label: 'Checkout opened', ok: prog.checkoutOpened },
          { label: 'Response verified', ok: prog.verifyDone && match === true }
        ]} />

        {err ? <p style={errTextStyle} role="alert">{err}</p> : null}
        {msg ? <p style={okTextStyle}>{msg}</p> : null}
        {copyMsg ? <p style={okTextStyle}>{copyMsg}</p> : null}

        {step === 0 ? renderOverview() : null}
        {step === 1 ? renderPrereq() : null}
        {step === 2 ? renderParams() : null}
        {step === 3 ? renderHash() : null}
        {step === 4 ? renderPost() : null}
        {step === 5 ? renderReturn() : null}
        {step === 6 ? renderVerify() : null}
        {step === 7 ? renderTest() : null}
        {step === 8 ? renderGolive() : null}

        <div style={navRowStyle}>
          {step > 0 ? (
            <button type="button" onClick={labClick(function () { goToStep(step - 1); })} style={btnSecondaryStyle}>Back</button>
          ) : (
            <span />
          )}
          {step < STEPS.length - 1 ? (
            <button
              type="button"
              disabled={!canAdvance()}
              onClick={labClick(function () {
                if (!canAdvance()) return;
                goToStep(step + 1);
              })}
              style={Object.assign({}, btnPrimaryStyle, { opacity: canAdvance() ? 1 : 0.45 })}
            >
              Continue
            </button>
          ) : null}
        </div>
        {continueHint() ? <p style={hintTextStyle}>{continueHint()}</p> : null}

        <p style={mutedTextStyle}>Wizard session only · fields and progress are not shared with other page widgets</p>
      </div>
    </SandboxAccordion>
  );
};
