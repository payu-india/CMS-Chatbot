function empty(s) {
  return s == null ? '' : String(s).trim();
}

function isBlank(s) {
  return empty(s) === '';
}

var MANDATORY_FIELD_KEYS = ['txnid', 'amount', 'productinfo', 'firstname', 'email', 'phone', 'surl', 'furl'];

var WIZARD_STEP_IDS = [

function defaultProgress() {
  return {
    paramsSaved: false,
    pipeShown: false,
    hashGenerated: false,
    checkoutOpened: false,
    returnReviewed: false,
    verifyDone: false
  };
}

function mergeParams(params, defaults) {
  var merged = Object.assign({}, defaults || {});
  var p = params || {};
  var k;
  for (k in merged) {
    if (!Object.prototype.hasOwnProperty.call(merged, k)) continue;
    if (k === 'key') {
      merged.key = isBlank(p.key) ? '' : String(p.key);
      continue;
    }
    if (!isBlank(p[k])) merged[k] = String(p[k]);
  }
  return merged;
}

function validateFieldErrors(data) {
  var errors = {};
  var p = data || {};
  if (isBlank(p.key)) errors.key = 'Enter your test merchant key from the PayU dashboard.';
  var i;
  for (i = 0; i < MANDATORY_FIELD_KEYS.length; i += 1) {
    var fk = MANDATORY_FIELD_KEYS[i];
    if (isBlank(p[fk])) errors[fk] = 'Required: ' + fk;
  }
  if (!isBlank(p.amount) && !/^\d+(\.\d{1,2})?$/.test(empty(p.amount))) {
    errors.amount = 'Amount must be a decimal string (e.g. 10.00).';
  }
  if (!isBlank(p.email) && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(empty(p.email))) {
    errors.email = 'Enter a valid email address.';
  }
  if (!isBlank(p.phone) && !/^\d{10}$/.test(empty(p.phone).replace(/\s/g, ''))) {
    errors.phone = 'Phone must be exactly 10 digits.';
  }
  if (!isBlank(p.txnid) && empty(p.txnid).length > 20) {
    errors.txnid = 'txnid must be 20 characters or fewer.';
  }
  return errors;
}

function buildForwardHashString(params, saltVal) {
  var p = params;
  return (
    empty(p.key) + '|' + empty(p.txnid) + '|' + empty(p.amount) + '|' + empty(p.productinfo) + '|' +
    empty(p.firstname) + '|' + empty(p.email) + '|' + empty(p.udf1) + '|' + empty(p.udf2) + '|' +
    empty(p.udf3) + '|' + empty(p.udf4) + '|' + empty(p.udf5) + '||||||' + empty(saltVal)
  );
}

function buildReverseHashString(saltVal, resp) {
  return (
    empty(saltVal) + '|' + empty(resp.status) + '||||||' + empty(resp.udf5) + '|' + empty(resp.udf4) + '|' +
    empty(resp.udf3) + '|' + empty(resp.udf2) + '|' + empty(resp.udf1) + '|' + empty(resp.email) + '|' +
    empty(resp.firstname) + '|' + empty(resp.productinfo) + '|' + empty(resp.amount) + '|' +
    empty(resp.txnid) + '|' + empty(resp.key)
  );
}

function diffHexStrings(computed, pasted) {
  var c = empty(computed).toLowerCase();
  var p = empty(pasted).toLowerCase();
  if (!/^[0-9a-f]{128}$/.test(p)) {
    return { kind: 'invalid_pasted', pastedLen: p.length };
  }
  if (c.length !== p.length) {
    return { kind: 'length_mismatch', pastedLen: p.length, computedLen: c.length };
  }
  for (var i = 0; i < c.length; i += 1) {
    if (c[i] !== p[i]) return { kind: 'mismatch', index: i, computed: c, pasted: p };
  }
  return { kind: 'match' };
}

function getProgressBlockers(step, progress, extras) {
  var p = progress || defaultProgress();
  var x = extras || {};
  var blockers = [];
  if (step === 1 && !x.prereqDone) blockers.push('PREREQ_INCOMPLETE');
  if (step === 2) {
    if (!p.paramsSaved) blockers.push('PARAMS_NOT_SAVED');
    if (!x.labAck) blockers.push('LAB_ACK_REQUIRED');
    if (x.fieldErrorCount > 0) blockers.push('FIELD_VALIDATION');
  }
  if (step === 3) {
    if (!p.paramsSaved) blockers.push('PARAMS_NOT_SAVED');
    if (isBlank(x.salt)) blockers.push('SALT_REQUIRED');
    if (!p.pipeShown) blockers.push('PIPE_NOT_SHOWN');
    if (!p.hashGenerated || !x.hashHex || x.hashHex.length !== 128) blockers.push('HASH_NOT_GENERATED');
  }
  if (step === 4 && !p.checkoutOpened) blockers.push('CHECKOUT_NOT_OPENED');
  if (step === 5 && !p.returnReviewed) blockers.push('RETURN_NOT_REVIEWED');
  if (step === 6) {
    if (!p.verifyDone || x.match !== true) blockers.push('VERIFY_NOT_DONE');
  }
  if (step === 7 && !x.testChecklistDone) blockers.push('TEST_CHECKLIST_INCOMPLETE');
  return blockers;
}

function computeCanAdvance(step, progress, extras) {
  return getProgressBlockers(step, progress, extras).length === 0;
}

function maskEmail(email) {
  var e = engineRef.empty(email);
  if (!e || e.indexOf('@') < 0) return e ? '[REDACTED]' : '';
  var parts = e.split('@');
  var local = parts[0];
  var domain = parts[1];
  var maskedLocal = local.length <= 2 ? '**' : local.charAt(0) + '***';
  return maskedLocal + '@' + domain;
}

function maskPhone(phone) {
  var p = engineRef.empty(phone).replace(/\s/g, '');
  if (p.length < 4) return p ? '[REDACTED]' : '';
  return '******' + p.slice(-4);
}

function maskHashTail(hex) {
  var h = engineRef.empty(hex);
  if (!h) return '';
  if (h.length <= 8) return '[REDACTED]';
  return '[REDACTED…' + h.slice(-4) + ']';
}

function redactParams(params, options) {
  var opts = options || {};
  var includeSecrets = !!opts.includeSecrets;
  var p = params || {};
  var out = {
    key: includeSecrets ? p.key : (engineRef.isBlank(p.key) ? '' : '[REDACTED]'),
    txnid: p.txnid || '',
    amount: p.amount || '',
    productinfo: p.productinfo || '',
    firstname: p.firstname || '',
    email: includeSecrets ? (p.email || '') : maskEmail(p.email),
    phone: includeSecrets ? (p.phone || '') : maskPhone(p.phone),
    surl: p.surl || '',
    furl: p.furl || '',
    udf1: p.udf1 || '',
    udf2: p.udf2 || '',
    udf3: p.udf3 || '',
    udf4: p.udf4 || '',
    udf5: p.udf5 || ''
  };
  return out;
}

function resolveLastErrorCode(input) {
  if (input.lastErrorCode) return input.lastErrorCode;
  if (input.hashDiff && input.hashDiff.kind === 'invalid_pasted') return ERROR_CODES.INVALID_PASTED_HASH;
  if (input.hashDiff && input.hashDiff.kind === 'mismatch') return ERROR_CODES.HASH_MISMATCH;
  if (input.fieldErrors && Object.keys(input.fieldErrors).length) return ERROR_CODES.FIELD_VALIDATION;
  if (input.blockers && input.blockers.length) return input.blockers[0];
  return null;
}

function exportLabContext(input) {
  var step = typeof input.step === 'number' ? input.step : 0;
  var stepId = engineRef.WIZARD_STEP_IDS[step] || 'welcome';
  var progress = input.progress || engineRef.defaultProgress();
  var extras = {
    prereqDone: !!input.prereqDone,
    labAck: !!input.labAck,
    fieldErrorCount: input.fieldErrors ? Object.keys(input.fieldErrors).length : 0,
    salt: input.salt,
    hashHex: input.hashHex,
    match: input.match,
    testChecklistDone: !!input.testChecklistDone
  };
  var blockers = engineRef.getProgressBlockers(step, progress, extras);
  var canAdvance = blockers.length === 0;

  return {
    schemaVersion: LAB_CONTEXT_SCHEMA_VERSION,
    exportedAt: Date.now(),
    step: step,
    stepId: stepId,
    stepLabel: input.stepLabel || String(step),
    progress: Object.assign({}, progress),
    canAdvance: canAdvance,
    blockers: blockers,
    params: redactParams(input.params, { includeSecrets: !!input.includeSecrets }),
    fieldErrors: input.fieldErrors || {},
    hashStage: input.hashStage || 0,
    hashDiff: input.hashDiff || null,
    hashTrimNote: !!input.hashTrimNote,
    match: input.match,
    lastErrorCode: resolveLastErrorCode({
      lastErrorCode: input.lastErrorCode,
      hashDiff: input.hashDiff,
      fieldErrors: input.fieldErrors,
      blockers: blockers
    }),
    lastUserMessage: input.err || input.msg || ''
  };
}

function explainHashMismatch(hashDiff, hashTrimNote) {
  if (!hashDiff || hashDiff.kind === 'match') return null;
  if (hashDiff.kind === 'invalid_pasted') {
    return {
      title: 'Invalid callback hash',
      body: 'The pasted hash is not a valid SHA-512 value (expected 128 hex characters, got ' + hashDiff.pastedLen + '). Copy the full **hash** field from PayU\'s POST to your surl/furl — no quotes or line breaks.',
      severity: 'error'
    };
  }
  if (hashDiff.kind === 'length_mismatch') {
    return {
      title: 'Hash length mismatch',
      body: 'Computed hash has ' + hashDiff.computedLen + ' characters but pasted has ' + hashDiff.pastedLen + '. You may have truncated the value when copying from logs or browser devtools.',
      severity: 'error'
    };
  }
  var idx = hashDiff.index;
  var hint = 'Compare salt, **status**, and field order against PayU docs.';
  if (idx < 16) {
    hint = 'Early mismatch (character ' + (idx + 1) + ') often means **wrong salt** or **wrong merchant key** in the reverse hash string.';
  } else if (idx > 100) {
    hint = 'Late mismatch (character ' + (idx + 1) + ') often means a **paste error** or truncated hash — re-copy from the callback POST body.';
  } else {
    hint = 'Mismatch at character ' + (idx + 1) + ' — verify **status**, **udf** field order, and that txnid/amount match the payment.';
  }
  if (hashTrimNote) {
    hint += ' Leading/trailing whitespace was ignored when comparing.';
  }
  return {
    title: 'Hash mismatch at character ' + (idx + 1),
    body: hint,
    severity: 'error'
  };
}

function getCopilotInsights(context) {
  var ctx = context || {};
  var insights = [];
  var stepId = ctx.stepId || 'welcome';

  insights.push({
    id: 'whats-next',
    title: 'What to do next',
    body: STEP_NEXT_ACTIONS[stepId] || 'Follow the current step in the Lab tab.',
    severity: 'info'
  });

  if (!ctx.canAdvance && ctx.blockers && ctx.blockers.length) {
    var i;
    for (i = 0; i < ctx.blockers.length; i += 1) {
      var code = ctx.blockers[i];
      insights.push({
        id: 'blocker-' + code,
        title: 'Continue is locked',
        body: BLOCKER_MESSAGES[code] || code,
        severity: 'warning'
      });
    }
  }

  if (ctx.fieldErrors && Object.keys(ctx.fieldErrors).length) {
    var keys = Object.keys(ctx.fieldErrors);
    var lines = keys.map(function (k) { return '**' + k + '**: ' + ctx.fieldErrors[k]; });
    insights.push({
      id: 'field-errors',
      title: 'Field validation',
      body: lines.join(' '),
      severity: 'error'
    });
  }

  var hashInsight = explainHashMismatch(ctx.hashDiff, ctx.hashTrimNote);
  if (hashInsight) insights.push(Object.assign({ id: 'hash-diff' }, hashInsight));

  if (ctx.progress && ctx.progress.hashGenerated && stepId === 'hash') {
    insights.push({
      id: 'hash-server-reminder',
      title: 'Production reminder',
      body: 'On your server, compute the forward hash with the same pipe order — never expose **salt** in client-side code.',
      severity: 'info'
    });
  }

  if (stepId === 'verify' && ctx.match === true) {
    insights.push({
      id: 'verify-success',
      title: 'Response verified',
      body: 'Hash matched for this lab session. In production, always verify before updating order status — never trust the callback alone.',
      severity: 'success'
    });
  }

  return insights;
}

  function sha512Hex(msg) {
    var enc = new TextEncoder();
    return crypto.subtle.digest('SHA-512', enc.encode(msg)).then(function (buf) {
      return Array.from(new Uint8Array(buf)).map(function (b) { return b.toString(16).padStart(2, '0'); }).join('');
    });
  }

  function diffHexStrings(computed, pasted) {
    var c = empty(computed).toLowerCase();
    var p = empty(pasted).toLowerCase();
    if (!/^[0-9a-f]{128}$/.test(p)) {
      return { kind: 'invalid_pasted', pastedLen: p.length };
    }
    if (c.length !== p.length) return { kind: 'length_mismatch', pastedLen: p.length, computedLen: c.length };
    for (var i = 0; i < c.length; i += 1) {
      if (c[i] !== p[i]) return { kind: 'mismatch', index: i, computed: c, pasted: p };
    }
    return { kind: 'match' };
  }

      if (e.key === 'Escape') {

      var focusables = getFocusableElements(modalPanelRef.current);

      var last = focusables[focusables.length - 1];
      if (e.shiftKey) {
        if (document.activeElement === first || !modalPanelRef.current.contains(document.activeElement)) {
          e.preventDefault();
          last.focus();
        }
      } else if (document.activeElement === last || !modalPanelRef.current.contains(document.activeElement)) {
        e.preventDefault();
        first.focus();
      }
    }

    lockDocumentBodyScroll();
    document.addEventListener('keydown', onKey);
    focusRaf = requestAnimationFrame(function () {
      if (closeBtnRef.current) closeBtnRef.current.focus();

    return function () {
      cancelAnimationFrame(focusRaf);

      setBusy(false);

  function renderHexSegment(str, highlightIndex) {
    if (!str) return null;
    if (highlightIndex < 0 || highlightIndex >= str.length) return <span>{str}</span>;
    return (
      <span>
        {str.slice(0, highlightIndex)}
        <span style={hashDiffMismatchStyle}>{str.charAt(highlightIndex)}</span>
        {str.slice(highlightIndex + 1)}
      </span>
    );
  }

  function renderHashDiffDiagnosis() {
    if (!hashDiff || hashDiff.kind === 'match') return null;
    if (hashDiff.kind === 'invalid_pasted') {
      return (
        <div style={infoBoxStyle} role="alert">
          <p style={{ margin: 0, fontSize: '14px', lineHeight: 1.6 }}>
            Pasted hash is not a valid SHA-512 — expected 128 hex characters, got {hashDiff.pastedLen}.
          </p>
        </div>
      );
    }
    if (hashDiff.kind === 'length_mismatch') {
      return (
        <div style={infoBoxStyle} role="alert">
          <p style={{ margin: 0, fontSize: '14px', lineHeight: 1.6 }}>
            Length mismatch — computed {hashDiff.computedLen} characters, pasted {hashDiff.pastedLen}.
          </p>
        </div>
      );
    }
    return (
      <div style={infoBoxStyle} role="alert">
        {hashTrimNote ? (
          <p style={{ margin: '0 0 8px', fontSize: '13px', color: '#b45309' }}>
            Leading/trailing whitespace was ignored when comparing hashes.
          </p>
        ) : null}
        <p style={{ margin: '0 0 8px', fontSize: '14px', fontWeight: 600 }}>First difference at character {hashDiff.index + 1}:</p>
        <p style={{ margin: '0 0 4px', fontSize: '12px', fontWeight: 700 }}>Computed</p>
        <code style={codeBlockStyle}>{renderHexSegment(hashDiff.computed, hashDiff.index)}</code>
        <p style={{ margin: '0 0 4px', fontSize: '12px', fontWeight: 700 }}>Pasted</p>
        <code style={codeBlockStyle}>{renderHexSegment(hashDiff.pasted, hashDiff.index)}</code>
      </div>
    );
  }

                    setGoDone(n);
                  }}
                  style={checkboxInputStyle}
                />
                <span>{label}</span>
              </label>
            );
          })}
        </div>
        <button type="button" onClick={labClick(restartLab)} style={btnPrimaryStyle}>Restart wizard</button>
      </div>
    );
  }

  return (
    <div className="min-w-0" style={{ fontFamily: 'system-ui, -apple-system, Segoe UI, Roboto, sans-serif' }}>
      <div style={launcherCardStyle} role="region" aria-label="PayU Hosted Checkout integration wizard launcher">
        <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'flex-start', gap: '16px', justifyContent: 'space-between' }}>
          <div style={{ flex: '1 1 280px', minWidth: 0 }}>
            <div style={{ fontSize: '11px', fontWeight: 700, textTransform: 'uppercase', letterSpacing: '0.08em', color: PAYU_GREEN, marginBottom: '6px' }}>
              Interactive lab
            </div>
            <h3 style={{ margin: '0 0 8px', fontSize: '1.25rem', fontWeight: 700, color: '#111827' }}>
              Hosted Checkout integration wizard
            </h3>
            <p style={{ margin: '0 0 12px', fontSize: '14px', lineHeight: 1.6, color: '#4b5563' }}>
              Run a full test payment on PayU sandbox — build a request, generate a hash, pay, and verify the callback. Opens in a focused popup so you can follow along without leaving the docs.
            </p>
            <ul style={{ margin: 0, paddingLeft: '18px', fontSize: '13px', lineHeight: 1.7, color: '#374151' }}>
              <li>Example fields pre-filled — you enter test key and salt</li>
              <li>Built-in Assistant tab for step guidance and error diagnosis</li>
              <li>Real SHA-512 and live test checkout</li>
              <li>~10 minutes · test credentials only</li>
            </ul>
          </div>
          <div style={{ flexShrink: 0, alignSelf: 'center' }}>
            <button type="button" ref={launcherBtnRef} onClick={labClick(openModal)} style={Object.assign({}, btnPrimaryStyle, { padding: '12px 24px', fontSize: '15px' })}>
              Open integration wizard
            </button>
          </div>
        </div>
      </div>

      {modalOpen ? (
        <div style={modalOverlayStyle} onClick={onOverlayClick} role="dialog" aria-modal="true" aria-label="PayU Hosted Checkout integration wizard">
          <div ref={modalPanelRef} style={modalPanelStyle} onClick={labStopBubble}>
            <div style={modalHeaderStyle}>
              <div style={{ display: 'flex', alignItems: 'flex-start', justifyContent: 'space-between', gap: '12px' }}>
                <div style={{ minWidth: 0 }}>
                  <div style={{ fontSize: '11px', fontWeight: 600, textTransform: 'uppercase', letterSpacing: '0.1em', opacity: 0.85, marginBottom: '4px' }}>
                    PayU Hosted Checkout
                  </div>
                  <h2 style={{ margin: 0, fontSize: '1.2rem', fontWeight: 700, lineHeight: 1.3 }}>{current.title}</h2>
                  {step > 0 ? (
                    <p style={{ margin: '6px 0 0', fontSize: '13px', opacity: 0.9 }}>Step {current.stepLabel} of 8</p>
                  ) : null}
                </div>
                <button
                  type="button"
                  ref={closeBtnRef}
                  onClick={labClick(closeModal)}
                  aria-label="Close wizard"
                  style={{ flexShrink: 0, cursor: 'pointer', border: 'none', background: 'rgba(255,255,255,0.2)', color: '#fff', borderRadius: '8px', width: '44px', height: '44px', fontSize: '18px', lineHeight: 1 }}
                >
                  ×
                </button>
              </div>
              {renderStepPills()}
              {renderModalTabs()}
              <div style={{ marginTop: '12px', height: '4px', background: 'rgba(255,255,255,0.25)', borderRadius: '4px', overflow: 'hidden' }}>
                <div style={{ height: '100%', width: pct + '%', background: '#fff', transition: 'width 0.25s ease' }} />
              </div>
            </div>
            <div style={modalBodyStyle}>
              {modalTab === 'help' ? renderHelpPanel() : renderWizardBody()}
            </div>
            <div style={modalFooterStyle}>
              {modalTab === 'lab' ? renderWizardFooter() : null}
              <p style={Object.assign({}, mutedTextStyle, { marginTop: modalTab === 'lab' ? '8px' : 0, marginBottom: 0 })}>
                {modalTab === 'help' ? 'Assistant uses local session only · Esc to close' : 'Session saved in this browser · Esc to close'}
              </p>
            </div>
          </div>
        </div>
      ) : null}
    </div>
  );
};
