  function sha512Hex(msg) {
    var enc = new TextEncoder();
    return crypto.subtle.digest('SHA-512', enc.encode(msg)).then(function (buf) {
      return Array.from(new Uint8Array(buf)).map(function (b) { return b.toString(16).padStart(2, '0'); }).join('');
    });
  }

    setErr('');

  function newTxnid() {
    var id = 'tran_' + Date.now().toString(36);
    setFormValues(function (prev) {
      var next = Object.assign({}, prev);
      next.txnid = id.slice(0, 20);
      persistFormValues(next);
      formValuesRef.current = next;
      return next;
    });
    invalidateLabAfterEdit();
    setMsg('');
    setErr('');
  }

  function showPipeString() {
    setErr('');
    if (!lab || !lab.step11Done) { setErr('Save payment parameters first.'); return; }
    if (isBlank(salt)) { setErr('Enter your test salt on this step.'); return; }
    var params = mergeParams(lab.params);
    var pipe = buildForwardHashString(params, salt);
    setPipeString(pipe);
    setHashStage(1);
    setMsg('Review the pre-hash string below, then compute SHA-512.');
  }

  function generateHash() {
    setErr('');
    if (!lab || !lab.step11Done) { setErr('Save payment parameters first.'); return; }
    if (isBlank(salt)) { setErr('Salt is required.'); return; }
    var params = mergeParams(lab.params);
    var pipe = buildForwardHashString(params, salt);
    setBusy(true);
    sha512Hex(pipe).then(function (h) {
      var labData = labEnsure();
      labData.salt = salt;
      labData.pipeString = pipe;
      labData.hashHex = h;
      labData.step12Done = true;
      labWrite(labData);
      setLab(labData);
      setPipeString(pipe);
      setHashHex(h);
      setHashStage(2);
      setMsg('Hash generated. Continue to open PayU test checkout.');
      setBusy(false);
      setErr('');
    }).catch(function () {
      setErr('SHA-512 unavailable in this browser context.');
      setBusy(false);
    });
  }

  function paymentReady(data) {
    return data && data.step11Done && data.step12Done && data.hashHex && data.hashHex.length === 128;
  }

  function syncPayForm(data) {
    var form = payFormRef.current;
    if (!form) return false;
    var p = mergeParams(data.params);
    var fields = {
      key: p.key, txnid: p.txnid, amount: p.amount, productinfo: p.productinfo,
      firstname: p.firstname, email: p.email, phone: p.phone, surl: p.surl, furl: p.furl,
      udf1: p.udf1 || '', udf2: p.udf2 || '', udf3: p.udf3 || '', udf4: p.udf4 || '', udf5: p.udf5 || '',
      hash: data.hashHex
    };
    var name;
    for (name in fields) {
      if (!Object.prototype.hasOwnProperty.call(fields, name)) continue;
      var existing = form.elements[name];
      if (!existing) {
        existing = document.createElement('input');
        existing.type = 'hidden';
        existing.name = name;
        form.appendChild(existing);
      }
      existing.value = fields[name];
    }
    return true;
  }

  function submitPayuCheckout(target) {
    var data = labEnsure();
    if (!paymentReady(data)) { setErr('Save parameters and generate hash first.'); setMsg(''); return; }
    if (!syncPayForm(data)) { setErr('Could not prepare checkout form. Refresh and try again.'); return; }
    payFormRef.current.target = target || '_blank';
    payFormRef.current.submit();
    data.step13Launched = true;
    labWrite(data);
    setLab(data);
    setErr('');
    setMsg('Test checkout launched. Complete payment on PayU or close the tab and continue.');
  }

  function buildCurl(data) {
    var p = mergeParams(data.params);
    return 'curl -X POST "' + TEST_PAYMENT + '" \n' +
      '  -H "Content-Type: application/x-www-form-urlencoded" \n' +
      '  -d "key=' + p.key + '" \n' +
      '  -d "txnid=' + p.txnid + '" \n' +
      '  -d "amount=' + p.amount + '" \n' +
      '  -d "productinfo=' + p.productinfo + '" \n' +
      '  -d "firstname=' + p.firstname + '" \n' +
      '  -d "email=' + p.email + '" \n' +
      '  -d "phone=' + p.phone + '" \n' +
      '  -d "surl=' + p.surl + '" \n' +
      '  -d "furl=' + p.furl + '" \n' +
      '  -d "hash=' + data.hashHex + '"';
  }

  function updateRevField(k) {
    return function (e) {
      var v = e.target.value;
      setRevResp(function (prev) {
        var next = Object.assign({}, prev);
        next[k] = v;
        return next;
      });
    };
  }

  function verifyReverseHash() {
    if (isBlank(salt)) { setErr('Enter test salt (same as hash step).'); return; }
    if (isBlank(responseHash)) { setErr('Paste the hash from PayU response.'); return; }
    setBusy(true);
    setErr('');
    setMatch(null);
    setComputedRev('');
    sha512Hex(buildReverseHashString(salt, revResp)).then(function (h) {
      setComputedRev(h);
      var ok = h.toLowerCase() === empty(responseHash).toLowerCase();
      setMatch(ok);
      if (ok) {
        var labData = labEnsure();
        labData.step14Done = true;
        labWrite(labData);
        setLab(labData);
        setMsg('Reverse hash verified. On your server, recompute and compare case-sensitively before updating orders.');
      } else {
        setMsg('');
        setErr('Hash mismatch — check field order, salt, and status.');
      }
      setBusy(false);
    }).catch(function () { setErr('SHA-512 unavailable.'); setBusy(false); });
  }

  function restartLab() {
    var next = Object.assign({}, DEFAULT_PARAMS);
    setMerchantKey('');
    merchantKeyRef.current = '';
    setFormValues(next);
    persistFormValues(next);
    formValuesRef.current = next;
    setSalt('');
    setHashHex('');
    setPipeString('');
    setHashStage(0);
    setPrereq({});
    setLabAck(false);
    setTestDone({});
    setGoDone({});
    setResponseHash('');
    setComputedRev('');
    setMatch(null);
    setErr('');
    setMsg('');
    var labData = labEnsure();
    labData.params = next;
    labData.salt = '';
    labData.hashHex = '';
    labData.pipeString = '';
    labData.step11Done = false;
    labData.step12Done = false;
    labData.step13Launched = false;
    labData.step14Done = false;
    labWrite(labData);
    setLab(labData);
    goToStep(0);
  }

  var current = STEPS[step];
