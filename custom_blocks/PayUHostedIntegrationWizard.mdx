  function sha512Hex(msg) {
    var enc = new TextEncoder();
    return crypto.subtle.digest('SHA-512', enc.encode(msg)).then(function (buf) {
      return Array.from(new Uint8Array(buf))
        .map(function (b) { return b.toString(16).padStart(2, '0'); })
        .join('');
    });
  }

    return false;

  function canAdvance() {
    if (step === 0) return true;
    if (step === 1) return lab && lab.step11Done;
    if (step === 2) return lab && lab.step12Done;
    if (step === 3) return lab && lab.step13Launched;
    if (step === 4) return match === true || (lab && lab.step14Done);
    if (step === 5) return stepDone(5);
    return true;
  }

  function fieldValue(k) {
    if (k === 'key') return merchantKey;
    return formValues[k] == null ? '' : formValues[k];
  }

  function updateField(k) {
    return function (e) {
      var v = e.target.value;
      setFormValues(function (prev) {
        var next = Object.assign({}, prev);
        next[k] = v;
        persistFormValues(next);
        formValuesRef.current = next;
        return next;
      });
      setSavedFlagsOff();
    };
  }

  function updateMerchantKey(e) {
    var v = e.target.value;
    setMerchantKey(v);
    merchantKeyRef.current = v;
    setFormValues(function (prev) {
      var next = Object.assign({}, prev);
      next.key = v;
      persistFormValues(next);
      formValuesRef.current = next;
      return next;
    });
    setSavedFlagsOff();
  }

  function setSavedFlagsOff() {
    var data = labEnsure();
    data.step11Done = false;
    data.step12Done = false;
    data.hashHex = '';
    data.pipeString = '';
    labWrite(data);
    setLab(data);
    setHashHex('');
    setPipeString('');
  }

  function readMerchantKeyValue() {
    if (!isBlank(merchantKey)) return merchantKey;
    if (!isBlank(merchantKeyRef.current)) return merchantKeyRef.current;
    return readFieldValue('key');
  }

  function readFieldValue(k) {
    var v = formValuesRef.current[k];
    if (!isBlank(v)) return v;
    v = formValues[k];
    return isBlank(v) ? '' : v;
  }

  function buildParamsFromState() {
    var data = Object.assign({}, DEFAULT_PARAMS);
    data.key = readMerchantKeyValue();
    var i;
    for (i = 0; i < MANDATORY.length; i += 1) {
      data[MANDATORY[i].k] = readFieldValue(MANDATORY[i].k);
    }
    for (i = 0; i < OPTIONAL.length; i += 1) {
      data[OPTIONAL[i].k] = readFieldValue(OPTIONAL[i].k);
    }
    return data;
  }

  function validateParams(data) {
    if (isBlank(data.key)) return 'Enter your test merchant key from the PayU dashboard.';
    for (var i = 0; i < MANDATORY.length; i += 1) {
      if (isBlank(data[MANDATORY[i].k])) {
        return 'Missing required field: ' + MANDATORY[i].l;
      }
    }
    if (!/^\d+(\.\d{1,2})?$/.test(empty(data.amount))) {
      return 'Amount must be a decimal string (e.g. 10.00).';
    }
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(empty(data.email))) {
      return 'Enter a valid email address.';
    }
    return '';
  }

  function saveParams() {
    var data = buildParamsFromState();
    persistFormValues(data);
    formValuesRef.current = data;
    var ve = validateParams(data);
    if (ve) { setErr(ve); setMsg(''); return; }
    var labData = labEnsure();
    labData.params = data;
    labData.step11Done = true;
    labData.hashHex = '';
    labData.pipeString = '';
    labData.step12Done = false;
    if (!labWrite(labData)) {
      setErr('Could not save lab state. Allow storage for this site and try again.');
      return;
    }
    setLab(labData);
    setMsg('Parameters saved. Continue to generate the request hash.');
    setErr('');
  }

  function newTxnid() {
    var id = 'tran_' + Date.now().toString(36);
    var txnid = id.slice(0, 20);
    setFormValues(function (prev) {
      var next = Object.assign({}, prev);
      next.txnid = txnid;
      persistFormValues(next);
      formValuesRef.current = next;
      return next;
    });
    setSavedFlagsOff();
    setMsg('');
    setErr('');
  }

  function resetExamples() {
    var next = Object.assign({}, DEFAULT_PARAMS);
    setMerchantKey('');
    merchantKeyRef.current = '';
    setFormValues(next);
    persistFormValues(next);
    formValuesRef.current = next;
    setSalt('');
    setHashHex('');
    setPipeString('');
    setErr('');
    setMsg('');
    var labData = labEnsure();
    labData.params = next;
    labData.salt = '';
    labData.hashHex = '';
    labData.pipeString = '';
    labData.step11Done = false;
    labData.step12Done = false;
    labData.step13Launched = false;
    labData.step14Done = false;
    labWrite(labData);
    setLab(labData);
  }

  function showPipeString() {
    setErr('');
    if (!lab || !lab.step11Done) {
      setErr('Save payment parameters first.');
      return;
    }
    if (isBlank(salt)) {
      setErr('Enter your test salt (dashboard → API keys). Never use production salt in the browser.');
      return;
    }
    var params = mergeParams(lab.params);
    var pipe = buildForwardHashString(params, salt);
    setPipeString(pipe);
    setMsg('Pre-hash string shown below. Review field order, then compute SHA-512.');
    setErr('');
  }

  function generateHash() {
    setErr('');
    if (!lab || !lab.step11Done) {
      setErr('Save payment parameters first.');
      return;
    }
    if (isBlank(salt)) {
      setErr('Enter test salt before generating hash.');
      return;
    }
    var params = mergeParams(lab.params);
    var pipe = buildForwardHashString(params, salt);
    setBusy(true);
    sha512Hex(pipe)
      .then(function (h) {
        var labData = labEnsure();
        labData.salt = salt;
        labData.pipeString = pipe;
        labData.hashHex = h;
        labData.step12Done = true;
        labWrite(labData);
        setLab(labData);
        setPipeString(pipe);
        setHashHex(h);
        setMsg('Hash generated. Continue to pay on test checkout.');
        setBusy(false);
        setErr('');
      })
      .catch(function () {
        setErr('SHA-512 unavailable in this browser context.');
        setBusy(false);
      });
  }

  function paymentReady(data) {
    if (!data) return false;
    return data.step11Done && data.step12Done && data.hashHex && data.hashHex.length === 128;
  }

  function syncPayForm(data) {
    var form = payFormRef.current;
    if (!form) return false;
    var p = mergeParams(data.params);
    var fields = {
      key: p.key,
      txnid: p.txnid,
      amount: p.amount,
      productinfo: p.productinfo,
      firstname: p.firstname,
      email: p.email,
      phone: p.phone,
      surl: p.surl,
      furl: p.furl,
      udf1: p.udf1 || '',
      udf2: p.udf2 || '',
      udf3: p.udf3 || '',
      udf4: p.udf4 || '',
      udf5: p.udf5 || '',
      hash: data.hashHex
    };
    var name;
    for (name in fields) {
      if (!Object.prototype.hasOwnProperty.call(fields, name)) continue;
      var existing = form.elements[name];
      if (!existing) {
        existing = document.createElement('input');
        existing.type = 'hidden';
        existing.name = name;
        form.appendChild(existing);
      }
      existing.value = fields[name];
    }
    return true;
  }

  function submitPayuCheckout(target) {
    var data = labEnsure();
    if (!paymentReady(data)) {
      setErr('Save parameters and generate hash first.');
      setMsg('');
      return;
    }
    if (!syncPayForm(data)) {
      setErr('Could not prepare checkout form. Refresh and try again.');
      return;
    }
    payFormRef.current.target = target || '_blank';
    payFormRef.current.submit();
    data.step13Launched = true;
    labWrite(data);
    setLab(data);
    setErr('');
    setMsg(target === '_top' ? 'Redirecting to PayU test checkout…' : 'Opening PayU test checkout in a new tab…');
  }

  function buildCurl(data) {
    var p = mergeParams(data.params);
    return 'curl -X POST "' + TEST_PAYMENT + '" \n' +
      '  -H "Content-Type: application/x-www-form-urlencoded" \n' +
      '  -d "key=' + p.key + '" \n' +
      '  -d "txnid=' + p.txnid + '" \n' +
      '  -d "amount=' + p.amount + '" \n' +
      '  -d "productinfo=' + p.productinfo + '" \n' +
      '  -d "firstname=' + p.firstname + '" \n' +
      '  -d "email=' + p.email + '" \n' +
      '  -d "phone=' + p.phone + '" \n' +
      '  -d "surl=' + p.surl + '" \n' +
      '  -d "furl=' + p.furl + '" \n' +
      '  -d "hash=' + data.hashHex + '"';
  }

  function updateRevField(k) {
    return function (e) {
      var v = e.target.value;
      setRevResp(function (prev) {
        var next = Object.assign({}, prev);
        next[k] = v;
        return next;
      });
    };
  }

  function verifyReverseHash() {
    if (isBlank(salt)) {
      setErr('Enter test salt (same as hash step).');
      return;
    }
    if (isBlank(responseHash)) {
      setErr('Paste the hash from PayU callback (surl/furl POST).');
      return;
    }
    setBusy(true);
    setErr('');
    setMatch(null);
    setComputedRev('');
    var chain = buildReverseHashString(salt, revResp);
    sha512Hex(chain)
      .then(function (h) {
        setComputedRev(h);
        var ok = h.toLowerCase() === empty(responseHash).toLowerCase();
        setMatch(ok);
        if (ok) {
          var labData = labEnsure();
          labData.step14Done = true;
          labWrite(labData);
          setLab(labData);
          setMsg('Reverse hash verified. Continue to the checklist.');
        } else {
          setMsg('');
          setErr('Hash mismatch — check salt, status, and field order.');
        }
        setBusy(false);
      })
      .catch(function () {
        setErr('SHA-512 unavailable.');
        setBusy(false);
      });
  }

  function restartLab() {
    resetExamples();
    setResponseHash('');
    setComputedRev('');
    setMatch(null);
    setDoneMap({});
    goToStep(0);
  }

  var readyPay = paymentReady(lab);
  var amountLabel = lab && lab.params ? mergeParams(lab.params).amount : '10.00';
  var current = WIZARD_STEPS[step];
  var stepNum = step === 0 ? '0' : String(step);

  function renderStepRail() {
    return (
      <ol style={stepRailStyle} aria-label="Integration steps">
        {WIZARD_STEPS.map(function (s, idx) {
          var done = idx < step || (idx !== step && stepDone(idx));
          var active = idx === step;
          var style = active ? stepPillActive : done ? stepPillDone : stepPillBase;
          var label = idx === 0 ? 'Start' : String(idx);
          return (
            <li key={s.id}>
              <button
                type="button"
                style={style}
                disabled={!done && idx !== step}
                onClick={labClick(function () {
                  if (done || idx === step) goToStep(idx);
                })}
                aria-current={active ? 'step' : undefined}
              >
                {idx === WIZARD_STEPS.length - 1 && stepDone(5) ? '✓ Done' : label + '. ' + s.title.split(' ')[0]}
              </button>
            </li>
          );
        })}
      </ol>
    );
  }

  function renderWelcome() {
    return (
      <div>
        <LabHeader stepNum="0" title={current.title} hint={current.hint} />
        <p style={bodyTextStyle}>
          Your server builds a signed POST to PayU → customer pays on the hosted page → PayU POSTs the result to your <strong>surl</strong> or <strong>furl</strong>.
          This lab lets you try the full loop on PayU test without writing backend code first.
        </p>
        <ol style={{ margin: '0 0 16px', paddingLeft: '20px', fontSize: '14px', lineHeight: 1.7, color: '#374151' }}>
          <li>Enter test key and payment fields (simulator URLs included)</li>
          <li>Generate SHA-512 request hash</li>
          <li>Pay on <code style={{ fontSize: '12px' }}>test.payu.in</code></li>
          <li>Verify reverse hash on the callback</li>
        </ol>
        <p style={hintTextStyle} role="note">
          Test credentials only. Production hashing must run on your server — never expose production salt in the browser.
        </p>
        <button type="button" onClick={labClick(function () { goToStep(1); })} style={btnPrimaryStyle}>
          Start integration lab
        </button>
      </div>
    );
  }

  function renderParams() {
    return (
      <div>
        <LabHeader stepNum="1" title={current.title} hint={current.hint} />
        <div ref={formRootRef} style={fieldStackStyle}>
          <div>
            <FieldLabel name="key" required={true} />
            <input type="text" name="merchantKey" value={merchantKey} onChange={updateMerchantKey} style={inputStyle} placeholder="Test merchant key" />
          </div>
          {MANDATORY.map(function (row) {
            return (
              <div key={row.k}>
                <FieldLabel name={row.l} required={true} />
                <input type={row.k === 'email' ? 'email' : 'text'} name={row.k} value={fieldValue(row.k)} onChange={updateField(row.k)} style={inputStyle} />
              </div>
            );
          })}
          <button type="button" onClick={labClick(function () { setShowUdf(!showUdf); })} style={summaryStyle}>
            {showUdf ? 'Hide optional UDF fields' : 'Show optional UDF fields (udf1–udf5)'}
          </button>
          {showUdf ? (
            <div style={fieldStackInnerStyle}>
              {OPTIONAL.map(function (row) {
                return (
                  <div key={row.k}>
                    <FieldLabel name={row.l} required={false} />
                    <input type="text" name={row.k} value={fieldValue(row.k)} onChange={updateField(row.k)} style={inputStyle} />
                  </div>
                );
              })}
            </div>
          ) : null}
        </div>
        <div style={btnRowStyle}>
          <button type="button" onClick={labClick(saveParams)} style={btnPrimaryStyle}>Save parameters</button>
          <button type="button" onClick={labClick(newTxnid)} style={btnSecondaryStyle}>New txnid</button>
          <button type="button" onClick={labClick(resetExamples)} style={btnSecondaryStyle}>Reset examples</button>
        </div>
      </div>
    );
  }

  function renderHash() {
    if (!lab || !lab.step11Done) {
      return (
        <div>
          <LabHeader stepNum="2" title={current.title} hint={current.hint} />
          <p style={errTextStyle}>Save payment parameters in Step 1 first.</p>
          <button type="button" onClick={labClick(function () { goToStep(1); })} style={btnSecondaryStyle}>Go to Step 1</button>
        </div>
      );
    }
    var params = mergeParams(lab.params);
    return (
      <div>
        <LabHeader stepNum="2" title={current.title} hint={current.hint} />
        <p style={bodyTextStyle}>
          Forward hash formula: <code className="text-xs">sha512(key|txnid|amount|…|udf5||||||SALT)</code>
        </p>
        <div style={fieldStackStyle}>
          <div>
            <FieldLabel name="salt" required={true} />
            <input
              type="password"
              value={salt}
              onChange={function (e) {
                setSalt(e.target.value);
                setHashHex('');
                setPipeString('');
                var labData = labEnsure();
                labData.step12Done = false;
                labData.hashHex = '';
                labWrite(labData);
                setLab(labData);
              }}
              style={inputStyle}
              autoComplete="off"
              placeholder="Test salt from dashboard"
            />
          </div>
        </div>
        <div style={btnRowSpacedStyle}>
          <button type="button" onClick={labClick(showPipeString)} style={btnSecondaryStyle}>Show pre-hash string</button>
          <button type="button" onClick={labClick(generateHash)} disabled={busy} style={btnPrimaryStyle}>
            {busy ? 'Computing…' : 'Generate SHA-512 hash'}
          </button>
          {hashHex ? (
            <button type="button" onClick={labClick(function () { copyText(hashHex, setCopyMsg); })} style={btnSecondaryStyle}>Copy hash</button>
          ) : null}
        </div>
        {pipeString ? <code style={codeBlockPreStyle}>{pipeString}</code> : null}
        {hashHex ? <code style={codeBlockStyle}>{hashHex}</code> : null}
        <a href={HASH_TOOL} target="_blank" rel="noopener noreferrer" style={linkBtnStyle}>Open Hash Verification Tool</a>
      </div>
    );
  }

  function renderPay() {
    return (
      <div>
        <LabHeader stepNum="3" title={current.title} hint={current.hint} />
        <form ref={payFormRef} method="post" action={TEST_PAYMENT} style={{ display: 'none' }} aria-hidden="true" />
        <div style={btnRowSpacedStyle}>
          <button type="button" onClick={labClick(function () { submitPayuCheckout('_blank'); })} disabled={!readyPay} style={btnPrimaryStyle}>
            {'Pay ₹' + amountLabel + ' on PayU Test Checkout'}
          </button>
          <button type="button" onClick={labClick(function () { submitPayuCheckout('_top'); })} disabled={!readyPay} style={btnSecondaryStyle}>
            Open in this tab
          </button>
          <button type="button" onClick={labClick(function () { setShowCurl(!showCurl); })} disabled={!readyPay} style={btnSecondaryStyle}>
            {showCurl ? 'Hide curl' : 'Show curl'}
          </button>
          {readyPay ? (
            <button type="button" onClick={labClick(function () { copyText(buildCurl(lab), setCopyMsg); })} style={btnSecondaryStyle}>Copy curl</button>
          ) : null}
        </div>
        {showCurl && readyPay ? <code style={codeBlockPreStyle}>{buildCurl(lab)}</code> : null}
        <p style={mutedTextStyle} role="note">
          {'Test card: 5123456789012346 · OTP 123456 · UPI: anything' + '@' + 'payu'}
        </p>
        <p style={bodyTextStyle}>
          Default <strong>surl</strong> / <strong>furl</strong> point to the PayU simulator — no backend needed for this try-out.
        </p>
      </div>
    );
  }

  function renderVerify() {
    return (
      <div>
        <LabHeader stepNum="4" title={current.title} hint={current.hint} />
        <p style={bodyTextStyle}>
          After payment, PayU POSTs fields to your callback URL. Recompute the reverse hash on your server before updating order status.
        </p>
        <div style={fieldStackStyle}>
          {REV_FIELDS.map(function (row) {
            var req = row.k.indexOf('udf') !== 0;
            return (
              <div key={row.k}>
                <FieldLabel name={row.l} required={req} />
                <input type="text" value={revResp[row.k] || ''} onChange={updateRevField(row.k)} style={inputStyle} />
              </div>
            );
          })}
          <div>
            <FieldLabel name="hash (from callback)" required={true} />
            <input
              type="text"
              value={responseHash}
              onChange={function (e) { setResponseHash(e.target.value); }}
              style={inputStyle}
              placeholder="Paste hash from surl/furl POST"
            />
          </div>
        </div>
        <div style={btnRowStyle}>
          <button type="button" onClick={labClick(verifyReverseHash)} disabled={busy} style={btnPrimaryStyle}>
            {busy ? 'Verifying…' : 'Verify reverse hash'}
          </button>
          <a href={HASH_TOOL} target="_blank" rel="noopener noreferrer" style={linkBtnStyle}>Open Hash Verification Tool</a>
        </div>
        {match === true ? <p style={okTextStyle}>Hash matches — response verified for this lab.</p> : null}
        {computedRev ? <code style={codeBlockStyle}>{computedRev}</code> : null}
      </div>
    );
  }

  function renderDone() {
    return (
      <div>
        <LabHeader stepNum="5" title={current.title} hint={current.hint} />
        <div style={{ display: 'flex', flexDirection: 'column', gap: '10px' }}>
          {DONE_CHECKS.map(function (label, idx) {
            return (
              <label key={label} style={checklistLabelStyle}>
                <input
                  type="checkbox"
                  checked={!!doneMap[idx]}
                  onChange={function (e) {
                    var n = Object.assign({}, doneMap);
                    n[idx] = e.target.checked;
                    setDoneMap(n);
                  }}
                  style={{ marginTop: '3px' }}
                />
                <span>{label}</span>
              </label>
            );
          })}
        </div>
        <div style={btnRowStyle}>
          <button type="button" onClick={labClick(restartLab)} style={btnSecondaryStyle}>Restart lab</button>
        </div>
        <p style={mutedTextStyle}>
          Going live: switch endpoint to <code>https://secure.payu.in/_payment</code>, use production key/salt on server only, and monitor surl/furl.
        </p>
      </div>
    );
  }

  return (
    <SandboxAccordion
      title="Try out · Hosted Checkout integration lab"
      intro="Walk through a real test payment: save parameters, generate hash, pay on PayU sandbox, and verify the callback. Matches the Integration Guide steps 1.1–1.5."
      ariaLabel="PayU Hosted Checkout integration wizard"
    >
      <div className={shellClass} role="region" aria-label="PayU Hosted Checkout integration wizard">
        <LabInteractiveBanner />
        <LabStatus items={[
          { label: 'Parameters saved', ok: lab && lab.step11Done },
          { label: 'Hash ready', ok: lab && lab.step12Done },
          { label: 'Test checkout opened', ok: lab && lab.step13Launched },
          { label: 'Response verified', ok: lab && (lab.step14Done || match === true) }
        ]} />
        {renderStepRail()}
        {err ? <p style={errTextStyle} role="alert">{err}</p> : null}
        {msg ? <p style={okTextStyle}>{msg}</p> : null}
        {copyMsg ? <p style={okTextStyle}>{copyMsg}</p> : null}
        {step === 0 ? renderWelcome() : null}
        {step === 1 ? renderParams() : null}
        {step === 2 ? renderHash() : null}
        {step === 3 ? renderPay() : null}
        {step === 4 ? renderVerify() : null}
        {step === 5 ? renderDone() : null}
        {step > 0 && step < WIZARD_STEPS.length - 1 ? (
          <div style={btnRowStyle}>
            <button type="button" onClick={labClick(function () { goToStep(step - 1); })} style={btnSecondaryStyle}>Back</button>
            <button
              type="button"
              disabled={!canAdvance()}
              onClick={labClick(function () {
                if (!canAdvance()) return;
                goToStep(step + 1);
              })}
              style={Object.assign({}, btnPrimaryStyle, { opacity: canAdvance() ? 1 : 0.5 })}
            >
              Continue
            </button>
          </div>
        ) : null}
        <p style={mutedTextStyle}>Shared lab storage · works with Step 1.1–1.5 widgets on this page</p>
      </div>
    </SandboxAccordion>
  );
};
