// PayUHostedCheckoutSandboxStep11 - Sandbox dashed box + accordion + lab (single paste)
// Regenerate: node scripts/build-payu-hosted-checkout-steps.mjs

import { useState, useEffect, useRef } from 'react';

export const PayUHostedCheckoutSandboxStep11 = function PayUHostedCheckoutSandboxStep11(props) {

  var LAB_CONFIG = JSON.parse('{"storageKey":"payu-hosted-checkout-lab-v1","testPaymentUrl":"https://test.payu.in/_payment","simulatorUrl":"https://test-payment-middleware.payu.in/simulatorResponse","hashVerifyToolUrl":"https://payu-hashverificationtool.onrender.com/","defaults":{"key":"","txnid":"tran_1234","amount":"10.00","productinfo":"iPhone","firstname":"Aarav","email":"aarav@example.com","phone":"1234567890","surl":"https://test-payment-middleware.payu.in/simulatorResponse","furl":"https://test-payment-middleware.payu.in/simulatorResponse","udf1":"","udf2":"","udf3":"","udf4":"","udf5":""},"sampleReverseResponse":{"status":"success","key":"","txnid":"tran_1234","amount":"10.00","productinfo":"iPhone","firstname":"Aarav","email":"aarav@example.com","udf1":"","udf2":"","udf3":"","udf4":"","udf5":""}}');
  var LAB_KEY = LAB_CONFIG.storageKey;
  var TEST_PAYMENT = LAB_CONFIG.testPaymentUrl;
  var DEFAULT_PARAMS = LAB_CONFIG.defaults;

  var PAYU_GREEN = '#0f766e';
  var PAYU_GREEN_LIGHT = '#ecfdf5';
  var RED_REQ = '#dc2626';

  var shellClass = 'min-w-0';
  var fieldStackStyle = { display: 'flex', flexDirection: 'column', gap: '16px' };
  var fieldStackInnerStyle = { display: 'flex', flexDirection: 'column', gap: '16px', marginTop: '16px' };
  var btnRowStyle = { display: 'flex', flexWrap: 'wrap', gap: '8px', marginTop: '20px', alignItems: 'center' };
  var labelStyle = { display: 'block', marginBottom: '8px', fontSize: '14px', fontWeight: 700, color: PAYU_GREEN };
  var reqStarStyle = { color: RED_REQ, fontWeight: 700 };
  var hintClass = 'mb-4 text-sm leading-relaxed text-gray-600 dark:text-gray-300';
  var noteClass = 'mb-4 text-sm leading-relaxed text-amber-800 dark:text-amber-200';
  var inputStyle = {
    boxSizing: 'border-box', width: '100%', minWidth: 0, borderRadius: '6px',
    border: '1px solid #d1d5db', background: '#fff', padding: '8px 12px',
    fontSize: '14px', color: '#111827'
  };
  var btnPrimaryStyle = {
    cursor: 'pointer', borderRadius: '6px', border: 'none', background: PAYU_GREEN,
    color: '#fff', padding: '8px 16px', fontSize: '14px', fontWeight: 600
  };
  var btnSecondaryStyle = {
    cursor: 'pointer', borderRadius: '6px', border: '1px solid ' + PAYU_GREEN,
    background: PAYU_GREEN_LIGHT, color: PAYU_GREEN, padding: '8px 16px',
    fontSize: '14px', fontWeight: 600
  };
  var summaryStyle = {
    cursor: 'pointer', borderRadius: '6px', border: '1px solid ' + PAYU_GREEN,
    background: PAYU_GREEN_LIGHT, color: PAYU_GREEN, padding: '8px 16px',
    fontSize: '14px', fontWeight: 600, listStyle: 'none'
  };
  var linkBtnStyle = {
    cursor: 'pointer', borderRadius: '6px', border: '1px solid ' + PAYU_GREEN,
    background: PAYU_GREEN_LIGHT, color: PAYU_GREEN, padding: '8px 16px',
    fontSize: '14px', fontWeight: 600, textDecoration: 'none', display: 'inline-flex', alignItems: 'center'
  };
  var checklistLabelStyle = {
    display: 'flex', cursor: 'pointer', alignItems: 'flex-start', gap: '8px',
    fontSize: '14px', fontWeight: 600, lineHeight: 1.5, color: PAYU_GREEN
  };
  var codeBlockClass =
    'block break-all rounded bg-gray-100 px-3 py-2 text-xs font-mono text-gray-800 dark:bg-gray-800/50 dark:text-gray-100';
  var linkClass =
    'cursor-pointer border-0 bg-transparent p-0 text-sm font-medium text-teal-700 underline hover:text-teal-800 dark:text-teal-400';

  function empty(s) {
    return s == null ? '' : String(s).trim();
  }

  function labRead() {
    try {
      if (typeof window === 'undefined' || !window.sessionStorage) return null;
      var raw = window.sessionStorage.getItem(LAB_KEY);
      if (!raw) return null;
      return JSON.parse(raw);
    } catch (e) {
      return null;
    }
  }

  function labWrite(data) {
    try {
      if (typeof window === 'undefined' || !window.sessionStorage) return false;
      data.updatedAt = Date.now();
      window.sessionStorage.setItem(LAB_KEY, JSON.stringify(data));
      return true;
    } catch (e2) {
      return false;
    }
  }

  function labEnsure() {
    var existing = labRead();
    if (existing && existing.params) return existing;
    return { params: Object.assign({}, DEFAULT_PARAMS), salt: '', hashHex: '', pipeString: '', step11Done: false, step12Done: false };
  }

  function buildForwardHashString(params, salt) {
    var p = params;
    return (
      empty(p.key) + '|' + empty(p.txnid) + '|' + empty(p.amount) + '|' + empty(p.productinfo) + '|' +
      empty(p.firstname) + '|' + empty(p.email) + '|' + empty(p.udf1) + '|' + empty(p.udf2) + '|' +
      empty(p.udf3) + '|' + empty(p.udf4) + '|' + empty(p.udf5) + '||||||' + empty(salt)
    );
  }

  function buildReverseHashString(salt, resp) {
    return (
      empty(salt) + '|' + empty(resp.status) + '||||||' + empty(resp.udf5) + '|' + empty(resp.udf4) + '|' +
      empty(resp.udf3) + '|' + empty(resp.udf2) + '|' + empty(resp.udf1) + '|' + empty(resp.email) + '|' +
      empty(resp.firstname) + '|' + empty(resp.productinfo) + '|' + empty(resp.amount) + '|' +
      empty(resp.txnid) + '|' + empty(resp.key)
    );
  }

  function sha512Hex(msg) {
    var enc = new TextEncoder();
    return crypto.subtle.digest('SHA-512', enc.encode(msg)).then(function (buf) {
      return Array.from(new Uint8Array(buf))
        .map(function (b) { return b.toString(16).padStart(2, '0'); })
        .join('');
    });
  }

  function escapeAttr(s) {
    return String(s)
      .replace(/&/g, '&amp;')
      .replace(/"/g, '&quot;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;');
  }

  function labStopBubble(e) {
    if (!e) return;
    if (e.stopPropagation) e.stopPropagation();
    if (e.nativeEvent && e.nativeEvent.stopImmediatePropagation) {
      e.nativeEvent.stopImmediatePropagation();
    }
  }

  function labStopEvent(e) {
    labStopBubble(e);
    if (e.preventDefault) e.preventDefault();
  }

  function labClick(fn) {
    return function (e) {
      labStopEvent(e);
      fn(e);
    };
  }

  function FieldLabel(props) {
    return (
      <label style={labelStyle}>
        {props.name}
        {props.required ? <span style={reqStarStyle}> *</span> : null}
      </label>
    );
  }

  function copyText(text, setCopyMsg) {
    var value = String(text || '');
    if (!value) return;
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(value).then(function () {
        setCopyMsg('Copied.');
        setTimeout(function () { setCopyMsg(''); }, 2000);
      });
    }
  }

  function LabInteractiveBanner() {
    return (
      <div
        className="mb-4 flex items-start gap-3 rounded-md border border-teal-200 bg-white/90 px-3 py-2.5 dark:border-teal-800 dark:bg-gray-900/50"
        role="note"
      >
        <span className="mt-0.5 shrink-0 rounded bg-teal-700 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wider text-white dark:bg-teal-600">
          Try it
        </span>
        <p className="mb-0 min-w-0 text-sm leading-snug text-gray-700 dark:text-gray-200">
          <span className="font-semibold text-gray-900 dark:text-gray-100">Interactive lab — </span>
          edit the fields below and use the buttons. This is a live widget, not static text. Values stay in your browser for the next step.
        </p>
      </div>
    );
  }

  function LabHeader(props) {
    return (
      <div className="mb-3 pb-1">
        <div className="mb-1 text-xs font-semibold uppercase tracking-wider text-teal-700 dark:text-teal-400">
          Integration lab · Step {props.step}
        </div>
        <p className="text-sm font-medium text-gray-900 dark:text-gray-100">{props.title}</p>
        {props.hint ? <p className="mt-1 text-sm text-gray-600 dark:text-gray-300">{props.hint}</p> : null}
      </div>
    );
  }

  function LabStatus(props) {
    var items = props.items || [];
    return (
      <div className="mb-4 flex flex-wrap gap-2 text-xs">
        {items.map(function (it, idx) {
          var ok = it.ok;
          return (
            <span
              key={idx}
              className={
                ok
                  ? 'rounded-full bg-teal-50 px-2 py-1 text-teal-800 dark:bg-teal-950 dark:text-teal-200'
                  : 'rounded-full bg-gray-100 px-2 py-1 text-gray-600 dark:bg-gray-800 dark:text-gray-300'
              }
            >
              {ok ? '✓ ' : '○ '}{it.label}
            </span>
          );
        })}
      </div>
    );
  }

  var sandboxOuterStyle = {
    margin: '24px 0', width: '100%', minWidth: 0, maxWidth: '100%', boxSizing: 'border-box',
    overflow: 'hidden', borderRadius: '12px', border: '2px dashed ' + PAYU_GREEN,
    background: 'rgba(240, 253, 250, 0.4)'
  };
  var sandboxSummaryStyle = {
    display: 'flex', alignItems: 'center', gap: '12px', cursor: 'pointer',
    padding: '12px 16px', listStyle: 'none', background: 'transparent', border: 'none'
  };
  var sandboxBadgeStyle = {
    flexShrink: 0, borderRadius: '4px', background: PAYU_GREEN, color: '#fff',
    padding: '2px 8px', fontSize: '10px', fontWeight: 700, textTransform: 'uppercase', letterSpacing: '0.05em'
  };
  var sandboxBodyStyle = {
    background: '#fff', padding: '16px', borderTop: '1px solid rgba(15, 118, 110, 0.15)'
  };

  function sandboxSyncChevron(details) {
    if (!details) return;
    var chevron = details.querySelector('[data-sandbox-chevron]');
    if (chevron) chevron.textContent = details.open ? '▼' : '▶';
  }

  function onSandboxToggle(e) {
    var details = e.currentTarget;
    sandboxSyncChevron(details);
  }

  function onSandboxSummaryClick(e) {
    var summary = e.currentTarget;
    var details = summary.parentNode;
    if (!details || details.tagName !== 'DETAILS') return;
    labStopEvent(e);
    details.open = !details.open;
    sandboxSyncChevron(details);
  }

  function SandboxAccordion(props) {
    var detailsRef = useRef(null);
    useEffect(function () {
      var details = detailsRef.current;
      if (!details) return;
      details.open = false;
      sandboxSyncChevron(details);
    }, []);
    return (
      <details
        ref={detailsRef}
        data-sandbox-id={props.ariaLabel || props.title || 'sandbox'}
        style={sandboxOuterStyle}
        onToggle={onSandboxToggle}
        aria-label={props.ariaLabel || 'PayU integration sandbox'}
      >
        <summary style={sandboxSummaryStyle} onClick={onSandboxSummaryClick}>
          <span style={sandboxBadgeStyle}>SANDBOX</span>
          <span style={{ minWidth: 0, flex: 1, fontSize: '14px', fontWeight: 600, color: '#111827' }}>{props.title}</span>
          <span data-sandbox-chevron={true} style={{ flexShrink: 0, color: PAYU_GREEN }}>▶</span>
        </summary>
        <div
          style={sandboxBodyStyle}
          onClick={labStopBubble}
          onMouseDown={labStopBubble}
          onPointerDown={labStopBubble}
        >
          <p style={{ margin: '0 0 20px', fontSize: '14px', lineHeight: 1.6, color: '#4b5563' }}>{props.intro}</p>
          <div style={{ display: 'flex', flexDirection: 'column', gap: '16px' }}>{props.children}</div>
        </div>
      </details>
    );
  }

  var MANDATORY = JSON.parse('[{"k":"key","l":"key"},{"k":"txnid","l":"txnid"},{"k":"amount","l":"amount"},{"k":"productinfo","l":"productinfo"},{"k":"firstname","l":"firstname"},{"k":"email","l":"email","t":"email"},{"k":"phone","l":"phone"},{"k":"surl","l":"surl"},{"k":"furl","l":"furl"}]');
  var OPTIONAL = JSON.parse('[{"k":"udf1","l":"udf1"},{"k":"udf2","l":"udf2"},{"k":"udf3","l":"udf3"},{"k":"udf4","l":"udf4"},{"k":"udf5","l":"udf5"}]');
  var FORM_UI_GLOBAL = '__payuHostedCheckoutStep11Form';

  function formStore() {
    if (typeof window !== 'undefined') {
      if (!window[FORM_UI_GLOBAL]) {
        var lab = labRead();
        var base = Object.assign({}, DEFAULT_PARAMS);
        if (lab && lab.params) Object.assign(base, lab.params);
        window[FORM_UI_GLOBAL] = base;
      }
      return window[FORM_UI_GLOBAL];
    }
    return Object.assign({}, DEFAULT_PARAMS);
  }

  function collectFormFromDom() {
    var store = formStore();
    if (typeof document === 'undefined') return store;
    var fields = document.querySelectorAll('[data-payu-step11-field]');
    for (var i = 0; i < fields.length; i += 1) {
      var el = fields[i];
      var k = el.getAttribute('data-payu-step11-field');
      if (k) store[k] = el.value;
    }
    return store;
  }

  function syncFieldToDom(k, v) {
    if (typeof document === 'undefined') return;
    var el = document.querySelector('[data-payu-step11-field="' + k + '"]');
    if (el) el.value = v == null ? '' : String(v);
  }

  function syncAllFieldsToDom(values) {
    var keys = Object.keys(values || {});
    for (var i = 0; i < keys.length; i += 1) {
      syncFieldToDom(keys[i], values[keys[i]]);
    }
  }

  function onFieldInput(k) {
    return function (e) {
      formStore()[k] = e.target.value;
    };
  }

  var form = formStore();
  var _msg = useState('');
  var msg = _msg[0];
  var setMsg = _msg[1];
  var _err = useState('');
  var err = _err[0];
  var setErr = _err[1];
  var _saved = useState(false);
  var saved = _saved[0];
  var setSaved = _saved[1];
  var UDF_SHOW = 'Show optional fields (udf1, udf2, udf3, udf4, udf5)';
  var UDF_HIDE = 'Hide optional fields (udf1, udf2, udf3, udf4, udf5)';
  var UDF_UI_KEY = 'payu-hosted-checkout-step11-udf-open-v1';
  var udfBtnRef = useRef(null);
  var udfPanelRef = useRef(null);

  function syncUdfPanel() {
    try {
      var panel = udfPanelRef.current;
      var btn = udfBtnRef.current;
      if (!panel || !btn || typeof window === 'undefined' || !window.sessionStorage) return;
      var open = window.sessionStorage.getItem(UDF_UI_KEY) === '1';
      panel.style.display = open ? 'block' : 'none';
      btn.textContent = open ? UDF_HIDE : UDF_SHOW;
    } catch (e0) {}
  }

  function toggleUdfPanel() {
    try {
      var panel = udfPanelRef.current;
      var btn = udfBtnRef.current;
      if (!panel || !btn) return;
      var open = panel.style.display !== 'block';
      panel.style.display = open ? 'block' : 'none';
      btn.textContent = open ? UDF_HIDE : UDF_SHOW;
      if (typeof window !== 'undefined' && window.sessionStorage) {
        window.sessionStorage.setItem(UDF_UI_KEY, open ? '1' : '0');
      }
    } catch (e1) {}
  }

  useEffect(function () {
    var lab = labEnsure();
    if (lab.params) {
      Object.assign(formStore(), DEFAULT_PARAMS, lab.params);
      syncAllFieldsToDom(formStore());
    }
    setSaved(!!lab.step11Done);
    syncUdfPanel();
  }, []);

  function newTxnid() {
    var id = 'tran_' + Date.now().toString(36);
    var txnid = id.slice(0, 20);
    formStore().txnid = txnid;
    syncFieldToDom('txnid', txnid);
    setSaved(false);
    setErr('');
  }

  function validateParams() {
    var data = collectFormFromDom();
    for (var i = 0; i < MANDATORY.length; i += 1) {
      if (!empty(data[MANDATORY[i].k])) continue;
      return 'Missing required field: ' + MANDATORY[i].l;
    }
    if (!/^\d+(\.\d{1,2})?$/.test(empty(data.amount))) {
      return 'Amount must be a decimal string (e.g. 10.00).';
    }
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(empty(data.email))) {
      return 'Enter a valid email address.';
    }
    return '';
  }

  function saveToLab() {
    var ve = validateParams();
    if (ve) { setErr(ve); return; }
    var lab = labEnsure();
    lab.params = Object.assign({}, collectFormFromDom());
    lab.step11Done = true;
    lab.hashHex = '';
    lab.pipeString = '';
    lab.step12Done = false;
    labWrite(lab);
    setSaved(true);
    setMsg('Saved to integration lab. Continue to Step 1.2 to generate the hash.');
    setErr('');
  }

  function resetExamples() {
    Object.assign(formStore(), DEFAULT_PARAMS);
    syncAllFieldsToDom(formStore());
    setSaved(false);
    setMsg('');
    setErr('');
  }

  function renderField(row) {
    var type = row.t === 'email' ? 'email' : 'text';
    return (
      <div key={row.k}>
        <FieldLabel name={row.k} required={true} />
        <input
          type={type}
          data-payu-step11-field={row.k}
          defaultValue={form[row.k] || ''}
          onInput={onFieldInput(row.k)}
          style={inputStyle}
        />
      </div>
    );
  }

  return (
    <SandboxAccordion title="Try Out! · Add request parameter values to test" intro="This is a live tool — not documentation. Enter your test key, edit values, and click Save parameters to lab. Values stay in your browser for Step 1.2." ariaLabel="Step 1.1 sandbox">
      <div className={shellClass}>
      <div style={fieldStackStyle}>
        {MANDATORY.map(renderField)}
      <button type="button" ref={udfBtnRef} onClick={labClick(toggleUdfPanel)} style={summaryStyle}>
        {UDF_SHOW}
      </button>
      <div
        ref={udfPanelRef}
        style={Object.assign({}, fieldStackInnerStyle, { display: 'none' })}
      >
          {OPTIONAL.map(function (row) {
            return (
              <div key={row.k}>
                <FieldLabel name={row.k} required={false} />
                <input
                  type="text"
                  data-payu-step11-field={row.k}
                  defaultValue={form[row.k] || ''}
                  onInput={onFieldInput(row.k)}
                  style={inputStyle}
                />
              </div>
            );
          })}
      </div>
      </div>
      <div style={btnRowStyle}>
        <button type="button" onClick={labClick(saveToLab)} style={btnPrimaryStyle}>Save parameters to lab</button>
        <button type="button" onClick={labClick(newTxnid)} style={btnSecondaryStyle}>New txnid</button>
        <button type="button" onClick={labClick(resetExamples)} style={btnSecondaryStyle}>Reset examples</button>
      </div>
      {err ? <p className="mb-3 text-sm text-red-600" role="alert">{err}</p> : null}
      {msg ? <p className="mb-3 text-sm text-teal-700">{msg}</p> : null}
      {saved ? <p className="mt-4 text-xs leading-relaxed text-gray-500">Step 1.1 complete — parameters stored for Steps 1.2 and 1.3.</p> : null}
      </div>
    </SandboxAccordion>
  );
};

<PayUHostedCheckoutSandboxStep11 />
