  function recommend(a) {
    var primaryKey = getPrimaryKey(a);
    var primaryProduct = PRODUCTS[primaryKey];
    var recs = [];
    if (primaryProduct) {
      recs.push({
        name: primaryKey,
        score: 100,
        reason: getPrimaryReason(a, primaryKey),
        product: primaryProduct,
        isPrimary: true,
      });
    }
    var companions = getCompanions(a, primaryKey);
    for (var i = 0; i < companions.length; i += 1) {
      var c = companions[i];
      recs.push({
        name: c.key,
        score: Math.max(40, 85 - i * 8),
        reason: c.reason,
        product: PRODUCTS[c.key],
        isPrimary: false,
      });
    }
    if (!recs.length) {
      recs.push({
        name: 'PayU Hosted Checkout',
        score: 50,
        reason: 'Default starting point for most PayU merchants.',
        product: PRODUCTS['PayU Hosted Checkout'],
        isPrimary: true,
      });
    }
    return recs;
  }

  var _done = useState(false);
  var done = _done[0];
  var setDone = _done[1];

  var currentId = done ? null : history[history.length - 1];
  var q = currentId ? resolveQuestion(currentId, answers) : null;
  var stepNum = history.length;
  var estTotal = estimateSteps(answers);
  var pct = done ? 100 : Math.min(95, Math.round((stepNum / estTotal) * 100));

  function onPick(oid) {
    if (!currentId) return;
    var nextAnswers = Object.assign({}, answers);
    nextAnswers[currentId] = oid;
    var nextId = getNext(nextAnswers, currentId);
    setAnswers(nextAnswers);
    if (nextId) {
      setHistory(history.concat([nextId]));
    } else {
      setDone(true);
    }
  }

  function goBack() {
    if (done) {
      setDone(false);
      return;
    }
    if (history.length <= 1) return;
    var prev = history.slice(0, -1);
    var removedId = history[history.length - 1];
    var nextAnswers = Object.assign({}, answers);
    delete nextAnswers[removedId];
    setHistory(prev);
    setAnswers(nextAnswers);
  }

  function restart() {
    setHistory([START]);
    setAnswers({});
    setDone(false);
  }

  var recs = done ? recommend(answers) : [];
  var primary = recs[0];
  var rest = recs.slice(1);

  function complexityStyle(c) {
    if (c === 'low') return { bg: '#d1fae5', color: '#065f46' };
    if (c === 'high') return { bg: '#fee2e2', color: '#991b1b' };
    return { bg: '#fef3c7', color: '#92400e' };
  }

  return (
    <div style={shell} role="region" aria-label="PayU Integration Path Recommender">
      <div style={{ fontSize: '11px', fontWeight: 600, textTransform: 'uppercase', letterSpacing: '0.04em', color: '#00A651', marginBottom: '6px' }}>
        PayU DevEx
      </div>
      <h2 style={{ margin: '0 0 6px', fontSize: '1.25rem', fontWeight: 700, color: '#1e3a5f' }}>
        Find your PayU integration path
      </h2>
      <p style={{ margin: '0 0 16px', color: '#6b7280', fontSize: '13px' }}>
        Workflow-based recommender aligned to PayU docs. Tap to answer - each step adapts to your previous choice.
      </p>

      {!done && history.length > 1 ? (
        <div style={{ marginBottom: '10px', fontSize: '12px', color: '#00A651', fontWeight: 500 }}>
          Your path: {buildPathSummary(answers)}
        </div>
      ) : null}

      {!done ? (
        <div style={{ marginBottom: '16px' }}>
          <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: '12px', color: '#6b7280', marginBottom: '6px' }}>
            <span>Step {stepNum}{estTotal ? ' - ~' + estTotal + ' questions' : ''}</span>
            <span>{pct}%</span>
          </div>
          <div style={{ height: '6px', background: '#e5e7eb', borderRadius: '999px', overflow: 'hidden' }}>
            <div style={{ height: '100%', width: pct + '%', background: '#00A651', borderRadius: '999px', transition: 'width 0.25s ease' }} />
          </div>
        </div>
      ) : null}

      {!done && q ? (
        <div>
          <h3 style={{ margin: '0 0 4px', fontSize: '1.05rem', fontWeight: 600, color: '#1e3a5f' }}>{q.text}</h3>
          <p style={{ margin: '0 0 12px', fontSize: '12px', color: '#6b7280' }}>{q.hint}</p>
          <div style={{ display: 'grid', gap: '8px' }}>
            {q.options.map(function (opt) {
              return (
                <button key={opt.id} type="button" onClick={function () { onPick(opt.id); }} style={optionBtn}>
                  <span style={{ fontWeight: 500 }}>{opt.label}</span>
                  <span style={{ display: 'block', marginTop: '3px', fontSize: '12px', color: '#6b7280', fontWeight: 400 }}>{opt.desc}</span>
                </button>
              );
            })}
          </div>
        </div>
      ) : null}

      {done && primary && primary.product ? (
        <div>
          <div style={{ marginBottom: '16px', padding: '12px 14px', borderRadius: '10px', border: '1px solid rgba(0,166,81,.25)', background: 'rgba(0,166,81,.06)', fontSize: '13px', color: '#374151' }}>
            {explain(answers, recs)}
          </div>
          <div style={{ border: '2px solid #00A651', borderRadius: '12px', overflow: 'hidden', marginBottom: '16px' }}>
            <div style={{ background: '#00A651', color: '#fff', fontSize: '12px', fontWeight: 600, padding: '8px 14px' }}>Primary integration</div>
            <div style={{ padding: '16px' }}>
              <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start', gap: '8px' }}>
                <div>
                  <h3 style={{ margin: 0, fontSize: '1.1rem', color: '#1e3a5f' }}>{primary.name}</h3>
                  <p style={{ margin: '4px 0 0', fontSize: '12px', color: '#6b7280' }}>{primary.product.category}</p>
                </div>
                <span style={{ fontSize: '1.5rem', fontWeight: 700, color: '#00A651' }}>{primary.score}%</span>
              </div>
              <p style={{ margin: '10px 0', fontSize: '13px' }}>{primary.product.description}</p>
              <p style={{ margin: '0 0 10px', fontSize: '12px', color: '#4b5563' }}>{primary.reason}</p>
              <div style={{ display: 'flex', flexWrap: 'wrap', gap: '6px', margin: '8px 0' }}>
                <span style={{ fontSize: '11px', fontWeight: 600, padding: '3px 8px', borderRadius: '999px', background: complexityStyle(primary.product.complexity).bg, color: complexityStyle(primary.product.complexity).color, textTransform: 'capitalize' }}>
                  {primary.product.complexity} complexity
                </span>
                <span style={{ fontSize: '11px', fontWeight: 600, padding: '3px 8px', borderRadius: '999px', background: '#f3f4f6', color: '#374151' }}>
                  ~{primary.product.goLiveDays} days to go-live
                </span>
              </div>
              <a href={primary.product.docsUrl} target="_blank" rel="noopener noreferrer" style={{ display: 'inline-block', marginTop: '10px', fontSize: '13px', fontWeight: 600, color: '#00A651' }}>
                View PayU documentation
              </a>
            </div>
          </div>
          {rest.length ? (
            <div>
              <p style={{ fontWeight: 600, color: '#1e3a5f', margin: '0 0 8px' }}>Also integrate</p>
              <div style={{ display: 'grid', gap: '8px' }}>
                {rest.map(function (r) {
                  return (
                    <div key={r.name} style={{ border: '1px solid #e5e7eb', borderRadius: '10px', padding: '12px', background: '#f9fafb' }}>
                      <h4 style={{ margin: '0 0 4px', fontSize: '14px', color: '#1e3a5f' }}>
                        {r.name}
                      </h4>
                      <p style={{ margin: '0 0 6px', fontSize: '12px', color: '#6b7280' }}>{r.reason}</p>
                      <a href={r.product.docsUrl} target="_blank" rel="noopener noreferrer" style={{ fontSize: '13px', fontWeight: 600, color: '#00A651' }}>Documentation</a>
                    </div>
                  );
                })}
              </div>
            </div>
          ) : null}
        </div>
      ) : null}

      <div style={{ display: 'flex', justifyContent: 'space-between', gap: '12px', marginTop: '16px', paddingTop: '12px', borderTop: '1px solid #e5e7eb' }}>
        {history.length > 1 || done ? (
          <button type="button" onClick={goBack} style={{ padding: '8px 14px', borderRadius: '8px', fontSize: '13px', fontWeight: 600, cursor: 'pointer', background: '#fff', border: '1px solid #e5e7eb', color: '#374151' }}>
            Back
          </button>
        ) : (
          <span />
        )}
        {done ? (
          <button type="button" onClick={restart} style={{ padding: '8px 14px', borderRadius: '8px', fontSize: '13px', fontWeight: 600, cursor: 'pointer', background: '#00A651', border: 'none', color: '#fff' }}>
            Start over
          </button>
        ) : null}
      </div>
    </div>
  );
};
