  function getTotalSteps(a) {
    var steps = 0;
    var cur = START;
    var sim = Object.assign({}, a);
    while (cur) {
      steps += 1;
      if (sim[cur] === undefined) {
        steps += getTailLength(cur, sim) - 1;
        break;
      }
      var next = getNext(sim, cur);
      if (!next) break;
      cur = next;
    }
    return steps;
  }

  function recommend(a) {
    var primaryKey = getPrimaryKey(a);
    var primaryProduct = PRODUCTS[primaryKey];
    var recs = [];
    if (primaryProduct) {
      recs.push({
        name: primaryKey,
        score: 100,
        reason: getPrimaryReason(a, primaryKey),
        product: primaryProduct,
        isPrimary: true,
      });
    }
    var companions = getCompanions(a, primaryKey);
    for (var i = 0; i < companions.length; i += 1) {
      var c = companions[i];
      recs.push({
        name: c.key,
        score: Math.max(40, 85 - i * 8),
        reason: c.reason,
        product: PRODUCTS[c.key],
        isPrimary: false,
      });
    }
    if (!recs.length) {
      recs.push({
        name: 'PayU Hosted Checkout',
        score: 50,
        reason: 'Default starting point for most PayU merchants.',
        product: PRODUCTS['PayU Hosted Checkout'],
        isPrimary: true,
      });
    }
    return recs;
  }

    setHistory(nextHistory);
    setAnswers(nextAnswers);
    setDone(false);
  }

      <div
        style={{

            position: 'fixed',
            inset: 0,

            display: 'flex',

            role="dialog"
            aria-modal="true"
            aria-labelledby="payu-ipr-modal-title"

              maxWidth: '560px',
              maxHeight: '90vh',
              overflow: 'auto',
              borderRadius: '16px',
              background: '#ffffff',
              boxShadow: '0 24px 64px rgba(0,0,0,0.28)',
              fontFamily: 'system-ui, -apple-system, Segoe UI, Roboto, sans-serif',
              fontSize: '14px',
              lineHeight: 1.5,

                </h2>
              </div>
              <button
                type="button"
                onClick={closeModal}
                aria-label="Close integration path finder"
                style={{ width: '36px', height: '36px', borderRadius: '8px', border: '1px solid #e5e7eb', background: '#fff', cursor: 'pointer', fontSize: '20px', lineHeight: 1, color: '#6b7280' }}
              >
                ×
              </button>
            </div>

            </div>
          </div>
        </div>
      ) : null}
    </div>
  );
};
