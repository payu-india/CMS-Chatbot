name: SearchableRichTable
---
import { useState, useMemo, useEffect } from 'react';

export const SearchableRichTable = (props) => {
  const {
    headers           = [],
    headersJson       = '',
    rowsJson          = '',
    boldColumnsJson   = '',
    codeColumnsJson   = '',
    placeholder       = 'Search in table...',
    maxHeight         = '480px',
    rowsPerPage       = 50,
    lastColumnMaxLines = 0,
    emptyMessage      = 'No results found. Try a different search.',
    className         = '',
    children,
  } = props;

  const [searchQuery, setSearchQuery] = useState('');
  const [currentPage, setCurrentPage] = useState(1);

  const parseJson = (text, fallback) => {
    if (text == null || text === '') return fallback;
    if (typeof text !== 'string') return text;
    const t = text.trim();
    if (!t) return fallback;
    try { return JSON.parse(t); } catch (e) { return fallback; }
  };

  const toArray = (c) => {
    if (c == null) return [];
    if (Array.isArray(c)) return c;
    return [c];
  };

  const isEl = (n) => (n != null && typeof n === 'object' && n.type != null && n.props != null);

  const extractText = (node) => {
    if (node == null || node === false || node === true) return '';
    if (typeof node === 'string')  return node;
    if (typeof node === 'number')  return String(node);
    if (Array.isArray(node))       return node.map(extractText).join(' ');
    if (isEl(node)) {
      const p = node.props;
      return [
        extractText(p.children),
        p.alt    ? String(p.alt)   : '',
        p.title  ? String(p.title) : '',
        p.label  ? String(p.label) : '',
        typeof p.value === 'string' ? p.value : '',
      ].join(' ');
    }
    return '';
  };

  const normalize = (str) =>
    String(str == null ? '' : str).toLowerCase().trim().replace(/\s+/g, ' ');

  const resolvedHeaders = useMemo(() => {
    const fromJson = parseJson(headersJson, null);
    if (Array.isArray(fromJson) && fromJson.length > 0)
      return fromJson.map((h) => String(h == null ? '' : h));
    if (Array.isArray(headers) && headers.length > 0)
      return headers.map((h) => String(h == null ? '' : h));
    return [];
  }, [headers, headersJson]);

  const jsonRows = useMemo(() => {
    const parsed = parseJson(rowsJson, null);
    if (Array.isArray(parsed)) return parsed.filter((r) => Array.isArray(r));
    return null;
  }, [rowsJson]);

  const boldCols = useMemo(() => {
    const p = parseJson(boldColumnsJson, null);
    return Array.isArray(p) ? p : [];
  }, [boldColumnsJson]);

  const codeCols = useMemo(() => {
    const p = parseJson(codeColumnsJson, null);
    return Array.isArray(p) ? p : [];
  }, [codeColumnsJson]);

  const isJsonMode = jsonRows !== null;

  const allRows = useMemo(() => {
    if (isJsonMode) return [];
    const result = [];
    const collect = (nodes) => {
      toArray(nodes).forEach((child) => {
        if (!isEl(child)) return;
        const tag = typeof child.type === 'string' ? child.type.toLowerCase() : '';
        if (tag === 'tr') {
          result.push(child);
        } else if (tag === 'tbody' || tag === 'thead' || tag === 'table') {
          collect(child.props.children);
        }
      });
    };
    collect(children);
    return result;
  }, [children, isJsonMode]);

  const rowTextIndex = useMemo(
    () => allRows.map((row) => normalize(extractText(row))),
    [allRows]
  );

  const filteredJsonRows = useMemo(() => {
    if (!isJsonMode) return [];
    const q = normalize(searchQuery);
    if (!q) return jsonRows;
    const words = q.split(/\s+/).filter(Boolean);
    return jsonRows.filter((row) => {
      const text = row.map((c) => normalize(c == null ? '' : String(c))).join(' ');
      return words.every((w) => text.includes(w));
    });
  }, [jsonRows, isJsonMode, searchQuery]);

  const filteredJsxRows = useMemo(() => {
    if (isJsonMode) return [];
    const q = normalize(searchQuery);
    if (!q) return allRows;
    const words = q.split(/\s+/).filter(Boolean);
    return allRows.filter((_, i) => words.every((w) => rowTextIndex[i].includes(w)));
  }, [allRows, rowTextIndex, isJsonMode, searchQuery]);

  useEffect(() => { setCurrentPage(1); }, [searchQuery]);

  const totalCount  = isJsonMode ? jsonRows.length  : allRows.length;
  const activeCount = isJsonMode ? filteredJsonRows.length : filteredJsxRows.length;

  const perPage    = Math.max(1, Number(rowsPerPage) || 50);
  const totalPages = Math.max(1, Math.ceil(activeCount / perPage));
  const safePage   = Math.min(currentPage, totalPages);
  const pageStart  = (safePage - 1) * perPage;
  const pageRows   = isJsonMode
    ? filteredJsonRows.slice(pageStart, pageStart + perPage)
    : filteredJsxRows.slice(pageStart, pageStart + perPage);

  const goTo = (page) => {
    if (page >= 1 && page <= totalPages) setCurrentPage(page);
  };

  const pageNumbers = useMemo(() => {
    if (totalPages <= 7) return Array.from({ length: totalPages }, (_, i) => i + 1);
    const list = [];
    const add  = (v) => { if (!list.includes(v)) list.push(v); };
    add(1);
    if (safePage > 3) add('...');
    const lo = Math.max(2, safePage - 1);
    const hi = Math.min(totalPages - 1, safePage + 1);
    for (let i = lo; i <= hi; i++) add(i);
    if (safePage < totalPages - 2) add('...');
    add(totalPages);
    return list;
  }, [totalPages, safePage]);

  const hasQuery   = normalize(searchQuery).length > 0;
  const showEmpty  = hasQuery && activeCount === 0;
  const showNoRows = totalCount === 0 && !showEmpty;
  const colCount   = resolvedHeaders.length || 3;

  const font     = '-apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, "Helvetica Neue", Arial, sans-serif';
  const fontMono = 'ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, monospace';
  const border   = '#e5e7eb';
  const blue     = '#3b82f6';
  const blueBg   = '#eff6ff';

  const thStyle = {
    padding: '10px 16px',
    textAlign: 'left',
    fontFamily: font,
    fontSize: '12px',
    fontWeight: '700',
    color: '#374151',
    textTransform: 'uppercase',
    letterSpacing: '0.06em',
    backgroundColor: '#f9fafb',
    borderBottom: '1px solid ' + border,
    borderTop: 'none', borderLeft: 'none', borderRight: 'none',
    verticalAlign: 'middle',
    boxSizing: 'border-box',
    whiteSpace: 'nowrap',
  };

  const tdBase = {
    padding: '12px 16px',
    fontFamily: font,
    fontSize: '14px',
    fontWeight: '400',
    lineHeight: '1.6',
    color: '#111827',
    borderTop: 'none', borderLeft: 'none', borderRight: 'none',
    verticalAlign: 'top',
    boxSizing: 'border-box',
    wordBreak: 'break-word',
    overflowWrap: 'break-word',
  };

  const codeStyle = {
    padding: '2px 6px',
    background: '#f3f4f6',
    color: '#1f2937',
    borderRadius: '4px',
    fontFamily: fontMono,
    fontSize: '13px',
  };

  const msgStyle = {
    padding: '24px 16px',
    fontFamily: font,
    fontSize: '14px',
    color: '#6b7280',
    border: 'none',
    textAlign: 'center',
    boxSizing: 'border-box',
  };

  const btnBase = {
    display: 'inline-flex',
    alignItems: 'center',
    justifyContent: 'center',
    height: '30px',
    padding: '0 10px',
    border: '1px solid ' + border,
    borderRadius: '6px',
    backgroundColor: '#ffffff',
    color: '#374151',
    fontFamily: font,
    fontSize: '13px',
    fontWeight: '400',
    cursor: 'pointer',
    boxSizing: 'border-box',
    userSelect: 'none',
    minWidth: '30px',
  };

  const renderJsonCell = (value, ci) => {
    const s = value == null ? '' : String(value);
    if (codeCols.indexOf(ci) >= 0) return <code style={codeStyle}>{s}</code>;
    if (boldCols.indexOf(ci) >= 0) return <strong>{s}</strong>;
    return s;
  };

  const renderRows = () => {
    if (showNoRows) {
      return (
        <tr>
          <td colSpan={colCount} style={msgStyle}>
            {isJsonMode
              ? 'No rows loaded. Pass rowsJson or add <tr> children.'
              : 'Add <tr> children to SearchableRichTable. Each <td> cell supports any ReadMe component.'}
          </td>
        </tr>
      );
    }
    if (showEmpty) {
      return (
        <tr>
          <td colSpan={colCount} role="alert"
            style={Object.assign({}, msgStyle, { backgroundColor: '#fef2f2', color: '#b91c1c' })}>
            {emptyMessage}
          </td>
        </tr>
      );
    }

    const maxLines = Number(lastColumnMaxLines);
    const clampStyle = {
      display: '-webkit-box',
      WebkitLineClamp: maxLines,
      WebkitBoxOrient: 'vertical',
      overflow: 'hidden',
      wordBreak: 'break-word',
    };

    if (isJsonMode) {
      return pageRows.map((row, rowIdx) => {
        const isLast = rowIdx === pageRows.length - 1;
        const bg     = rowIdx % 2 === 0 ? '#ffffff' : '#fafafa';
        return (
          <tr key={pageStart + rowIdx} style={{ backgroundColor: bg }}>
            {row.map((cell, ci) => {
              const clampLast = maxLines > 0 && ci === row.length - 1;
              const cellStyle = Object.assign(
                {},
                tdBase,
                { borderBottom: isLast ? 'none' : '1px solid ' + border },
                clampLast ? { whiteSpace: 'normal' } : {}
              );
              const content = renderJsonCell(cell, ci);
              return (
                <td key={ci} style={cellStyle}>
                  {clampLast ? <div style={clampStyle}>{content}</div> : content}
                </td>
              );
            })}
          </tr>
        );
      });
    }

    return pageRows.map((row, rowIdx) => {
      const isLast    = rowIdx === pageRows.length - 1;
      const bg        = rowIdx % 2 === 0 ? '#ffffff' : '#fafafa';
      const cellNodes = toArray(row.props.children).filter(isEl);
      const cells     = cellNodes.map((cell, ci) => {
        const clampLast = maxLines > 0 && ci === cellNodes.length - 1;
        const cellStyle = Object.assign(
          {},
          tdBase,
          { borderBottom: isLast ? 'none' : '1px solid ' + border },
          cell.props.style || {},
          clampLast ? { whiteSpace: 'normal' } : {}
        );
        return (
          <td
            key={ci}
            colSpan={cell.props.colSpan}
            rowSpan={cell.props.rowSpan}
            style={cellStyle}
          >
            {clampLast ? (
              <div style={clampStyle}>{cell.props.children}</div>
            ) : cell.props.children}
          </td>
        );
      });
      return (
        <tr key={pageStart + rowIdx} style={{ backgroundColor: bg }}>
          {cells}
        </tr>
      );
    });
  };

  return (
    <div
      className={className}
      style={{ width: '100%', fontFamily: font, fontSize: '14px', color: '#111827' }}
    >
      <div style={{
        width: '100%',
        border: '1px solid ' + border,
        borderRadius: '8px',
        overflow: 'hidden',
        boxSizing: 'border-box',
        backgroundColor: '#ffffff',
      }}>

        <div style={{
          padding: '12px 16px',
          backgroundColor: '#f9fafb',
          borderBottom: '1px solid ' + border,
          display: 'flex',
          alignItems: 'center',
          gap: '12px',
          flexWrap: 'wrap',
        }}>
          <div style={{
            display: 'flex',
            alignItems: 'center',
            flex: '1 1 200px',
            maxWidth: '400px',
            border: '1px solid #d1d5db',
            borderRadius: '6px',
            backgroundColor: '#ffffff',
            boxShadow: '0 1px 2px rgba(0,0,0,0.05)',
          }}>
            <span aria-hidden="true" style={{ display: 'flex', alignItems: 'center', paddingLeft: '12px', color: '#9ca3af', flexShrink: 0 }}>
              <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24"
                fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
                <circle cx="11" cy="11" r="8" /><path d="m21 21-4.35-4.35" />
              </svg>
            </span>
            <input
              type="search"
              placeholder={placeholder}
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              aria-label="Search table rows"
              style={{
                flex: 1, minWidth: 0,
                padding: '8px 8px 8px 8px',
                border: 0, backgroundColor: 'transparent', outline: 'none',
                fontSize: '14px', fontFamily: font, color: '#111827',
              }}
            />
            {searchQuery ? (
              <button
                onClick={() => setSearchQuery('')}
                aria-label="Clear search"
                type="button"
                style={{ display: 'flex', alignItems: 'center', padding: '0 10px', border: 0, background: 'transparent', cursor: 'pointer', color: '#9ca3af', flexShrink: 0 }}
              >
                <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round">
                  <path d="M18 6 6 18M6 6l12 12" />
                </svg>
              </button>
            ) : null}
          </div>
          {!showNoRows ? (
            <span style={{ fontSize: '12px', color: '#6b7280', fontFamily: font, flexShrink: 0, whiteSpace: 'nowrap' }}>
              {hasQuery
                ? activeCount + ' of ' + totalCount + ' rows match'
                : totalCount + (totalCount === 1 ? ' row' : ' rows')}
            </span>
          ) : null}
        </div>

        <div style={{ overflowX: 'auto', overflowY: 'auto', maxHeight: showEmpty || showNoRows ? 'none' : maxHeight, width: '100%' }}>
          <table style={{ width: '100%', borderCollapse: 'collapse', borderSpacing: 0, margin: 0, tableLayout: 'auto', fontFamily: font }}>
            {resolvedHeaders.length > 0 ? (
              <thead>
                <tr>
                  {resolvedHeaders.map((h, i) => (
                    <th key={i} scope="col" style={thStyle}>{h}</th>
                  ))}
                </tr>
              </thead>
            ) : null}
            <tbody>
              {renderRows()}
            </tbody>
          </table>
        </div>

        {!showNoRows && !showEmpty && totalPages > 1 ? (
          <div style={{
            padding: '10px 16px',
            borderTop: '1px solid ' + border,
            backgroundColor: '#f9fafb',
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'space-between',
            flexWrap: 'wrap',
            gap: '8px',
          }}>
            <span style={{ fontSize: '12px', color: '#6b7280', fontFamily: font, whiteSpace: 'nowrap' }}>
              {(pageStart + 1) + '-' + Math.min(pageStart + perPage, activeCount) + ' of ' + activeCount + ' rows'}
            </span>
            <div style={{ display: 'flex', alignItems: 'center', gap: '4px', flexWrap: 'wrap' }}>
              <button
                onClick={() => goTo(safePage - 1)}
                disabled={safePage === 1}
                type="button"
                style={Object.assign({}, btnBase, { opacity: safePage === 1 ? 0.4 : 1, cursor: safePage === 1 ? 'default' : 'pointer' })}
              >
                Prev
              </button>
              {pageNumbers.map((p, i) => {
                if (p === '...') {
                  return (
                    <span key={i} style={{ padding: '0 4px', color: '#9ca3af', fontFamily: font, fontSize: '13px' }}>...</span>
                  );
                }
                return (
                  <button
                    key={i}
                    onClick={() => goTo(p)}
                    type="button"
                    style={Object.assign({}, btnBase, {
                      backgroundColor: p === safePage ? blueBg : '#ffffff',
                      color: p === safePage ? blue : '#374151',
                      borderColor: p === safePage ? blue : border,
                      fontWeight: p === safePage ? '600' : '400',
                    })}
                  >
                    {p}
                  </button>
                );
              })}
              <button
                onClick={() => goTo(safePage + 1)}
                disabled={safePage === totalPages}
                type="button"
                style={Object.assign({}, btnBase, { opacity: safePage === totalPages ? 0.4 : 1, cursor: safePage === totalPages ? 'default' : 'pointer' })}
              >
                Next
              </button>
            </div>
          </div>
        ) : null}

        {!showNoRows && !showEmpty && totalPages === 1 && activeCount > 0 ? (
          <div style={{ padding: '8px 16px', borderTop: '1px solid ' + border, backgroundColor: '#f9fafb' }}>
            <span style={{ fontSize: '12px', color: '#6b7280', fontFamily: font }}>
              {activeCount + ' of ' + totalCount + (totalCount === 1 ? ' row' : ' rows') + (hasQuery ? ' match' : '')}
            </span>
          </div>
        ) : null}

      </div>
    </div>
  );
};
