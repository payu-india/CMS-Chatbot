/**
 * HoverCardGrid - ReadMe CMS Custom Component
 *
 * Responsive card grid with cycling pastel backgrounds, soft floating shadows,
 * and lift on hover (Razorpay-style step cards).
 *
 * ReadMe: Settings → Custom Components → paste this file. One registration; pass `items`.
 *
 * Usage: pass columns and items. Each item: title, href, optional text, icon, image,
 * imageAlt, target, bg (optional CSS color / Tailwind bg-* class override).
 * If `image` (URL) is set, it shows above the title; otherwise `icon` (Font Awesome class) is used.
 *
 * ReadMe Refactored (MDX): use plain strings only in items — do not pass JSX in `text`.
 * See README_HOVER_CARD_GRID.md for a working 3-card example.
 *
 * Props: items, columns 1–4, className.
 *
 * Do not add helpers above `export` — ReadMe MDX parses that region and Acorn fails on `{`.
 */

export const HoverCardGrid = (props) => {
  const { items = [], columns = 3, className = '' } = props;

  const safeCols = [1, 2, 3, 4].includes(Number(columns)) ? Number(columns) : 3;
  const gridClass =
    safeCols === 1
      ? 'grid-cols-1'
      : safeCols === 2
        ? 'grid-cols-1 sm:grid-cols-2'
        : safeCols === 4
          ? 'grid-cols-1 sm:grid-cols-2 xl:grid-cols-4'
          : 'grid-cols-1 sm:grid-cols-2 lg:grid-cols-3';

  // Soft pastel fills with a light vertical wash for subtle texture (not flat solids).
  const pastelStyles = [
    {
      background:
        'linear-gradient(165deg, #eef3ff 0%, #e4ecff 48%, #dce6fb 100%)',
    },
    {
      background:
        'linear-gradient(165deg, #fff6ec 0%, #ffefdf 48%, #ffe8d4 100%)',
    },
    {
      background:
        'linear-gradient(165deg, #eefbf3 0%, #e4f7eb 48%, #daf2e3 100%)',
    },
    {
      background:
        'linear-gradient(165deg, #f5f0ff 0%, #eee6ff 48%, #e6dcfa 100%)',
    },
  ];

  // Resting float + deeper lift on hover (matches soft bottom-heavy drop shadows).
  const cardShadow =
    'shadow-[0_10px_28px_-8px_rgba(15,23,42,0.14),0_4px_10px_-4px_rgba(15,23,42,0.08)]';
  const cardShadowHover =
    'hover:shadow-[0_22px_44px_-12px_rgba(15,23,42,0.2),0_10px_20px_-8px_rgba(15,23,42,0.12)]';

  const cardBase =
    'group flex h-full flex-col rounded-xl p-5 text-left !no-underline hover:!no-underline ' +
    cardShadow +
    ' ' +
    cardShadowHover +
    ' transition-[transform,box-shadow] duration-500 ease-[cubic-bezier(0.22,1,0.36,1)] ' +
    'hover:-translate-y-1.5 ' +
    'focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-blue-500';

  const mediaTransition =
    'transition-[box-shadow,transform] duration-500 ease-[cubic-bezier(0.22,1,0.36,1)]';

  const hoverLinkUnderline =
    'transition-[text-decoration,text-decoration-color] duration-300 ' +
    'group-hover:underline group-hover:decoration-[#00b058] group-hover:underline-offset-[3px]';

  return (
    <div className={`my-6 ${className}`.trim()}>
      <ul className={`grid gap-5 list-none p-0 m-0 ${gridClass}`}>
        {items.map((item, index) => {
          const title = item?.title != null ? String(item.title) : '';
          const href = item?.href != null ? String(item.href) : '';
          const content = item?.text;
          const hasContent =
            content != null && (typeof content === 'string' ? content.trim() !== '' : true);
          const rawIcon = item?.icon != null ? String(item.icon).trim() : '';
          const strippedIcon = rawIcon ? rawIcon.replace(/^fa\s+/i, '').trim() : '';
          const icon =
            !strippedIcon ? '' : strippedIcon.indexOf('fa-') === 0 ? strippedIcon : 'fa-' + strippedIcon;
          const rawImage =
            item?.image != null
              ? String(item.image).trim()
              : item?.imageSrc != null
                ? String(item.imageSrc).trim()
                : item?.img != null
                  ? String(item.img).trim()
                  : '';
          const imageSrc = rawImage;
          const imageAlt =
            item?.imageAlt != null && String(item.imageAlt).trim() !== ''
              ? String(item.imageAlt).trim()
              : title || 'Card illustration';
          // Default to opening links in a new tab.
          // If you want same-tab navigation, pass `target="_self"` explicitly.
          const target =
            item?.target != null && String(item.target).trim() !== ''
              ? String(item.target).trim()
              : '_blank';
          const key = item?.key != null ? String(item.key) : `card-${index}-${title}`;

          const customBg = item?.bg != null ? String(item.bg).trim() : '';
          const usesTailwindBg = customBg.indexOf('bg-') === 0;
          const pastel = pastelStyles[index % pastelStyles.length];
          const cardStyle = customBg
            ? usesTailwindBg
              ? undefined
              : { background: customBg }
            : pastel;
          const cardClass =
            cardBase + (usesTailwindBg ? ' ' + customBg : '');

          const renderContentNode = () => {
            if (content == null) return null;

            if (Array.isArray(content)) {
              const listItems = content
                .filter((v) => v != null && String(v).trim() !== '')
                .map((v) => String(v).trim());
              if (listItems.length < 1) return null;
              return (
                <ul className={'list-disc pl-5 space-y-1' + (href ? ' ' + hoverLinkUnderline : '')}>
                  {listItems.map((li, liIndex) => (
                    <li key={`${liIndex}-${li}`}>{li}</li>
                  ))}
                </ul>
              );
            }

            if (typeof content === 'string') {
              const raw = content.trim();
              if (raw === '') return null;

              const lines = raw
                .split(/\\n|\r?\n/)
                .map((l) => l.trim())
                .filter(Boolean);

              const bulletRe = /^(-|\*|•)\s+/;
              const bulletLines = lines.filter((l) => bulletRe.test(l));

              const looksLikeBullets =
                bulletLines.length >= 2 &&
                bulletLines.every((l) => bulletRe.test(l)) &&
                bulletLines.length === lines.length;

              if (looksLikeBullets) {
                const listItems = bulletLines.map((l) => l.replace(bulletRe, '').trim());
                if (listItems.length < 1) return null;

                return (
                  <ul className={'list-disc pl-5 space-y-1' + (href ? ' ' + hoverLinkUnderline : '')}>
                    {listItems.map((li, liIndex) => (
                      <li key={`${liIndex}-${li}`}>{li}</li>
                    ))}
                  </ul>
                );
              }

              return <span className={href ? hoverLinkUnderline : ''}>{content}</span>;
            }

            return <>{content}</>;
          };

          const titleClass =
            'min-w-0 flex-1 text-base font-semibold leading-snug text-slate-900 !no-underline ' +
            (href ? hoverLinkUnderline : '');

          const iconBoxClass =
            'inline-flex h-10 w-10 flex-shrink-0 items-center justify-center rounded-md bg-white/75 text-[15px] text-slate-700 shadow-sm ring-1 ring-slate-900/5 ' +
            mediaTransition +
            ' group-hover:scale-105 group-hover:shadow-md';

          const inner = (
            <>
              {imageSrc ? (
                <span
                  className={`mb-3 block overflow-hidden rounded-md bg-white/60 shadow-sm ring-1 ring-slate-900/5 ${mediaTransition} group-hover:shadow-md group-hover:ring-slate-900/10`}
                >
                  <img
                    src={imageSrc}
                    alt={imageAlt}
                    className={`h-32 w-full object-cover ${mediaTransition} group-hover:scale-[1.03]`}
                    loading="lazy"
                    decoding="async"
                  />
                </span>
              ) : null}
              {imageSrc && title ? (
                <span className={'mb-2 block ' + titleClass}>{title}</span>
              ) : null}
              {!imageSrc && (icon || title) ? (
                <div className={'flex items-center gap-2 ' + (hasContent ? 'mb-2' : '')}>
                  {icon ? (
                    <span className={iconBoxClass} aria-hidden="true">
                      <i className={`fa ${icon} text-[#00b386]`} aria-hidden="true" />
                    </span>
                  ) : null}
                  {title ? <span className={titleClass}>{title}</span> : null}
                </div>
              ) : null}
              {hasContent ? (
                <div
                  className={
                    'mt-2 prose prose-sm max-w-none text-slate-600 ' +
                    (href
                      ? hoverLinkUnderline +
                        ' [&_a]:!no-underline group-hover:[&_a]:underline group-hover:[&_a]:decoration-[#00b058] group-hover:[&_a]:underline-offset-[3px]'
                      : ' [&_a]:underline [&_a]:decoration-[#00b058] [&_a]:underline-offset-[3px] hover:[&_a]:underline')
                  }
                >
                  {renderContentNode()}
                </div>
              ) : null}
            </>
          );

          return (
            <li key={key} className="m-0 list-none p-0">
              {href ? (
                <a
                  href={href}
                  target={target}
                  rel={target === '_blank' ? 'noopener noreferrer' : undefined}
                  className={cardClass}
                  style={{
                    textDecoration: 'none',
                    textDecorationLine: 'none',
                    ...(cardStyle || {}),
                  }}
                >
                  {inner}
                </a>
              ) : (
                <div className={cardClass} style={cardStyle}>
                  {inner}
                </div>
              )}
            </li>
          );
        })}
      </ul>
    </div>
  );
};
