/**
 * HoverCardGrid - ReadMe CMS Custom Component
 *
 * Responsive card grid with soft pastel backgrounds, floating shadows, and hover lift.
 * Card headers use a fixed-height row (inline styles) so titles align across cards
 * even when ReadMe global CSS fights Tailwind utilities.
 *
 * ReadMe: Settings → Custom Components → paste this file. One registration; pass `items`.
 *
 * Usage: pass columns and items. Each item: title, href, optional text, icon, image,
 * imageAlt, target, bg (optional CSS color / Tailwind bg-* class override).
 *
 * ReadMe Refactored (MDX): use plain strings only in items — do not pass JSX in `text`.
 * See README_HOVER_CARD_GRID.md for a working 3-card example.
 *
 * Props: items, columns 1–4, className.
 *
 * Do not add helpers above `export` — ReadMe MDX parses that region and Acorn fails on `{`.
 */

export const HoverCardGrid = (props) => {
  const { items = [], columns = 3, className = '' } = props;

  const safeCols = [1, 2, 3, 4].includes(Number(columns)) ? Number(columns) : 3;
  const gridClass =
    safeCols === 1
      ? 'grid-cols-1'
      : safeCols === 2
        ? 'grid-cols-1 sm:grid-cols-2'
        : safeCols === 4
          ? 'grid-cols-1 sm:grid-cols-2 xl:grid-cols-4'
          : 'grid-cols-1 sm:grid-cols-2 lg:grid-cols-3';

  // Soft pastels (muted blue / peach / mint / lavender) — not vibrant fills.
  const themes = [
    {
      background: 'linear-gradient(165deg, #eef3ff 0%, #e8effc 50%, #e3ebf8 100%)',
      icon: '#3b5bdb',
      iconBox: 'rgba(255,255,255,0.72)',
    },
    {
      background: 'linear-gradient(165deg, #fff6ec 0%, #fff2e4 50%, #ffedd9 100%)',
      icon: '#c2410c',
      iconBox: 'rgba(255,255,255,0.72)',
    },
    {
      background: 'linear-gradient(165deg, #eefbf3 0%, #e8f7ee 50%, #e3f3e9 100%)',
      icon: '#00b386',
      iconBox: 'rgba(255,255,255,0.72)',
    },
    {
      background: 'linear-gradient(165deg, #f5f0ff 0%, #f0eafc 50%, #ebe5f8 100%)',
      icon: '#5b21b6',
      iconBox: 'rgba(255,255,255,0.72)',
    },
  ];

  const cardShadow =
    'shadow-[0_10px_28px_-8px_rgba(15,23,42,0.14),0_4px_10px_-4px_rgba(15,23,42,0.08)]';
  const cardShadowHover =
    'hover:shadow-[0_22px_44px_-12px_rgba(15,23,42,0.2),0_10px_20px_-8px_rgba(15,23,42,0.12)]';

  const cardClass =
    'group rounded-xl text-left !no-underline hover:!no-underline ' +
    cardShadow +
    ' ' +
    cardShadowHover +
    ' transition-[transform,box-shadow] duration-500 ease-[cubic-bezier(0.22,1,0.36,1)] ' +
    'hover:-translate-y-1.5 ' +
    'focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-blue-500';

  const mediaTransition =
    'transition-[box-shadow,transform] duration-500 ease-[cubic-bezier(0.22,1,0.36,1)]';

  const hoverLinkUnderline =
    'transition-[text-decoration,text-decoration-color] duration-300 ' +
    'group-hover:underline group-hover:decoration-[#00b058] group-hover:underline-offset-[3px]';

  // Inline layout beats ReadMe global `a` / list styles that were shifting titles.
  const cardLayoutStyle = {
    display: 'grid',
    gridTemplateRows: '40px auto',
    alignContent: 'start',
    alignItems: 'stretch',
    justifyItems: 'stretch',
    gap: '12px',
    padding: '20px',
    height: '100%',
    width: '100%',
    boxSizing: 'border-box',
    margin: 0,
    lineHeight: 'normal',
    textDecoration: 'none',
    textDecorationLine: 'none',
  };

  const headerStyle = {
    display: 'flex',
    flexDirection: 'row',
    alignItems: 'center',
    justifyContent: 'flex-start',
    gap: '10px',
    height: '40px',
    maxHeight: '40px',
    minHeight: '40px',
    margin: 0,
    padding: 0,
    overflow: 'hidden',
    boxSizing: 'border-box',
  };

  const iconBoxStyle = {
    display: 'inline-flex',
    alignItems: 'center',
    justifyContent: 'center',
    width: '40px',
    height: '40px',
    minWidth: '40px',
    minHeight: '40px',
    maxWidth: '40px',
    maxHeight: '40px',
    margin: 0,
    padding: 0,
    borderRadius: '8px',
    overflow: 'hidden',
    lineHeight: 1,
    flexShrink: 0,
    boxSizing: 'border-box',
    boxShadow: '0 1px 2px rgba(15,23,42,0.06)',
  };

  const iconGlyphStyle = {
    display: 'block',
    lineHeight: 1,
    fontSize: '15px',
    width: '15px',
    height: '15px',
    margin: 0,
    padding: 0,
    textAlign: 'center',
  };

  const titleStyle = {
    display: 'block',
    margin: 0,
    padding: 0,
    border: 0,
    minWidth: 0,
    flex: '1 1 auto',
    fontSize: '16px',
    fontWeight: 600,
    lineHeight: '40px',
    height: '40px',
    overflow: 'hidden',
    textOverflow: 'ellipsis',
    whiteSpace: 'nowrap',
    color: '#0f172a',
    textDecoration: 'none',
  };

  const bodyStyle = {
    margin: 0,
    padding: 0,
    fontSize: '14px',
    lineHeight: '22px',
    color: '#475569',
  };

  return (
    <div className={`my-6 ${className}`.trim()}>
      <ul
        className={`m-0 grid list-none gap-5 p-0 ${gridClass}`}
        style={{ alignItems: 'stretch', margin: 0, padding: 0 }}
      >
        {items.map((item, index) => {
          const title = item?.title != null ? String(item.title) : '';
          const href = item?.href != null ? String(item.href) : '';
          const content = item?.text;
          const hasContent =
            content != null && (typeof content === 'string' ? content.trim() !== '' : true);
          const rawIcon = item?.icon != null ? String(item.icon).trim() : '';
          const strippedIcon = rawIcon ? rawIcon.replace(/^fa\s+/i, '').trim() : '';
          const icon =
            !strippedIcon ? '' : strippedIcon.indexOf('fa-') === 0 ? strippedIcon : 'fa-' + strippedIcon;
          const rawImage =
            item?.image != null
              ? String(item.image).trim()
              : item?.imageSrc != null
                ? String(item.imageSrc).trim()
                : item?.img != null
                  ? String(item.img).trim()
                  : '';
          const imageSrc = rawImage;
          const imageAlt =
            item?.imageAlt != null && String(item.imageAlt).trim() !== ''
              ? String(item.imageAlt).trim()
              : title || 'Card illustration';
          const target =
            item?.target != null && String(item.target).trim() !== ''
              ? String(item.target).trim()
              : '_blank';
          const key = item?.key != null ? String(item.key) : `card-${index}-${title}`;

          const theme = themes[index % themes.length];
          const customBg = item?.bg != null ? String(item.bg).trim() : '';
          const usesTailwindBg = customBg.indexOf('bg-') === 0;
          const background = customBg && !usesTailwindBg ? customBg : theme.background;

          const renderContentNode = () => {
            if (content == null) return null;

            if (Array.isArray(content)) {
              const listItems = content
                .filter((v) => v != null && String(v).trim() !== '')
                .map((v) => String(v).trim());
              if (listItems.length < 1) return null;
              return (
                <ul
                  className={href ? hoverLinkUnderline : ''}
                  style={{ margin: 0, paddingLeft: '20px', listStyleType: 'disc' }}
                >
                  {listItems.map((li, liIndex) => (
                    <li key={`${liIndex}-${li}`} style={{ margin: 0 }}>
                      {li}
                    </li>
                  ))}
                </ul>
              );
            }

            if (typeof content === 'string') {
              const raw = content.trim();
              if (raw === '') return null;

              const lines = raw
                .split(/\\n|\r?\n/)
                .map((l) => l.trim())
                .filter(Boolean);

              const bulletRe = /^(-|\*|•)\s+/;
              const bulletLines = lines.filter((l) => bulletRe.test(l));

              const looksLikeBullets =
                bulletLines.length >= 2 &&
                bulletLines.every((l) => bulletRe.test(l)) &&
                bulletLines.length === lines.length;

              if (looksLikeBullets) {
                const listItems = bulletLines.map((l) => l.replace(bulletRe, '').trim());
                if (listItems.length < 1) return null;

                return (
                  <ul
                    className={href ? hoverLinkUnderline : ''}
                    style={{ margin: 0, paddingLeft: '20px', listStyleType: 'disc' }}
                  >
                    {listItems.map((li, liIndex) => (
                      <li key={`${liIndex}-${li}`} style={{ margin: 0 }}>
                        {li}
                      </li>
                    ))}
                  </ul>
                );
              }

              return (
                <span className={href ? hoverLinkUnderline : ''} style={{ margin: 0, padding: 0 }}>
                  {content}
                </span>
              );
            }

            return <>{content}</>;
          };

          // Image cards: image on row 1, title+body stacked in row 2.
          // Icon cards: fixed 40px header row, body in row 2 — titles share the same Y.
          const hasHeader = Boolean(title || (!imageSrc && icon));

          const header = hasHeader ? (
            <div style={headerStyle}>
              {!imageSrc && icon ? (
                <span
                  className={mediaTransition + ' group-hover:scale-105'}
                  style={{
                    ...iconBoxStyle,
                    backgroundColor: theme.iconBox,
                    color: theme.icon,
                  }}
                  aria-hidden="true"
                >
                  <i
                    className={`fa ${icon}`}
                    style={{ ...iconGlyphStyle, color: theme.icon }}
                    aria-hidden="true"
                  />
                </span>
              ) : null}
              {title ? (
                <span
                  className={
                    '!no-underline ' + (href ? hoverLinkUnderline : '')
                  }
                  style={titleStyle}
                  title={title}
                >
                  {title}
                </span>
              ) : null}
            </div>
          ) : (
            <div style={{ height: '40px', margin: 0, padding: 0 }} aria-hidden="true" />
          );

          const body = hasContent ? (
            <div
              className={
                href
                  ? hoverLinkUnderline +
                    ' [&_a]:!no-underline group-hover:[&_a]:underline group-hover:[&_a]:decoration-[#00b058] group-hover:[&_a]:underline-offset-[3px]'
                  : '[&_a]:underline [&_a]:decoration-[#00b058] [&_a]:underline-offset-[3px]'
              }
              style={bodyStyle}
            >
              {renderContentNode()}
            </div>
          ) : (
            <div style={{ margin: 0, padding: 0 }} />
          );

          const inner = imageSrc ? (
            <>
              <span
                className={`block overflow-hidden rounded-md bg-white/60 shadow-sm ring-1 ring-slate-900/5 ${mediaTransition} group-hover:shadow-md`}
                style={{ margin: 0, height: '128px' }}
              >
                <img
                  src={imageSrc}
                  alt={imageAlt}
                  className={`${mediaTransition} group-hover:scale-[1.03]`}
                  style={{ display: 'block', width: '100%', height: '128px', objectFit: 'cover', margin: 0 }}
                  loading="lazy"
                  decoding="async"
                />
              </span>
              <div style={{ display: 'grid', gap: '12px', margin: 0, padding: 0, alignContent: 'start' }}>
                {title ? (
                  <span
                    className={'!no-underline ' + (href ? hoverLinkUnderline : '')}
                    style={{ ...titleStyle, height: 'auto', lineHeight: '24px', whiteSpace: 'normal' }}
                  >
                    {title}
                  </span>
                ) : null}
                {body}
              </div>
            </>
          ) : (
            <>
              {header}
              {body}
            </>
          );

          const mergedCardStyle = {
            ...cardLayoutStyle,
            ...(usesTailwindBg ? {} : { background }),
            ...(imageSrc
              ? {
                  gridTemplateRows: '128px auto',
                }
              : {}),
          };

          return (
            <li
              key={key}
              className="m-0 list-none p-0"
              style={{
                display: 'flex',
                margin: 0,
                padding: 0,
                listStyle: 'none',
                height: '100%',
                minHeight: 0,
              }}
            >
              {href ? (
                <a
                  href={href}
                  target={target}
                  rel={target === '_blank' ? 'noopener noreferrer' : undefined}
                  className={
                    cardClass +
                    ' w-full' +
                    (usesTailwindBg ? ' ' + customBg : '')
                  }
                  style={mergedCardStyle}
                >
                  {inner}
                </a>
              ) : (
                <div
                  className={
                    cardClass +
                    ' w-full' +
                    (usesTailwindBg ? ' ' + customBg : '')
                  }
                  style={mergedCardStyle}
                >
                  {inner}
                </div>
              )}
            </li>
          );
        })}
      </ul>
    </div>
  );
};
