import { useState, useMemo } from 'react';

export const SearchableTable = (props) => {
  const {
    headers = [],
    rows = [],
    placeholder = 'Search in table...',
    maxHeight = '400px',
    emptyMessage = 'No results found. Try a different search.',
    className = '',
    columnWidths = null,
  } = props;

  const [searchQuery, setSearchQuery] = useState('');

  const normalize = (str) =>
    str == null ? '' : String(str).toLowerCase().trim().replace(/\s+/g, ' ');

  const rowMatchesQuery = (row, query) => {
    const q = normalize(query);
    if (!q) return true;
    const words = q.split(/\s+/).filter(Boolean);
    const rowText = row.map((cell) => normalize(cell)).join(' ');
    return words.every((word) => rowText.includes(word));
  };

  const filteredRows = useMemo(
    () => rows.filter((row) => rowMatchesQuery(row, searchQuery)),
    [rows, searchQuery]
  );

  const hasSearchQuery = normalize(searchQuery).length > 0;
  const showEmptyState = hasSearchQuery && filteredRows.length === 0;
  const colCount = headers.length;

  const autoWidths = useMemo(() => {
    if (headers.length === 0) return [];
    const stripBackticks = (s) => String(s == null ? '' : s).replace(/`/g, '');
    const SAMPLE_SIZE = 200;
    const sample = rows.length > SAMPLE_SIZE ? rows.slice(0, SAMPLE_SIZE) : rows;

    const weights = headers.map((header, i) => {
      const headerLen = stripBackticks(header).length;
      let total = 0;
      let count = 0;
      for (const row of sample) {
        if (row && row[i] != null) {
          total += stripBackticks(row[i]).length;
          count++;
        }
      }
      const avg = count > 0 ? total / count : 0;
      return Math.max(headerLen, avg, 4);
    });

    const sumWeights = weights.reduce((a, b) => a + b, 0) || 1;
    let pcts = weights.map((w) => (w / sumWeights) * 100);

    const minPct = Math.min(8, 100 / headers.length);
    const totalDeficit = pcts.reduce((acc, p) => acc + Math.max(0, minPct - p), 0);
    if (totalDeficit > 0) {
      const surplusTotal = pcts.reduce((acc, p) => acc + Math.max(0, p - minPct), 0);
      if (surplusTotal > 0) {
        pcts = pcts.map((p) =>
          p < minPct ? minPct : p - ((p - minPct) / surplusTotal) * totalDeficit
        );
      }
    }
    return pcts.map((p) => `${p.toFixed(2)}%`);
  }, [headers, rows]);

  const getColWidth = (i) => {
    if (columnWidths && columnWidths[i]) return columnWidths[i];
    if (autoWidths[i]) return autoWidths[i];
    return `${100 / Math.max(1, headers.length)}%`;
  };

  const renderCellContent = (value) => {
    if (value == null) return '';
    if (typeof value !== 'string') return value;
    if (value.indexOf('`') === -1) return value;
    const parts = value.split('`');
    return parts.map((part, i) =>
      i % 2 === 1 ? (
        <code key={i} style={{ padding: '2px 6px', background: '#f3f4f6', color: '#1f2937', borderRadius: '4px', fontFamily: 'ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, monospace', fontSize: '0.875em' }}>
          {part}
        </code>
      ) : part
    );
  };

  const headerCellStyle = {
    padding: '12px 16px', textAlign: 'left', fontSize: '12px', fontWeight: 600,
    color: '#374151', textTransform: 'uppercase', letterSpacing: '0.05em',
    background: '#f9fafb', borderBottom: '1px solid #e5e7eb',
    verticalAlign: 'top', wordBreak: 'break-word', overflowWrap: 'break-word',
    position: 'sticky', top: 0, zIndex: 10,
  };

  const bodyCellStyle = {
    padding: '8px 16px', fontSize: '14px', color: '#111827',
    verticalAlign: 'top', wordBreak: 'break-word', overflowWrap: 'break-word',
    borderBottom: '1px solid #f3f4f6',
  };

  return (
    <div className={className} style={{ width: '100%', display: 'block' }}>
      <div style={{ width: '100%', border: '1px solid #e5e7eb', borderRadius: '6px', overflow: 'hidden', boxSizing: 'border-box' }}>
        <div style={{ padding: '12px 16px', background: '#f9fafb', borderBottom: '1px solid #e5e7eb' }}>
          <div style={{ display: 'flex', alignItems: 'center', width: '100%', maxWidth: '420px', border: '1px solid #d1d5db', borderRadius: '6px', background: '#ffffff', boxShadow: '0 1px 2px 0 rgba(0,0,0,0.05)' }}>
            <span aria-hidden="true" style={{ display: 'flex', alignItems: 'center', paddingLeft: '12px', color: '#9ca3af', flexShrink: 0 }}>
              <svg xmlns="http://www.w3.org/2000/svg" width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
                <circle cx="11" cy="11" r="8" />
                <path d="m21 21-4.35-4.35" />
              </svg>
            </span>
            <input
              type="search"
              placeholder={placeholder}
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              aria-label="Search table"
              style={{ flex: 1, minWidth: 0, padding: '8px 12px 8px 8px', border: 0, background: 'transparent', outline: 'none', fontSize: '14px' }}
            />
          </div>
        </div>

        <div style={{ width: '100%', overflowY: 'auto', overflowX: 'hidden', position: 'relative', maxHeight: showEmptyState ? 'none' : maxHeight, boxSizing: 'border-box' }}>
          <table style={{ width: '100%', tableLayout: 'fixed', borderCollapse: 'separate', borderSpacing: 0 }}>
            <colgroup>
              {headers.map((_, i) => (
                <col key={i} style={{ width: getColWidth(i) }} />
              ))}
            </colgroup>
            <thead>
              <tr>
                {headers.map((header, i) => (
                  <th key={i} scope="col" style={{ ...headerCellStyle, width: getColWidth(i) }}>
                    {renderCellContent(header)}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody style={{ background: '#ffffff' }}>
              {showEmptyState ? (
                <tr>
                  <td colSpan={colCount} role="alert" style={{ padding: '16px', background: '#fef2f2', color: '#b91c1c', borderBottom: '1px solid #fecaca' }}>
                    {emptyMessage}
                  </td>
                </tr>
              ) : (
                filteredRows.map((row, rowIndex) => (
                  <tr key={rowIndex}>
                    {row.map((cell, cellIndex) => (
                      <td key={cellIndex} style={bodyCellStyle}>
                        {renderCellContent(cell)}
                      </td>
                    ))}
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>

      {filteredRows.length > 0 && (
        <p style={{ marginTop: '8px', fontSize: '12px', color: '#6b7280' }}>
          Showing {filteredRows.length} of {rows.length} row{rows.length !== 1 ? 's' : ''}
          {hasSearchQuery && ' (filtered by search)'}.
        </p>
      )}
    </div>
  );
};

  placeholder="Search errors..."
