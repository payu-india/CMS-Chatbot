import { useState, useMemo } from 'react';

export const SearchableTable = (props) => {
  const {
    headers = [],
    rows = [],
    placeholder = 'Search in table...',
    maxHeight = '400px',
    emptyMessage = 'No results found. Try a different search.',
    className = '',
    columnWidths = null,
  } = props;

  const [searchQuery, setSearchQuery] = useState('');

  const normalize = (str) =>
    str == null ? '' : String(str).toLowerCase().trim().replace(/\s+/g, ' ');

  const rowMatchesQuery = (row, query) => {
    const q = normalize(query);
    if (!q) return true;
    const words = q.split(/\s+/).filter(Boolean);
    const rowText = row.map((cell) => normalize(cell)).join(' ');
    return words.every((word) => rowText.includes(word));
  };

  const filteredRows = useMemo(
    () => rows.filter((row) => rowMatchesQuery(row, searchQuery)),
    [rows, searchQuery]
  );

  const hasSearchQuery = normalize(searchQuery).length > 0;
  const showEmptyState = hasSearchQuery && filteredRows.length === 0;
  const colCount = headers.length;

  const renderCellContent = (value) => {
    if (value == null) return '';
    if (typeof value !== 'string') return value;
    if (value.indexOf('`') === -1) return value;
    const parts = value.split('`');
    return parts.map((part, i) =>
      i % 2 === 1 ? (
        <code key={i} className="px-1.5 py-0.5 bg-gray-100 text-gray-800 rounded text-[0.875em] font-mono">
          {part}
        </code>
      ) : (
        part
      )
    );
  };

  return (
    <div className={`w-full ${className}`}>
      <div className="border border-gray-200 rounded-md overflow-hidden">
        <div className="px-4 py-3 bg-gray-50 border-b border-gray-200">
          <div className="flex items-center w-full max-w-md border border-gray-300 rounded-md bg-white shadow-sm focus-within:ring-2 focus-within:ring-blue-500 focus-within:border-blue-500">
            <span className="flex items-center justify-center pl-3 text-gray-400 flex-shrink-0" aria-hidden="true">
              <svg xmlns="http://www.w3.org/2000/svg" width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
                <circle cx="11" cy="11" r="8" />
                <path d="m21 21-4.35-4.35" />
              </svg>
            </span>
            <input
              type="search"
              placeholder={placeholder}
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className="flex-1 min-w-0 py-2 pr-3 border-0 bg-transparent focus:ring-0 focus:outline-none"
              aria-label="Search table"
            />
          </div>
        </div>

        <div
          className="overflow-y-auto overflow-x-hidden relative"
          style={showEmptyState ? {} : { maxHeight }}
        >
          <table className="w-full border-separate border-spacing-0" style={{ tableLayout: 'fixed' }}>
            <thead>
              <tr>
                {headers.map((header, i) => {
                  const widthStyle =
                    columnWidths && columnWidths[i] ? { width: columnWidths[i] } : {};
                  return (
                    <th
                      key={i}
                      scope="col"
                      style={{ position: 'sticky', top: 0, zIndex: 10, ...widthStyle }}
                      className="px-4 py-3 text-left text-xs font-medium text-gray-700 uppercase tracking-wider bg-gray-50 border-b border-gray-200 align-top break-words"
                    >
                      {renderCellContent(header)}
                    </th>
                  );
                })}
              </tr>
            </thead>
            <tbody className="bg-white divide-y divide-gray-200">
              {showEmptyState ? (
                <tr>
                  <td colSpan={colCount} className="px-4 py-4 bg-red-50 text-red-700 border-b border-red-200" role="alert">
                    {emptyMessage}
                  </td>
                </tr>
              ) : (
                filteredRows.map((row, rowIndex) => (
                  <tr key={rowIndex} className="hover:bg-gray-50">
                    {row.map((cell, cellIndex) => (
                      <td key={cellIndex} className="px-4 py-2 text-sm text-gray-900 align-top break-words border-b border-gray-100">
                        {renderCellContent(cell)}
                      </td>
                    ))}
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>

      {filteredRows.length > 0 && (
        <p className="mt-2 text-xs text-gray-500">
          Showing {filteredRows.length} of {rows.length} row{rows.length !== 1 ? 's' : ''}
          {hasSearchQuery && ' (filtered by search)'}.
        </p>
      )}
    </div>
  );
};

<SearchableTable
  headers={['Bank Code', 'Description', 'Recommended Fix']}
  columnWidths={['18%', '32%', '50%']}
  rows={[
    ['`AUCNEGATIVE`', 'OTP Generated Successfully', 'Verify final payment status, show a customer-safe retry or alternate-payment message, and use a new txnid for any new payment attempt.'],
    ['`E348`', 'ISSUER_DECLINED', 'Verify final payment status, show a customer-safe retry or alternate-payment message, and use a new txnid for any new payment attempt.'],
  ]}
  placeholder="Search errors..."
/>
