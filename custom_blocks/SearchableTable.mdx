export const SearchableTable = (props) => {
  const {
    headers = [],
    rows = [],
    rowsJson = '',
    dataUrl = null,
    tableKey = '',
    placeholder = 'Search in table...',
    maxHeight = '400px',
    emptyMessage = 'No results found. Try a different search.',
    loadErrorMessage =
      'Could not load table data. Check the GitHub raw URL and try again.',
    className = '',
    columnWidths = null,
    maxLines = 0,
    codeColumns = null,
  } = props;

  const [searchQuery, setSearchQuery] = useState('');
  const [remoteRows, setRemoteRows] = useState(null);
  const [loadState, setLoadState] = useState(dataUrl ? 'loading' : 'idle');
  const [loadError, setLoadError] = useState('');

  useEffect(() => {
    const url = dataUrl;
    if (!url) {
      setRemoteRows(null);
      setLoadState('idle');
      setLoadError('');
      return undefined;
    }

    let cancelled = false;
    setLoadState('loading');
    setLoadError('');
    setRemoteRows(null);

    const primary = String(url).trim();
    const rawToJsDelivr = (rawUrl) => {
      const m = rawUrl.match(
        /^https:\/\/raw\.githubusercontent\.com\/([^/]+)\/([^/]+)\/([^/]+)\/(.+)$/
      );
      if (!m) return null;
      return (
        'https://cdn.jsdelivr.net/gh/' +
        m[1] +
        '/' +
        m[2] +
        '@' +
        m[3] +
        '/' +
        m[4]
      );
    };

    const parseRowsJson = (json) => {
      const loaded =
        json && Array.isArray(json.rows)
          ? json.rows
          : Array.isArray(json)
            ? json
            : null;
      if (!loaded) {
        throw new Error('JSON must be rows array or object with rows key');
      }
      return loaded;
    };

    const fetchOne = (fetchUrl) =>
      fetch(fetchUrl).then((res) =>
        res.ok ? res.json() : Promise.reject(new Error('HTTP ' + res.status))
      );

    const urls = [primary];
    const cdn = rawToJsDelivr(primary);
    if (cdn && cdn !== primary) urls.push(cdn);

    const loadSequential = (index, lastErr) => {
      if (index >= urls.length) {
        return Promise.reject(lastErr || new Error('Failed to fetch'));
      }
      return fetchOne(urls[index])
        .then(parseRowsJson)
        .catch((err) => loadSequential(index + 1, err));
    };

    loadSequential(0, null)
      .then((loaded) => {
        if (cancelled) return;
        setRemoteRows(loaded);
        setLoadState('ready');
      })
      .catch((err) => {
        if (cancelled) return;
        const msg = err && err.message ? err.message : 'Load failed';
        const hint =
          msg === 'Failed to fetch'
            ? ' (browser blocked the URL — use the jsDelivr link from GITHUB_HOSTING.md, or confirm the repo is public)'
            : msg.indexOf('HTTP 404') >= 0
              ? ' (file not found — check repo path: data/payment-error-codes.json)'
              : '';
        setLoadError(msg + hint);
        setLoadState('error');
      });

    return () => {
      cancelled = true;
    };
  }, [dataUrl]);

  const coerceRows = (input) => {
    if (input == null || input === '') return [];
    if (Array.isArray(input)) {
      return input.filter((row) => Array.isArray(row));
    }
    if (typeof input === 'string') {
      const text = input.trim();
      if (!text) return [];
      try {
        return coerceRows(JSON.parse(text));
      } catch (e) {
        return [];
      }
    }
    if (typeof input === 'object') {
      if (Array.isArray(input.rows)) return coerceRows(input.rows);
      if (typeof input.length === 'number' && input.length >= 0) {
        const list = [];
        for (let i = 0; i < input.length; i += 1) {
          list.push(input[i]);
        }
        return coerceRows(list);
      }
      const vals = Object.keys(input)
        .sort((a, b) => Number(a) - Number(b))
        .map((k) => input[k]);
      if (vals.length > 0 && vals.every((v) => Array.isArray(v))) return vals;
    }
    return [];
  };

  const builtinRowsForKey = (key) => {
    const k = key && String(key).trim();
    if (k === 'paymentErrors') {
      return typeof paymentErrorsTableRows !== 'undefined'
        ? coerceRows(paymentErrorsTableRows)
        : [];
    }
    return [];
  };

  const inlineRows = useMemo(() => {
    const fromKey = builtinRowsForKey(tableKey);
    if (fromKey.length > 0) return fromKey;
    const fromRows = coerceRows(rows);
    if (fromRows.length > 0) return fromRows;
    const fromJson = coerceRows(rowsJson);
    if (fromJson.length > 0) return fromJson;
    return [];
  }, [tableKey, rows, rowsJson]);

  const tableRows = remoteRows != null ? remoteRows : inlineRows;

  const normalize = (str) =>
    str == null ? '' : String(str).toLowerCase().trim().replace(/\s+/g, ' ');

  const rowMatchesQuery = (row, query) => {
    const q = normalize(query);
    if (!q) return true;
    const words = q.split(/\s+/).filter(Boolean);
    const rowText = row.map((cell) => normalize(cell)).join(' ');
    return words.every((word) => rowText.includes(word));
  };

  const filteredRows = useMemo(
    () => tableRows.filter((row) => rowMatchesQuery(row, searchQuery)),
    [tableRows, searchQuery]
  );

  const hasSearchQuery = normalize(searchQuery).length > 0;
  const isLoading = loadState === 'loading';
  const hasLoadError = loadState === 'error';
  const showEmptyState =
    !isLoading && !hasLoadError && hasSearchQuery && filteredRows.length === 0;
  const colCount = headers.length;

  const codeColIndexes = useMemo(() => {
    if (Array.isArray(codeColumns)) return codeColumns;
    const idx = headers.findIndex((h) =>
      /error/i.test(String(h == null ? '' : h))
    );
    return idx >= 0 ? [idx] : headers.length > 0 ? [0] : [];
  }, [codeColumns, headers]);

  const isCodeCol = (i) => codeColIndexes.indexOf(i) >= 0;

  const layoutWidths = useMemo(() => {
    const n = headers.length;
    if (n === 0) return null;
    if (Array.isArray(columnWidths) && columnWidths.length > 0) {
      const widths = columnWidths.slice();
      while (widths.length < n) {
        widths.push('auto');
      }
      return widths;
    }

    const sampleSize = Math.min(tableRows.length, 80);
    const weights = headers.map((header, colIdx) => {
      let maxLen = String(header == null ? '' : header).length;
      for (let r = 0; r < sampleSize; r += 1) {
        const cell = tableRows[r][colIdx];
        if (cell != null) {
          maxLen = Math.max(maxLen, String(cell).length);
        }
      }
      if (isCodeCol(colIdx)) {
        return Math.min(Math.max(maxLen, 6), 14) + 6;
      }
      return Math.min(maxLen, 120) + 12;
    });

    const total = weights.reduce((sum, w) => sum + w, 0) || 1;
    return weights.map((w) => ((w / total) * 100).toFixed(2) + '%');
  }, [columnWidths, headers, tableRows, codeColIndexes]);

  const useComputedWidths = layoutWidths != null && layoutWidths.length > 0;

  const fontSans =
    '-apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, "Helvetica Neue", Arial, sans-serif';
  const fontMono =
    'ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, monospace';
  const borderColor = '#e5e7eb';

  const codeStyle = {
    padding: '2px 6px',
    background: '#f3f4f6',
    color: '#1f2937',
    borderRadius: '4px',
    fontFamily: fontMono,
    fontSize: '13px',
    fontWeight: '400',
  };

  const renderCellContent = (value, colIndex) => {
    if (value == null) return '';
    if (typeof value !== 'string') return value;
    const plain = value.indexOf('`') >= 0 ? value.replace(/`/g, '') : value;
    return isCodeCol(colIndex) ? (
      <code style={codeStyle}>{plain}</code>
    ) : (
      plain
    );
  };

  const headerLabelStyle = {
    display: 'block',
    margin: 0,
    padding: 0,
    fontFamily: fontSans,
    fontSize: '12px',
    fontWeight: '700',
    lineHeight: '1.25',
    color: '#374151',
    textTransform: 'uppercase',
    letterSpacing: '0.05em',
  };

  const headerCellStyle = {
    padding: '12px 16px',
    textAlign: 'left',
    fontFamily: fontSans,
    fontSize: '12px',
    fontWeight: '700',
    color: '#374151',
    textTransform: 'uppercase',
    letterSpacing: '0.05em',
    backgroundColor: '#f9fafb',
    border: '1px solid ' + borderColor,
    verticalAlign: 'middle',
    wordBreak: 'normal',
    overflowWrap: 'break-word',
    position: 'sticky',
    top: 0,
    zIndex: 10,
    boxSizing: 'border-box',
  };

  const bodyCellStyle = {
    padding: '12px 16px',
    fontFamily: fontSans,
    fontSize: '14px',
    fontWeight: '400',
    lineHeight: '1.5',
    color: '#111827',
    backgroundColor: '#ffffff',
    border: '1px solid ' + borderColor,
    verticalAlign: 'top',
    wordBreak: 'normal',
    overflowWrap: 'break-word',
    boxSizing: 'border-box',
  };

  const bodyTextStyle = {
    margin: 0,
    padding: 0,
    fontFamily: fontSans,
    fontSize: '14px',
    fontWeight: '400',
    lineHeight: '1.5',
    color: '#111827',
  };

  const clampStyle =
    typeof maxLines === 'number' && maxLines > 0
      ? {
          display: '-webkit-box',
          WebkitBoxOrient: 'vertical',
          WebkitLineClamp: maxLines,
          overflow: 'hidden',
          textOverflow: 'ellipsis',
        }
      : null;

  const cellPlainText = (value) =>
    value == null ? '' : String(value).replace(/`/g, '');

  const mergeStyle = (base, extra) =>
    extra ? Object.assign({}, base, extra) : base;

  const showNoRows =
    !isLoading && !hasLoadError && tableRows.length === 0;

  return (
    <div
      className={className}
      style={{
        width: '100%',
        display: 'block',
        fontFamily: fontSans,
        fontSize: '14px',
        lineHeight: '1.5',
        color: '#111827',
      }}
    >
      <div
        style={{
          width: '100%',
          border: '1px solid ' + borderColor,
          borderRadius: '6px',
          overflow: 'hidden',
          boxSizing: 'border-box',
          backgroundColor: '#ffffff',
        }}
      >
        <div
          style={{
            padding: '12px 16px',
            backgroundColor: '#f9fafb',
            borderBottom: '1px solid ' + borderColor,
          }}
        >
          <div
            style={{
              display: 'flex',
              alignItems: 'center',
              width: '100%',
              maxWidth: '420px',
              border: '1px solid #d1d5db',
              borderRadius: '6px',
              backgroundColor: '#ffffff',
              boxShadow: '0 1px 2px 0 rgba(0,0,0,0.05)',
            }}
          >
            <span
              aria-hidden="true"
              style={{
                display: 'flex',
                alignItems: 'center',
                paddingLeft: '12px',
                color: '#9ca3af',
                flexShrink: 0,
              }}
            >
              <svg
                xmlns="http://www.w3.org/2000/svg"
                width="18"
                height="18"
                viewBox="0 0 24 24"
                fill="none"
                stroke="currentColor"
                strokeWidth="2"
                strokeLinecap="round"
                strokeLinejoin="round"
              >
                <circle cx="11" cy="11" r="8" />
                <path d="m21 21-4.35-4.35" />
              </svg>
            </span>
            <input
              type="search"
              placeholder={placeholder}
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              aria-label="Search table"
              style={{
                flex: 1,
                minWidth: 0,
                padding: '8px 12px 8px 8px',
                border: 0,
                backgroundColor: 'transparent',
                outline: 'none',
                fontSize: '14px',
                fontFamily: fontSans,
                fontWeight: '400',
                color: '#111827',
              }}
            />
          </div>
        </div>

        <div
          style={{
            width: '100%',
            overflowY: 'auto',
            overflowX: 'auto',
            position: 'relative',
            maxHeight: showEmptyState ? 'none' : maxHeight,
            boxSizing: 'border-box',
            borderTop: '1px solid ' + borderColor,
          }}
        >
          <table
            style={{
              width: '100%',
              minWidth: '100%',
              tableLayout: 'fixed',
              borderCollapse: 'collapse',
              borderSpacing: 0,
              margin: 0,
              fontFamily: fontSans,
            }}
          >
            {useComputedWidths ? (
              <colgroup>
                {headers.map((_, i) => (
                  <col key={i} style={{ width: layoutWidths[i] }} />
                ))}
              </colgroup>
            ) : null}
            <thead>
              <tr>
                {headers.map((header, i) => (
                  <th
                    key={i}
                    scope="col"
                    style={mergeStyle(headerCellStyle, {
                      width: useComputedWidths ? layoutWidths[i] : undefined,
                    })}
                  >
                    <span style={headerLabelStyle}>
                      {renderCellContent(header, i)}
                    </span>
                  </th>
                ))}
              </tr>
            </thead>
            <tbody style={{ backgroundColor: '#ffffff' }}>
              {showNoRows ? (
                <tr>
                  <td
                    colSpan={colCount || 1}
                    style={{
                      padding: '16px',
                      fontFamily: fontSans,
                      fontSize: '14px',
                      color: '#6b7280',
                      border: '1px solid ' + borderColor,
                    }}
                  >
                    No rows loaded. Set dataUrl to your GitHub raw JSON URL, or
                    pass rows for small tables.
                  </td>
                </tr>
              ) : isLoading ? (
                <tr>
                  <td
                    colSpan={colCount || 1}
                    style={{
                      padding: '16px',
                      fontFamily: fontSans,
                      fontSize: '14px',
                      color: '#6b7280',
                      border: '1px solid ' + borderColor,
                    }}
                  >
                    Loading table data...
                  </td>
                </tr>
              ) : hasLoadError ? (
                <tr>
                  <td
                    colSpan={colCount || 1}
                    role="alert"
                    style={{
                      padding: '16px',
                      background: '#fef2f2',
                      color: '#b91c1c',
                      border: '1px solid ' + borderColor,
                    }}
                  >
                    {loadErrorMessage}
                    {loadError ? ' (' + loadError + ')' : ''}
                  </td>
                </tr>
              ) : showEmptyState ? (
                <tr>
                  <td
                    colSpan={colCount}
                    role="alert"
                    style={{
                      padding: '16px',
                      background: '#fef2f2',
                      color: '#b91c1c',
                      borderBottom: '1px solid #fecaca',
                    }}
                  >
                    {emptyMessage}
                  </td>
                </tr>
              ) : (
                filteredRows.map((row, rowIndex) => (
                  <tr key={rowIndex}>
                    {row.map((cell, cellIndex) => (
                      <td
                        key={cellIndex}
                        style={mergeStyle(bodyCellStyle, {
                          width: useComputedWidths
                            ? layoutWidths[cellIndex]
                            : undefined,
                        })}
                      >
                        {clampStyle ? (
                          <div style={clampStyle} title={cellPlainText(cell)}>
                            {isCodeCol(cellIndex) ? (
                              renderCellContent(cell, cellIndex)
                            ) : (
                              <span style={bodyTextStyle}>
                                {renderCellContent(cell, cellIndex)}
                              </span>
                            )}
                          </div>
                        ) : isCodeCol(cellIndex) ? (
                          renderCellContent(cell, cellIndex)
                        ) : (
                          <span style={bodyTextStyle}>
                            {renderCellContent(cell, cellIndex)}
                          </span>
                        )}
                      </td>
                    ))}
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>

      {!isLoading && !hasLoadError && !showNoRows && filteredRows.length > 0 ? (
        <p
          style={{
            marginTop: '8px',
            fontSize: '12px',
            fontFamily: fontSans,
            color: '#6b7280',
          }}
        >
          Showing {filteredRows.length} of {tableRows.length} row
          {tableRows.length !== 1 ? 's' : ''}
          {hasSearchQuery ? ' (filtered by search)' : ''}.
        </p>
      ) : null}
    </div>
  );
};

<SearchableTable
  headers={['Errors', 'Description']}
  codeColumns={[0]}
  rows={[['E000', 'NO_ERROR']]}
  placeholder="Search"
/>
