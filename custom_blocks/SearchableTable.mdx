// SearchableTable - Readme CMS Custom Component
// Readme parses this entire file as MDX. Never use curly braces in comments here.
// No helpers above export. Preview uses rowsJson only (not nested rows arrays).

import { useState, useMemo, useEffect } from 'react';

export const SearchableTable = (props) => {
  const {
    headers = [],
    rows = [],
    rowsJson = '',
    dataUrl = null,
    placeholder = 'Search in table...',
    maxHeight = '400px',
    emptyMessage = 'No results found. Try a different search.',
    loadErrorMessage = 'Could not load table data. Check the data URL or try again later.',
    className = '',
    columnWidths = null,
    maxLines = 3,
    codeColumns = null,
  } = props;

  const [searchQuery, setSearchQuery] = useState('');
  const [remoteRows, setRemoteRows] = useState(null);
  const [loadState, setLoadState] = useState(dataUrl ? 'loading' : 'idle');
  const [loadError, setLoadError] = useState('');

  useEffect(() => {
    const url = dataUrl;
    if (!url) {
      setRemoteRows(null);
      setLoadState('idle');
      setLoadError('');
      return undefined;
    }

    let cancelled = false;
    setLoadState('loading');
    setLoadError('');
    setRemoteRows(null);

    fetch(url)
      .then((res) => (res.ok ? res.json() : Promise.reject(new Error('HTTP ' + res.status))))
      .then((json) => {
        if (cancelled) return;
        const loaded =
          json && Array.isArray(json.rows)
            ? json.rows
            : Array.isArray(json)
              ? json
              : null;
        if (!loaded) {
          throw new Error('JSON must be { rows: [...] } or a row array');
        }
        setRemoteRows(loaded);
        setLoadState('ready');
      })
      .catch((err) => {
        if (cancelled) return;
        setLoadError(err && err.message ? err.message : 'Load failed');
        setLoadState('error');
      });

    return () => {
      cancelled = true;
    };
  }, [dataUrl]);

  const inlineRows = useMemo(() => {
    const raw = rowsJson && String(rowsJson).trim();
    if (!raw) return Array.isArray(rows) ? rows : [];
    try {
      const parsed = JSON.parse(raw);
      return Array.isArray(parsed) ? parsed : [];
    } catch (e) {
      return Array.isArray(rows) ? rows : [];
    }
  }, [rowsJson, rows]);

  const tableRows = remoteRows != null ? remoteRows : inlineRows;

  const normalize = (str) =>
    str == null ? '' : String(str).toLowerCase().trim().replace(/\s+/g, ' ');

  const rowMatchesQuery = (row, query) => {
    const q = normalize(query);
    return !q
      ? true
      : q
          .split(/\s+/)
          .filter(Boolean)
          .every((word) =>
            row.map((cell) => normalize(cell)).join(' ').includes(word)
          );
  };

  const filteredRows = useMemo(
    () => tableRows.filter((row) => rowMatchesQuery(row, searchQuery)),
    [tableRows, searchQuery]
  );

  const hasSearchQuery = normalize(searchQuery).length > 0;
  const isLoading = loadState === 'loading';
  const hasLoadError = loadState === 'error';
  const showEmptyState =
    !isLoading && !hasLoadError && hasSearchQuery && filteredRows.length === 0;
  const colCount = headers.length;

  const useFixedLayout = Array.isArray(columnWidths) && columnWidths.length > 0;

  const isCodeCol = (i) =>
    Array.isArray(codeColumns) ? codeColumns.indexOf(i) >= 0 : false;

  const codeStyle = {
    padding: '2px 6px',
    background: '#f3f4f6',
    color: '#1f2937',
    borderRadius: '4px',
    fontFamily:
      'ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, monospace',
    fontSize: '0.875em',
  };

  const renderCellContent = (value, colIndex) => {
    if (value == null) return '';
    if (typeof value !== 'string') return value;
    const plain = value.indexOf('`') >= 0 ? value.replace(/`/g, '') : value;
    return isCodeCol(colIndex) ? (
      <code style={codeStyle}>{plain}</code>
    ) : (
      plain
    );
  };

  const headerCellStyle = {
    padding: '12px 16px',
    textAlign: 'left',
    fontSize: '12px',
    fontWeight: 600,
    color: '#374151',
    textTransform: 'uppercase',
    letterSpacing: '0.05em',
    background: '#f9fafb',
    borderBottom: '1px solid #e5e7eb',
    verticalAlign: 'top',
    wordBreak: 'normal',
    overflowWrap: 'break-word',
    position: 'sticky',
    top: 0,
    zIndex: 10,
  };

  const bodyCellStyle = {
    padding: '8px 16px',
    fontSize: '14px',
    color: '#111827',
    verticalAlign: 'top',
    wordBreak: 'normal',
    overflowWrap: 'break-word',
    borderBottom: '1px solid #f3f4f6',
  };

  const clampStyle =
    typeof maxLines === 'number' && maxLines > 0
      ? {
          display: '-webkit-box',
          WebkitBoxOrient: 'vertical',
          WebkitLineClamp: maxLines,
          overflow: 'hidden',
          textOverflow: 'ellipsis',
          maxWidth: '480px',
        }
      : null;

  const cellPlainText = (value) =>
    value == null ? '' : String(value).replace(/`/g, '');

  const mergeStyle = (base, extra) =>
    extra ? Object.assign({}, base, extra) : base;

  return (
    <div className={className} style={{ width: '100%', display: 'block' }}>
      <div
        style={{
          width: '100%',
          border: '1px solid #e5e7eb',
          borderRadius: '6px',
          overflow: 'hidden',
          boxSizing: 'border-box',
        }}
      >
        <div
          style={{
            padding: '12px 16px',
            background: '#f9fafb',
            borderBottom: '1px solid #e5e7eb',
          }}
        >
          <div
            style={{
              display: 'flex',
              alignItems: 'center',
              width: '100%',
              maxWidth: '420px',
              border: '1px solid #d1d5db',
              borderRadius: '6px',
              background: '#ffffff',
              boxShadow: '0 1px 2px 0 rgba(0,0,0,0.05)',
            }}
          >
            <span
              aria-hidden="true"
              style={{
                display: 'flex',
                alignItems: 'center',
                paddingLeft: '12px',
                color: '#9ca3af',
                flexShrink: 0,
              }}
            >
              <svg
                xmlns="http://www.w3.org/2000/svg"
                width="18"
                height="18"
                viewBox="0 0 24 24"
                fill="none"
                stroke="currentColor"
                strokeWidth="2"
                strokeLinecap="round"
                strokeLinejoin="round"
              >
                <circle cx="11" cy="11" r="8" />
                <path d="m21 21-4.35-4.35" />
              </svg>
            </span>
            <input
              type="search"
              placeholder={placeholder}
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              aria-label="Search table"
              style={{
                flex: 1,
                minWidth: 0,
                padding: '8px 12px 8px 8px',
                border: 0,
                background: 'transparent',
                outline: 'none',
                fontSize: '14px',
              }}
            />
          </div>
        </div>

        <div
          style={{
            width: '100%',
            overflowY: 'auto',
            overflowX: 'auto',
            position: 'relative',
            maxHeight: showEmptyState ? 'none' : maxHeight,
            boxSizing: 'border-box',
          }}
        >
          <table
            style={{
              width: useFixedLayout ? '100%' : 'max-content',
              minWidth: '100%',
              tableLayout: useFixedLayout ? 'fixed' : 'auto',
              borderCollapse: 'separate',
              borderSpacing: 0,
            }}
          >
            {useFixedLayout ? (
              <colgroup>
                {headers.map((_, i) => (
                  <col key={i} style={{ width: columnWidths[i] }} />
                ))}
              </colgroup>
            ) : null}
            <thead>
              <tr>
                {headers.map((header, i) => (
                  <th
                    key={i}
                    scope="col"
                    style={mergeStyle(
                      headerCellStyle,
                      useFixedLayout ? { width: columnWidths[i] } : null
                    )}
                  >
                    {renderCellContent(header, i)}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody style={{ background: '#ffffff' }}>
              {isLoading ? (
                <tr>
                  <td
                    colSpan={colCount}
                    style={{
                      padding: '16px',
                      color: '#6b7280',
                      borderBottom: '1px solid #f3f4f6',
                    }}
                  >
                    Loading table data...
                  </td>
                </tr>
              ) : hasLoadError ? (
                <tr>
                  <td
                    colSpan={colCount}
                    role="alert"
                    style={{
                      padding: '16px',
                      background: '#fef2f2',
                      color: '#b91c1c',
                      borderBottom: '1px solid #fecaca',
                    }}
                  >
                    {loadErrorMessage}
                    {loadError ? ' (' + loadError + ')' : ''}
                  </td>
                </tr>
              ) : showEmptyState ? (
                <tr>
                  <td
                    colSpan={colCount}
                    role="alert"
                    style={{
                      padding: '16px',
                      background: '#fef2f2',
                      color: '#b91c1c',
                      borderBottom: '1px solid #fecaca',
                    }}
                  >
                    {emptyMessage}
                  </td>
                </tr>
              ) : (
                filteredRows.map((row, rowIndex) => (
                  <tr key={rowIndex}>
                    {row.map((cell, cellIndex) => (
                      <td
                        key={cellIndex}
                        style={bodyCellStyle}
                      >
                        {clampStyle ? (
                          <div style={clampStyle} title={cellPlainText(cell)}>
                            {renderCellContent(cell, cellIndex)}
                          </div>
                        ) : (
                          renderCellContent(cell, cellIndex)
                        )}
                      </td>
                    ))}
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>

      {!isLoading && !hasLoadError && filteredRows.length > 0 ? (
        <p style={{ marginTop: '8px', fontSize: '12px', color: '#6b7280' }}>
          Showing {filteredRows.length} of {tableRows.length} row
          {tableRows.length !== 1 ? 's' : ''}
          {hasSearchQuery ? ' (filtered by search)' : ''}.
        </p>
      ) : null}
    </div>
  );
};

<SearchableTable headers={['Code', 'Description']} codeColumns={[0]} rowsJson='[["E000","NO_ERROR"]]' />
