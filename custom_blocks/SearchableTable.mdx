/**
 * SearchableTable - Readme CMS Custom Component
 * 
 * A table with:
 * - Search input that filters rows by any column (case-insensitive, multi-word)
 * - "No results found" error state when search has no matches
 * - Scrollable table body (fixed max height with overflow)
 * 
 * Usage in Readme: Add this code in Settings > Custom Components, then use
 * <SearchableTable headers={["Column A","Column B"]} rows={[["a1","b1"],["a2","b2"]]} />
 * 
 * Reference: Razorpay-style supported banks/cards table
 */

import { useState, useMemo } from 'react';

export const SearchableTable = (props) => {
  const {
    headers = [],
    rows = [],
    placeholder = 'Search in table...',
    maxHeight = '400px',
    emptyMessage = 'No results found. Try a different search.',
    className = '',
  } = props;

  const [searchQuery, setSearchQuery] = useState('');

  const normalize = (str) =>
    str == null ? '' : String(str).toLowerCase().trim().replace(/\s+/g, ' ');

  const rowMatchesQuery = (row, query) => {
    const q = normalize(query);
    if (!q) return true;
    const words = q.split(/\s+/).filter(Boolean);
    const rowText = row.map((cell) => normalize(cell)).join(' ');
    return words.every((word) => rowText.includes(word));
  };

  const filteredRows = useMemo(
    () => rows.filter((row) => rowMatchesQuery(row, searchQuery)),
    [rows, searchQuery]
  );

  const hasSearchQuery = normalize(searchQuery).length > 0;
  const showEmptyState = hasSearchQuery && filteredRows.length === 0;

  const colCount = headers.length;

  // Renders a string with `inline code` segments as <code> elements.
  // Non-string values (numbers, JSX) pass through unchanged.
  const renderCellContent = (value) => {
    if (value == null) return '';
    if (typeof value !== 'string') return value;
    if (value.indexOf('`') === -1) return value;
    const parts = value.split('`');
    return parts.map((part, i) =>
      i % 2 === 1 ? (
        <code
          key={i}
          className="px-1.5 py-0.5 bg-gray-100 text-gray-800 rounded text-[0.875em] font-mono"
        >
          {part}
        </code>
      ) : (
        part
      )
    );
  };

  return (
    <div className={`w-fit max-w-full ${className}`}>
      <div
        className="border border-gray-200 rounded-md overflow-hidden"
        style={showEmptyState ? {} : { maxHeight }}
      >
        <div
          className="overflow-auto"
          style={showEmptyState ? {} : { maxHeight }}
        >
          <table className="min-w-full divide-y divide-gray-200 border-collapse">
            <thead>
              {/* Frozen row 1: search bar */}
              <tr className="sticky top-0 z-20 bg-gray-50 border-b border-gray-200">
                <th
                  colSpan={colCount}
                  className="px-4 py-3 bg-gray-50"
                >
                  <div className="flex items-center w-full max-w-md border border-gray-300 rounded-md bg-white shadow-sm focus-within:ring-2 focus-within:ring-blue-500 focus-within:border-blue-500">
                    <span className="flex items-center justify-center pl-3 text-gray-400 flex-shrink-0" aria-hidden="true">
                      <svg xmlns="http://www.w3.org/2000/svg" width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
                        <circle cx="11" cy="11" r="8" />
                        <path d="m21 21-4.35-4.35" />
                      </svg>
                    </span>
                    <input
                      type="search"
                      placeholder={placeholder}
                      value={searchQuery}
                      onChange={(e) => setSearchQuery(e.target.value)}
                      className="flex-1 min-w-0 py-2 pr-3 border-0 bg-transparent focus:ring-0 focus:outline-none"
                      aria-label="Search table"
                    />
                  </div>
                </th>
              </tr>
              {/* Frozen row 2: column headers */}
              <tr className="sticky top-[3.5rem] z-10 bg-gray-50 shadow-[0_2px_0_0_rgba(0,0,0,0.05)]">
                {headers.map((header, i) => (
                  <th
                    key={i}
                    scope="col"
                    className="px-4 py-3 text-left text-xs font-medium text-gray-700 uppercase tracking-wider bg-gray-50 border-b border-gray-200"
                  >
                    {renderCellContent(header)}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody className="bg-white divide-y divide-gray-200">
              {showEmptyState ? (
                <tr>
                  <td
                    colSpan={colCount}
                    className="px-4 py-4 bg-red-50 text-red-700 border-b border-red-200"
                    role="alert"
                  >
                    {emptyMessage}
                  </td>
                </tr>
              ) : (
                filteredRows.map((row, rowIndex) => (
                  <tr key={rowIndex} className="hover:bg-gray-50">
                    {row.map((cell, cellIndex) => (
                      <td
                        key={cellIndex}
                        className="px-4 py-2 text-sm text-gray-900"
                      >
                        {renderCellContent(cell)}
                      </td>
                    ))}
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>

      {filteredRows.length > 0 && (
        <p className="mt-2 text-xs text-gray-500">
          Showing {filteredRows.length} of {rows.length} row{rows.length !== 1 ? 's' : ''}
          {hasSearchQuery && ' (filtered by search)'}.
        </p>
      )}
    </div>
  );
};

// Default preview for Readme CMS (component must appear after exports)
<SearchableTable
  headers={['Bank', 'Type', 'Status']}
  rows={[
    ['HDFC Bank', 'Netbanking / Cards', 'Active'],
    ['ICICI Bank', 'Netbanking / Cards', 'Active'],
    ['State Bank of India', 'Netbanking', 'Active'],
    ['Axis Bank', 'Cards', 'Active'],
  ]}
  placeholder="Search banks..."
/>
