/**
 * SearchableTable - Readme CMS Custom Component
 * 
 * A table with:
 * - Search input that filters rows by any column (case-insensitive, multi-word)
 * - "No results found" error state when search has no matches
 * - Scrollable table body (fixed max height with overflow)
 * 
 * Usage in Readme: Add this code in Settings > Custom Components, then use
 * <SearchableTable headers={["Column A","Column B"]} rows={[["a1","b1"],["a2","b2"]]} />
 * 
 * Reference: Razorpay-style supported banks/cards table
 */

import { useState, useMemo } from 'react';

export const SearchableTable = (props) => {
  const {
    headers = [],
    rows = [],
    placeholder = 'Search in table...',
    maxHeight = '400px',
    emptyMessage = 'No results found. Try a different search.',
    className = '',
  } = props;

  const [searchQuery, setSearchQuery] = useState('');

  const normalize = (str) =>
    str == null ? '' : String(str).toLowerCase().trim().replace(/\s+/g, ' ');

  const rowMatchesQuery = (row, query) => {
    const q = normalize(query);
    if (!q) return true;
    const words = q.split(/\s+/).filter(Boolean);
    const rowText = row.map((cell) => normalize(cell)).join(' ');
    return words.every((word) => rowText.includes(word));
  };

  const filteredRows = useMemo(
    () => rows.filter((row) => rowMatchesQuery(row, searchQuery)),
    [rows, searchQuery]
  );

  const hasSearchQuery = normalize(searchQuery).length > 0;
  const showEmptyState = hasSearchQuery && filteredRows.length === 0;

  const colCount = headers.length;

  return (
    <div className={`w-full ${className}`}>
      <div
        className="border border-gray-200 rounded-md overflow-hidden"
        style={{ maxHeight: showEmptyState ? 'none' : maxHeight }}
      >
        <div
          className="overflow-auto"
          style={showEmptyState ? {} : { maxHeight }}
        >
          <table className="min-w-full divide-y divide-gray-200">
            <thead className="bg-gray-50 sticky top-0 z-10">
              <tr>
                <th
                  colSpan={colCount}
                  className="px-4 py-3 bg-gray-50 border-b border-gray-200"
                >
                  <div className="relative max-w-md">
                    <span className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400 pointer-events-none">
                      <svg xmlns="http://www.w3.org/2000/svg" width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
                        <circle cx="11" cy="11" r="8" />
                        <path d="m21 21-4.35-4.35" />
                      </svg>
                    </span>
                    <input
                      type="search"
                      placeholder={placeholder}
                      value={searchQuery}
                      onChange={(e) => setSearchQuery(e.target.value)}
                      className="w-full pl-10 pr-3 py-2 border border-gray-300 rounded-md shadow-sm focus:ring-2 focus:ring-blue-500 focus:border-blue-500"
                      aria-label="Search table"
                    />
                  </div>
                </th>
              </tr>
              <tr>
                {headers.map((header, i) => (
                  <th
                    key={i}
                    scope="col"
                    className="px-4 py-3 text-left text-xs font-medium text-gray-700 uppercase tracking-wider"
                  >
                    {header}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody className="bg-white divide-y divide-gray-200">
              {showEmptyState ? (
                <tr>
                  <td
                    colSpan={colCount}
                    className="px-4 py-4 bg-red-50 text-red-700 border-b border-red-200"
                    role="alert"
                  >
                    {emptyMessage}
                  </td>
                </tr>
              ) : (
                filteredRows.map((row, rowIndex) => (
                  <tr key={rowIndex} className="hover:bg-gray-50">
                    {row.map((cell, cellIndex) => (
                      <td
                        key={cellIndex}
                        className="px-4 py-2 text-sm text-gray-900"
                      >
                        {cell}
                      </td>
                    ))}
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>

      {filteredRows.length > 0 && (
        <p className="mt-2 text-xs text-gray-500">
          Showing {filteredRows.length} of {rows.length} row{rows.length !== 1 ? 's' : ''}
          {hasSearchQuery && ' (filtered by search)'}.
        </p>
      )}
    </div>
  );
};

<SearchableTable
  headers={['Bank', 'Type', 'Status']}
  rows={[
    ['HDFC Bank', 'Netbanking / Cards', 'Active'],
    ['ICICI Bank', 'Netbanking / Cards', 'Active'],
    ['State Bank of India', 'Netbanking', 'Active'],
    ['Axis Bank', 'Cards', 'Active'],
  ]}
  placeholder="Search"
/>
