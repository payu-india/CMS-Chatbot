---
name: RequiredStar
---
export const RequiredStar = (props) => {
  const name = props.param || props.name || props.field || '';
  const codeStyle = {
    display: 'inline-block',
    whiteSpace: 'nowrap',
    padding: '2px 6px',
    background: '#f3f4f6',
    color: '#1f2937',
    borderRadius: '4px',
    fontFamily: 'ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, monospace',
    fontSize: '0.875em',
  };
  const wrapStyle = {
    display: 'inline-block',
    position: 'relative',
    whiteSpace: 'nowrap',
    verticalAlign: 'baseline',
  };
  const starStyle = {
    position: 'absolute',
    top: '-0.2em',
    right: '-0.45em',
    color: '#e53e3e',
    fontWeight: 700,
    fontSize: '0.7em',
    lineHeight: 1,
    pointerEvents: 'none',
  };
  const starOnlyStyle = {
    color: '#e53e3e',
    fontWeight: 700,
    fontSize: '0.85em',
    lineHeight: 0,
    verticalAlign: 'super',
    display: 'inline',
    margin: 0,
    padding: 0,
    border: 'none',
    background: 'transparent',
  };
  if (props.htmlblock && name) {
    return (
      <HTMLBlock>{`<span style="display:inline-block;position:relative;white-space:nowrap;vertical-align:baseline"><code style="display:inline-block;white-space:nowrap;padding:2px 6px;background:#f3f4f6;color:#1f2937;border-radius:4px;font-family:ui-monospace,Menlo,monospace;font-size:0.875em">${name}</code><span aria-hidden="true" style="position:absolute;top:-0.2em;right:-0.45em;color:#e53e3e;font-weight:700;font-size:0.7em;line-height:1">*</span></span>`}</HTMLBlock>
    );
  }
  if (name) {
    return (
      <span style={wrapStyle}>
        <code style={codeStyle}>{name}</code>
        <span aria-hidden="true" style={starStyle}>
          *
        </span>
      </span>
    );
  }
  return (
    <sup aria-hidden="true" style={starOnlyStyle}>
      *
    </sup>
  );
};

<RequiredStar param="merchant_id" />
