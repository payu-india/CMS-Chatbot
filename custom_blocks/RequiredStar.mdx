export const RequiredStar = (props) => {
  const name = props.param || props.name || props.field || '';
  if (name) {
    return (
      <code className="inline whitespace-nowrap rounded px-1.5 py-0.5 bg-gray-100 text-gray-800 font-mono text-sm">
        {name}
        <span className="text-red-600 font-bold ml-0.5" aria-hidden="true">
          *
        </span>
      </code>
    );
  }
  return (
    <span className="inline text-red-600 font-bold align-super text-xs ml-0.5" aria-hidden="true">
      *
    </span>
  );
};

<RequiredStar param="merchant_id" />
