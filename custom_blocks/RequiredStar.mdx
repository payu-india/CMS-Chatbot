// Star only: <RequiredStar />   Code + star: <RequiredStar code="merchant_id" />

var React =
  (typeof require === 'function' && require('react')) ||
  (typeof React !== 'undefined' ? React : typeof window !== 'undefined' ? window.React : null);

var mp = function (a) {
  return a.reduce(function (o, x, i) {
    if (i % 2 === 0 && i + 1 < a.length) o[x] = a[i + 1];
    return o;
  }, Object.create(null));
};

export var RequiredStar = function (props) {
  var label = props.code || props.name || props.children || '';
  label = label ? String(label).trim() : '';
  if (label !== '') {
    return React.createElement(
      'code',
      mp(['className', 'inline whitespace-nowrap rounded px-1.5 py-0.5 bg-gray-100 text-gray-800 font-mono text-sm']),
      label,
      React.createElement('span', mp(['className', 'text-red-600 font-bold ml-0.5', 'aria-hidden', 'true']), '*')
    );
  }
  return React.createElement(
    'span',
    mp(['className', 'inline text-red-600 font-bold align-super text-xs ml-0.5', 'aria-hidden', 'true']),
    '*'
  );
};
