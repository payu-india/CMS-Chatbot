  function empty(s) {
    return s == null ? '' : String(s).trim();
  }

  function labRead() {

  function labWrite(data) {
    try {
      if (typeof window === 'undefined' || !window.sessionStorage) return false;
      data.updatedAt = Date.now();
      window.sessionStorage.setItem(LAB_KEY, JSON.stringify(data));

      return false;
    }
  }

  function labEnsure() {
    var existing = labRead();
    if (existing && existing.params) return existing;

  }

  function buildForwardHashString(params, salt) {
    var p = params;
    return (
      empty(p.key) + '|' + empty(p.txnid) + '|' + empty(p.amount) + '|' + empty(p.productinfo) + '|' +
      empty(p.firstname) + '|' + empty(p.email) + '|' + empty(p.udf1) + '|' + empty(p.udf2) + '|' +
      empty(p.udf3) + '|' + empty(p.udf4) + '|' + empty(p.udf5) + '||||||' + empty(salt)
    );
  }

  function buildReverseHashString(salt, resp) {
    return (
      empty(salt) + '|' + empty(resp.status) + '||||||' + empty(resp.udf5) + '|' + empty(resp.udf4) + '|' +
      empty(resp.udf3) + '|' + empty(resp.udf2) + '|' + empty(resp.udf1) + '|' + empty(resp.email) + '|' +
      empty(resp.firstname) + '|' + empty(resp.productinfo) + '|' + empty(resp.amount) + '|' +
      empty(resp.txnid) + '|' + empty(resp.key)
    );
  }

  function sha512Hex(msg) {
    var enc = new TextEncoder();
    return crypto.subtle.digest('SHA-512', enc.encode(msg)).then(function (buf) {
      return Array.from(new Uint8Array(buf))
        .map(function (b) { return b.toString(16).padStart(2, '0'); })
        .join('');
    });
  }

  function escapeAttr(s) {
    return String(s)
      .replace(/&/g, '&amp;')
      .replace(/"/g, '&quot;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;');
  }

  function labStopBubble(e) {
    if (!e) return;
    if (e.stopPropagation) e.stopPropagation();
    if (e.nativeEvent && e.nativeEvent.stopImmediatePropagation) {
      e.nativeEvent.stopImmediatePropagation();
    }
  }

  function labStopEvent(e) {
    labStopBubble(e);
    if (e.preventDefault) e.preventDefault();
  }

  function labClick(fn) {
    return function (e) {
      labStopBubble(e);
      fn(e);
    };
  }

  function copyText(text, setCopyMsg) {
    var value = String(text || '');
    if (!value) return;
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(value).then(function () {
        setCopyMsg('Copied.');
        setTimeout(function () { setCopyMsg(''); }, 2000);
      });
    }
  }

  var SANDBOX_OPEN_GLOBAL = '__payuHostedCheckoutSandboxOpen';

  function sandboxOpenStore() {
    if (typeof window !== 'undefined') {
      if (!window[SANDBOX_OPEN_GLOBAL]) window[SANDBOX_OPEN_GLOBAL] = {};
      return window[SANDBOX_OPEN_GLOBAL];
    }
    return {};

  function sandboxOpenRead(id) {
    return sandboxOpenStore()[id] === true;
  }

  function sandboxOpenWrite(id, isOpen) {
    sandboxOpenStore()[id] = !!isOpen;
  }

  function sandboxSyncChevron(details) {
    if (!details) return;
    var chevron = details.querySelector('[data-sandbox-chevron]');
    if (chevron) chevron.textContent = details.open ? '▼' : '▶';
  }

  function onSandboxToggle(e) {
    var details = e.currentTarget;
    var id = details.getAttribute('data-sandbox-id') || 'sandbox';
    sandboxOpenWrite(id, details.open);
    sandboxSyncChevron(details);
  }

  function onSandboxSummaryClick(e) {
    var summary = e.currentTarget;
    var details = summary.parentNode;
    if (!details || details.tagName !== 'DETAILS') return;
    labStopEvent(e);
    details.open = !details.open;
    var id = details.getAttribute('data-sandbox-id') || 'sandbox';
    sandboxOpenWrite(id, details.open);
    sandboxSyncChevron(details);
  }

  function SandboxAccordion(props) {

    useEffect(function () {
      var details = detailsRef.current;
      if (!details) return;
      if (sandboxOpenRead(sandboxId)) details.open = true;
      sandboxSyncChevron(details);

        data-sandbox-id={sandboxId}

  var SAMPLE = LAB_CONFIG.sampleReverseResponse;
  var STEP14_UI_GLOBAL = '__payuHostedCheckoutStep14Form';

  function step14Store() {
    if (typeof window !== 'undefined') {
      if (!window[STEP14_UI_GLOBAL]) {
        var lab = labRead();
        var resp = Object.assign({}, SAMPLE);
        if (lab && lab.params) {
          Object.assign(resp, lab.params);
          resp.status = 'success';
        }
        window[STEP14_UI_GLOBAL] = {
          salt: lab && lab.salt ? lab.salt : '',
          responseHash: '',
          resp: resp
        };
      }
      return window[STEP14_UI_GLOBAL];
    }
    return { salt: '', responseHash: '', resp: Object.assign({}, SAMPLE) };
  }

  function onStep14Input(kind, fieldKey) {
    return function (e) {
      var store = step14Store();
      if (kind === 'salt') store.salt = e.target.value;
      else if (kind === 'hash') store.responseHash = e.target.value;
      else store.resp[fieldKey] = e.target.value;
      if (typeof window !== 'undefined') {
        if (!window.__payuHostedCheckoutSandboxOpen) window.__payuHostedCheckoutSandboxOpen = {};
        window.__payuHostedCheckoutSandboxOpen['Step 1.4 sandbox'] = true;
      }
    };
  }

  function collectStep14FromDom() {
    var store = step14Store();
    if (typeof document === 'undefined') return store;
    var fields = document.querySelectorAll('[data-payu-step14-field], [data-payu-step14-kind]');
    for (var i = 0; i < fields.length; i += 1) {
      var el = fields[i];
      var kind = el.getAttribute('data-payu-step14-kind');
      if (kind === 'salt') store.salt = el.value;
      else if (kind === 'hash') store.responseHash = el.value;
      else {
        var k = el.getAttribute('data-payu-step14-field');
        if (k) store.resp[k] = el.value;
      }
    }
    return store;
  }

  function syncStep14FieldToDom(kind, fieldKey, v) {
    if (typeof document === 'undefined') return;
    var el = kind === 'salt'
      ? document.querySelector('[data-payu-step14-kind="salt"]')
      : kind === 'hash'
      ? document.querySelector('[data-payu-step14-kind="hash"]')
      : document.querySelector('[data-payu-step14-field="' + fieldKey + '"]');
    if (el) el.value = v == null ? '' : String(v);
  }

  function syncAllStep14ToDom(store) {
    syncStep14FieldToDom('salt', '', store.salt);
    syncStep14FieldToDom('hash', '', store.responseHash);
    var keys = Object.keys(store.resp || {});
    for (var i = 0; i < keys.length; i += 1) {
      syncStep14FieldToDom('resp', keys[i], store.resp[keys[i]]);
    }
  }

  var store = step14Store();
  var _computed = useState('');
  var computed = _computed[0];
  var setComputed = _computed[1];
  var _busy = useState(false);
  var busy = _busy[0];
  var setBusy = _busy[1];
  var _err = useState('');
  var err = _err[0];
  var setErr = _err[1];
  var _match = useState(null);
  var match = _match[0];
  var setMatch = _match[1];

  var REV_FIELDS = JSON.parse('[{"k":"status","l":"status"},{"k":"txnid","l":"txnid"},{"k":"amount","l":"amount"},{"k":"productinfo","l":"productinfo"},{"k":"firstname","l":"firstname"},{"k":"email","l":"email"},{"k":"key","l":"key"},{"k":"udf1","l":"udf1"},{"k":"udf2","l":"udf2"},{"k":"udf3","l":"udf3"},{"k":"udf4","l":"udf4"},{"k":"udf5","l":"udf5"}]');

  function isRevFieldRequired(k) {
    return k.indexOf('udf') !== 0;
  }

  useEffect(function () {
    var lab = labEnsure();
    var data = step14Store();
    if (lab.salt) data.salt = lab.salt;
    if (lab.params) {
      data.resp = Object.assign({}, SAMPLE, lab.params);
      data.resp.status = 'success';
    }
    syncAllStep14ToDom(data);
  }, []);

  function verifyReverseHash() {
    var data = collectStep14FromDom();
    if (empty(data.salt)) {
      setErr('Enter test salt (same as Step 1.2).');
      return;
    }
    if (empty(data.responseHash)) {
      setErr('Paste the hash from PayU response.');
      return;
    }
    setBusy(true);
    setErr('');
    setMatch(null);
    setComputed('');
    var chain = buildReverseHashString(data.salt, data.resp);
    sha512Hex(chain)
      .then(function (h) {
        setComputed(h);
        setMatch(h.toLowerCase() === empty(data.responseHash).toLowerCase());
        setBusy(false);
      })
      .catch(function () {
        setErr('SHA-512 unavailable.');
        setBusy(false);
      });
  }

  return (
    <SandboxAccordion title="Try Out! · Verify reverse hash" intro="Paste callback values from surl/furl and verify the reverse hash. Test only — production verification runs on your server." ariaLabel="Step 1.4 sandbox">
      <div className={shellClass}>
      <div style={fieldStackStyle}>
        <div>
          <FieldLabel name="salt" required={true} />
          <input
            type="password"
            data-payu-step14-kind="salt"
            defaultValue={store.salt || ''}
            onInput={onStep14Input('salt', '')}
            style={inputStyle}
            autoComplete="off"
          />
        </div>
        {REV_FIELDS.map(function (row) {
          return (
            <div key={row.k}>
              <FieldLabel name={row.k} required={isRevFieldRequired(row.k)} />
              <input
                type="text"
                data-payu-step14-field={row.k}
                data-payu-step14-kind="resp"
                defaultValue={store.resp[row.k] || ''}
                onInput={onStep14Input('resp', row.k)}
                style={inputStyle}
              />
            </div>
          );
        })}
        <div>
          <FieldLabel name="hash" required={true} />
          <input
            type="text"
            data-payu-step14-kind="hash"
            defaultValue={store.responseHash || ''}
            onInput={onStep14Input('hash', '')}
            style={inputStyle}
            placeholder="Paste hash from callback"
          />
        </div>
      </div>
      <div style={btnRowStyle}>
        <button type="button" onClick={labClick(verifyReverseHash)} disabled={busy} style={btnPrimaryStyle}>
          {busy ? 'Verifying…' : 'Verify reverse hash'}
        </button>
        <a href={LAB_CONFIG.hashVerifyToolUrl} target="_blank" rel="noopener noreferrer" style={linkBtnStyle}>
          Open Hash Verification Tool
        </a>
      </div>
      {err ? <p style={errTextStyle} role="alert">{err}</p> : null}
      {match === true ? <p style={okTextStyle}>Hash matches — response verified for this lab.</p> : null}
      {match === false ? <p style={errTextStyle}>Hash mismatch — check field order, salt, and status.</p> : null}
      {computed ? <code style={codeBlockStyle}>{computed}</code> : null}
      </div>
    </SandboxAccordion>
  );
};
