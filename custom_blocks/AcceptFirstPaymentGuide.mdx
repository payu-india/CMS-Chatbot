  function copyText(key, value) {
    if (!value) return;
    var done = function () {
      setCopied(key);
      if (copyTimer.current) clearTimeout(copyTimer.current);
      copyTimer.current = setTimeout(function () {
        setCopied('');
      }, 1800);
    };
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(value).then(done).catch(function () {
        fallbackCopy(value, done);
      });
    } else {
      fallbackCopy(value, done);
    }
  }

  function fallbackCopy(value, done) {
    try {
      var ta = document.createElement('textarea');
      ta.value = value;
      ta.setAttribute('readonly', '');
      ta.style.position = 'fixed';
      ta.style.left = '-9999px';
      document.body.appendChild(ta);
      ta.select();
      document.execCommand('copy');
      document.body.removeChild(ta);
      done();
    } catch (e) {
      // ignore
    }
  }

        </pre>
      </div>
    );
  }

  function sectionShell(id, title, children) {
    return (
      <section
        key={id}
        data-section-id={id}
        aria-labelledby={'afp-' + id}
        style={{
          marginTop: '28px',
          paddingTop: '8px',
          borderTop: '1px solid ' + line,
          scrollMarginTop: '96px',
        }}

        <h3 id={'afp-' + id} style={{ margin: '0 0 8px', fontSize: '1.15rem', color: ink }}>

        {children}
      </section>

  }

  var createCode =
    (CREATE_SNIPPETS[integration] && CREATE_SNIPPETS[integration][lang]) ||
    (CREATE_SNIPPETS.hosted && CREATE_SNIPPETS.hosted[lang]) ||
    '';
  var hashCode = (HASH_SNIPPETS[lang] && HASH_SNIPPETS[lang].code) || '';
  var verifyCode = VERIFY_SNIPPETS[lang] || '';

  var shell = {
    border: '1px solid ' + line,
    borderRadius: '14px',
    background: '#fff',
    padding: '20px',
    margin: '20px 0',
    fontFamily: 'system-ui, -apple-system, Segoe UI, Roboto, sans-serif',
    color: '#111827',
    fontSize: '14px',
    lineHeight: 1.55,
  };

  return (
    <div ref={rootRef} style={shell} role="region" aria-label="Accept your first payment guide">

        Canonical onboarding
      </div>
      <h2 style={{ margin: '0 0 8px', fontSize: '1.45rem', color: ink }}>Accept Your First Payment</h2>

      <div role="progressbar" aria-valuemin={1} aria-valuemax={SECTIONS.length} aria-valuenow={stepIndex + 1} aria-label="Guide progress" style={{ marginBottom: '16px' }}>

          <span>
            {SECTIONS[stepIndex].title} · Step {stepIndex + 1} of {SECTIONS.length}
          </span>
          <span>{pct}%</span>
        </div>
        <div style={{ height: '6px', background: '#e5e7eb', borderRadius: '999px', overflow: 'hidden' }}>
          <div style={{ height: '100%', width: pct + '%', background: brand, transition: 'width 0.25s ease' }} />
        </div>
      </div>

      <div
        style={{
          display: 'flex',
          flexWrap: 'wrap',
          gap: '16px',
          alignItems: 'start',
        }}
      >

            position: 'sticky',
            top: '72px',
            flex: '1 1 200px',
            maxWidth: '240px',
            minWidth: '180px',

            borderRadius: '12px',

            background: '#f9fafb',

          <div style={{ fontSize: '11px', fontWeight: 700, letterSpacing: '0.08em', textTransform: 'uppercase', color: muted, marginBottom: '8px' }}>
            Journey
          </div>
          <ul style={{ listStyle: 'none', margin: 0, padding: 0, display: 'grid', gap: '2px' }}>
            {SECTIONS.map(function (section) {
              var isActive = section.id === activeSection;
              return (
                <li key={section.id}>
                  <button
                    type="button"
                    onClick={function () {
                      jump(section.id);
                    }}
                    aria-current={isActive ? 'true' : undefined}
                    style={{
                      width: '100%',
                      textAlign: 'left',
                      padding: '7px 10px',
                      borderRadius: '8px',
                      border: 'none',
                      background: isActive ? 'rgba(0,166,81,.1)' : 'transparent',
                      color: isActive ? '#067a3d' : '#374151',
                      fontWeight: isActive ? 700 : 500,
                      fontSize: '12.5px',
                      cursor: 'pointer',
                      borderLeft: '3px solid ' + (isActive ? brand : 'transparent'),

                    {section.title}

        <div style={{ flex: '1 1 320px', minWidth: 0 }}>

              <p style={{ margin: 0, padding: '12px 14px', borderRadius: '10px', background: 'rgba(0,166,81,.06)', border: '1px solid rgba(0,166,81,.2)', color: '#374151' }}>

              </p>
              <div role="tablist" aria-label="Programming language" style={{ display: 'flex', flexWrap: 'wrap', gap: '6px' }}>
                {LANGS.map(function (item) {

                      key={item.id}

                        selectLang(item.id);

            'create-payment-request',
            'Create Payment Request',

              <p style={{ margin: '0 0 8px', color: muted }}>{content.createHint}</p>

            'handle-payment-response',

              <ul style={{ margin: 0, paddingLeft: '18px', color: '#374151' }}>

                </li>
                <li>
                  <strong>Webhooks</strong> — configure server notifications for reliable reconciliation in production.
                </li>
                <li>
                  Treat browser return as a UX signal only. Fulfilment waits for verification.
                </li>
              </ul>
            </div>
          )}

          {sectionShell(
            'verify-payment',
            'Verify Payment',
            <div>
              <p style={{ margin: '0 0 8px', color: muted }}>
                Validate the reverse hash from the return POST, then confirm status with <code>verify_payment</code> before fulfilling the order.
              </p>
              {codePanel('Verify Payment · ' + ((LANGS.find(function (l) { return l.id === lang; }) || {}).label || lang), 'verify', verifyCode)}
            </div>
          )}

          {sectionShell(
            'test',
            'Test',
            <ul style={{ margin: 0, paddingLeft: '18px', color: '#374151' }}>
              <li>Complete a successful Test Mode payment for your selected integration.</li>
              <li>Confirm surl receives the expected txnid and amount.</li>
              <li>Force a failure path and confirm furl / failure handling.</li>
              <li>Confirm reverse hash validation and verify_payment both pass.</li>
              <li>Use a fresh txnid for every attempt.</li>
            </ul>
          )}

          {sectionShell(
            'go-live',
            'Go Live',
            <ol style={{ margin: 0, paddingLeft: '18px', color: '#374151' }}>
              <li>Switch endpoints from <code>https://test.payu.in</code> to production (<code>https://secure.payu.in</code> for hosted).</li>
              <li>Replace test key/salt with production credentials on secure servers only.</li>
              <li>Enable webhooks and monitor surl/furl error rates.</li>
              <li>Run a small live transaction before opening full traffic.</li>
            </ol>
          )}

          {sectionShell(
            'next-steps',
            'Next Steps',
            <div>
              <p style={{ margin: '0 0 12px', color: muted }}>
                Branch into the product-specific guide for your selected path. These are continuations of this journey — not separate starting points.
              </p>
              <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(220px, 1fr))', gap: '12px' }}>
                {content.nextCards.map(function (card) {
                  return (
                    <a
                      key={card.id}
                      href={card.href}
                      style={{
                        display: 'block',
                        textDecoration: 'none',
                        color: 'inherit',
                        border: '1px solid ' + line,
                        borderRadius: '12px',
                        padding: '14px',
                        background: '#f9fafb',

                      <div style={{ fontWeight: 700, color: ink, marginBottom: '6px' }}>{card.title}</div>

                    </a>

                })}
              </div>
              <p style={{ margin: '14px 0 0', fontSize: '13px' }}>
                Primary docs for this path:{' '}
                <a href={content.docsUrl} style={{ color: brand, fontWeight: 700 }}>
                  {content.docsLabel}
                </a>
              </p>
            </div>
          )}
        </div>
      </div>

      <span role="status" aria-live="polite" style={{ position: 'absolute', width: 1, height: 1, overflow: 'hidden', clip: 'rect(0,0,0,0)' }}>
        {copied ? 'Copied to clipboard' : ''}
      </span>
    </div>
  );
