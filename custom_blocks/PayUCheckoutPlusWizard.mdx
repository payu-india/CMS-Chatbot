  // ─── Utilities ────────────────────────────────────────────────────────────────
  function empty(s) { return s == null ? '' : String(s).trim(); }
  function isBlank(s) { return empty(s) === ''; }

  function escapeJsString(s) {
    return String(s||'').replace(/\\/g,'\\\\').replace(/'/g,"\\'");
  }

  function sha512Hex(msg) {
    var enc = new TextEncoder();
    return crypto.subtle.digest('SHA-512', enc.encode(msg)).then(function(buf) {
      return Array.from(new Uint8Array(buf)).map(function(b){ return b.toString(16).padStart(2,'0'); }).join('');
    });
  }

  function buildHashString(params, saltVal) {
    var p = params;
    return (
      empty(p.key)+'|'+empty(p.txnid)+'|'+empty(p.amount)+'|'+empty(p.productinfo)+'|'+
      empty(p.firstname)+'|'+empty(p.email)+'|'+empty(p.udf1)+'|'+empty(p.udf2)+'|'+
      empty(p.udf3)+'|'+empty(p.udf4)+'|'+empty(p.udf5)+'||||||'+empty(saltVal)
    );
  }

  function validateFields(data) {
    var errors = {};
    var p = data || {};
    if (isBlank(p.key)) errors.key = 'Add your test merchant key from the PayU dashboard.';
    var i;
    for (i=0; i<MANDATORY_KEYS.length; i++) {
      if (isBlank(p[MANDATORY_KEYS[i]])) errors[MANDATORY_KEYS[i]] = MANDATORY_KEYS[i]+' is required.';
    }
    if (!isBlank(p.amount) && !/^\d+(\.\d{1,2})?$/.test(empty(p.amount))) {
      errors.amount = 'Use a decimal string like 10.00';
    }
    if (!isBlank(p.email) && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(empty(p.email))) {
      errors.email = 'Not a valid email address.';
    }
    if (!isBlank(p.phone) && !/^\d{10}$/.test(empty(p.phone).replace(/\s/g,''))) {
      errors.phone = 'Phone should be 10 digits.';
    }
    if (!isBlank(p.txnid) && empty(p.txnid).length > 25) {
      errors.txnid = 'txnid should be under 25 characters.';
    }
    return errors;
  }

  function mergeParams(params) {
    var merged = Object.assign({}, DEFAULT_PARAMS);
    var p = params || {};
    var k;
    for (k in merged) {
      if (!Object.prototype.hasOwnProperty.call(merged,k)) continue;
      if (k==='key') { merged.key = isBlank(p.key) ? '' : String(p.key); continue; }
      if (!isBlank(p[k])) merged[k] = String(p[k]);
    }
    return merged;
  }

  // ─── Storage ──────────────────────────────────────────────────────────────────
  function labWindow() {
    if (typeof window === 'undefined') return null;
    try { if (window.top && window.top !== window) return window.top; } catch(e) {}
    return window;
  }

  function defaultProgress() {
    return { paramsSaved:false, pipeShown:false, hashGenerated:false, launchAck:false };
  }

  function wizardFresh() {
    return { params:Object.assign({},DEFAULT_PARAMS), salt:'', hashHex:'', pipeString:'', progress:defaultProgress(), wizardStep:0 };
  }

  function hydrateStored(raw) {
    if (!raw || typeof raw !== 'object') return null;
    var params = mergeParams(raw.params || {});
    params.key = '';
    return { params:params, salt:'', hashHex:'', pipeString:'', progress:defaultProgress(), wizardStep:0 };
  }

  function labRead() {
    try {
      var lw = labWindow();
      if (!lw) return null;
      if (lw[LAB_GLOBAL_STATE]) return Object.assign({}, lw[LAB_GLOBAL_STATE]);
      if (lw.localStorage) {
        var raw = lw.localStorage.getItem(LAB_KEY);
        if (raw) return hydrateStored(JSON.parse(raw));
      }
      return null;
    } catch(e) { return null; }
  }

  function labWrite(data) {
    try {
      if (typeof window === 'undefined') return false;
      var snapshot = Object.assign({}, data, { updatedAt:Date.now() });
      var lw = labWindow();
      if (lw) lw[LAB_GLOBAL_STATE] = snapshot;
      var persisted = { params:mergeParams(snapshot.params), updatedAt:snapshot.updatedAt };
      persisted.params.key = '';
      var payload = JSON.stringify(persisted);
      if (lw && lw.localStorage) lw.localStorage.setItem(LAB_KEY, payload);
      var evt = new CustomEvent(LAB_UPDATE_EVENT, { detail:snapshot });
      if (typeof window !== 'undefined') window.dispatchEvent(evt);
      return true;
    } catch(e2) { return false; }
  }

  function labEnsure() {
    var existing = labRead();
    if (existing) {
      if (!existing.progress) { var r = wizardFresh(); labWrite(r); return r; }
      existing.params = mergeParams(existing.params);
      return existing;
    }
    return wizardFresh();
  }

  // ─── Body scroll lock ─────────────────────────────────────────────────────────
  var LOCK_ATTR = 'data-payu-cp-wizard-lock';
  function lockScroll() {
    if (typeof document === 'undefined') return;
    var c = parseInt(document.body.getAttribute(LOCK_ATTR)||'0',10)+1;
    document.body.setAttribute(LOCK_ATTR,String(c));
    if (c===1) document.body.style.overflow='hidden';
  }
  function unlockScroll() {
    if (typeof document === 'undefined') return;
    var c = parseInt(document.body.getAttribute(LOCK_ATTR)||'0',10);
    if (c<=1) { document.body.removeAttribute(LOCK_ATTR); document.body.style.overflow=''; }
    else document.body.setAttribute(LOCK_ATTR,String(c-1));

  useEffect(function() {
    if (!modalOpen || typeof document === 'undefined') return;
    var raf = 0;
    function onKey(e) {
      if (e.key==='Escape') { closeModal(); return; }
      if (e.key!=='Tab' || !modalPanelRef.current) return;
      var els = getFocusable(modalPanelRef.current);
      if (!els.length) return;
      var first=els[0], last=els[els.length-1];
      if (e.shiftKey) {
        if (document.activeElement===first || !modalPanelRef.current.contains(document.activeElement)) { e.preventDefault(); last.focus(); }
      } else if (document.activeElement===last || !modalPanelRef.current.contains(document.activeElement)) {
        e.preventDefault(); first.focus();
      }
    }
    lockScroll();
    document.addEventListener('keydown', onKey);
    raf = requestAnimationFrame(function(){ if(closeBtnRef.current) closeBtnRef.current.focus(); });
    return function() {
      cancelAnimationFrame(raf);
      document.removeEventListener('keydown', onKey);
      unlockScroll();
      if (launcherBtnRef.current) launcherBtnRef.current.focus();
    };
  }, [modalOpen]);

  function renderInsightBody(text) {
    var parts=String(text||'').split('**');
    return parts.map(function(part,idx){ if(!part) return null; return idx%2===1?<strong key={idx}>{part}</strong>:<span key={idx}>{part}</span>; });

  function helpBorder(sev) { if(sev==='error') return '1px solid #fecaca'; if(sev==='warning') return '1px solid #fde68a'; if(sev==='success') return '1px solid rgba(15,118,110,0.35)'; return '1px solid #e5e7eb'; }
  function helpBg(sev) { if(sev==='error') return '#fef2f2'; if(sev==='warning') return '#fffbeb'; if(sev==='success') return PAYU_GREEN_LIGHT; return '#fff'; }

  function buildInsights() {
    var stepId = STEPS[step] ? STEPS[step].id : 'welcome';
    var insights = [];
    insights.push({ id:'next', title:'Do this next', body:STEP_NEXT_ACTIONS[stepId]||'Follow the current step.', sev:'info' });
    if(!canAdvance()) {
      var hint=continueHint();
      if(hint) insights.push({ id:'blocker', title:'Cannot continue yet', body:hint, sev:'warning' });
    }
    if(stepId==='hash' && wizardProgress().hashGenerated) {
      insights.push({ id:'server', title:'Production reminder', body:'Generate the hash on your **backend server**, not in the browser. Never include the merchant salt in frontend JavaScript.', sev:'info' });
    }
    if(stepId==='response') {
      insights.push({ id:'verify', title:'Server-side verification is required', body:'After every SUCCESS in responseHandler(), call the Verify Payment API from your backend before fulfilling the order. The txnStatus in the browser can be spoofed.', sev:'warning' });
    }
    if(stepId==='golive') {
      insights.push({ id:'smoke', title:'Run a smoke test', body:'After completing the checklist, make one real end-to-end payment with a small amount before opening to full traffic.', sev:'info' });
    }
    return insights;
  }
