// SearchableTableSmall - Readme CMS Custom Component
// Register in Readme as component name: SearchableTableSmall
// Small tables: tableKey for embedded presets, or rowsJson on doc pages when short enough.
// Regenerate presets: npm run build:searchable-table:small
// Do not put curly braces inside line comments in this file.

import { useState, useMemo } from 'react';

export const SearchableTableSmall = (props) => {
  const demoHeadersJson = '["Bank","Type","Status"]';
  const demoRowsJson =
    '[["HDFC Bank","Netbanking / Cards","Active"],["ICICI Bank","Netbanking / Cards","Active"],["State Bank of India","Netbanking","Active"],["Axis Bank","Cards","Active"]]';

  const {
    headers = [],
    headersJson = '',
    rows = [],
    rowsJson = '',
    tableKey = '',
    codeColumns = null,
    codeColumnsJson = '',
    placeholder = 'Search in table...',
    maxHeight = '400px',
    emptyMessage = 'No results found. Try a different search.',
    className = '',
  } = props;

  const [searchQuery, setSearchQuery] = useState('');

  // PRESET_LOOKUP_START
  const lookupPreset = (key) => {
  const k = key && String(key).trim();
  if (!k) return null;
  if (k === 'netBankingCodes') {
    return {
      headersJson: "[\"Bank Name\",\"bankcode\"]",
      rowsJson: "[[\"Airtel Payments Bank\",\"AIRNB\"],[\"AU Small Finance Bank\",\"AUSFNB\"],[\"AU Small Finance Bank - Corporate\",\"AUSFCNB\"],[\"AXIS Bank\",\"AXIB\"],[\"Axis Corporate Netbanking\",\"AXISCNB\"],[\"Bandhan Bank\",\"BANDNB\"],[\"Bank of Baroda\",\"BBRB\"],[\"Bank of India -  Corporate\",\"BOIB\"],[\"Bank of India  - Retail\",\"BOINB\"],[\"Bank of Maharashtra\",\"BOMB\"],[\"Canara Bank\",\"CABB\"],[\"Canara Bank (Erstwhile - Syndicate Bank)\",\"SYNDB\"],[\"Capital Small Finance Bank Corporate\",\"CSFBC\"],[\"Capital Small Finance Bank Retail\",\"CSFBR\"],[\"Catholic Syrian Bank\",\"CSBN\"],[\"Central Bank Of India\",\"CBIB\"],[\"City Union Bank\",\"CUBB\"],[\"Cosmos Bank\",\"CSMSNB\"],[\"DCB Bank\",\"DCBB\"],[\"Deutsche Bank\",\"DSHB\"],[\"Dhanlaxmi Bank - Corporate\",\"DLNBCORP\"],[\"Dhanlaxmi Bank - Retail\",\"DLSB\"],[\"Dhanlaxmi Bank - TPV\",\"DLSNBTPV\"],[\"Federal Bank\",\"FEDB\"],[\"Federal Bank Corporate\",\"FEDCORP\"],[\"HDFC Bank\",\"HDFB\"],[\"HDFC Bank - Corporate Banking\",\"HDFCCONB\"],[\"ICICI Bank\",\"ICIB\"],[\"ICICI Corporate Netbanking\",\"ICICICNB\"],[\"ICICI Corporate Netbanking TPV\",\"ICICTPV\"],[\"IDBI Bank\",\"IDBB\"],[\"IDBI Corp Netbanking\",\"IDBICORP\"],[\"IDFC FIRST Bank\",\"IDFCNB\"],[\"Indian Bank\",\"INDB\"],[\"Indian Bank (Erstwhile Allahabad Bank)\",\"ALLB\"],[\"Indian Overseas Bank\",\"INOB\"],[\"IndusInd Bank\",\"INIB\"],[\"Jammu & Kashmir Bank\",\"JAKB\"],[\"Jana Small Finance Bank\",\"JANANB\"],[\"Janata Sahakari Bank Pune\",\"JSBNB\"],[\"Karnataka Bank\",\"KRKB\"],[\"Karnataka Gramin Bank\",\"KGBNB\"],[\"Karur Vysya - Corporate Banking\",\"KRVBC\"],[\"Karur Vysya Bank\",\"KRVB\"],[\"Kerala Gramin Bank\",\"KEGBNB\"],[\"Kotak Mahindra Bank\",\"162B\"],[\"Kotak Mahindra Bank - Corp Net Banking\",\"KTKBCORP\"],[\"Karur Vysya Bank TPV\",\"KVBNBTPV\"],[\"PNB (Erstwhile -Oriental Bank of Commerce)\",\"OBCB\"],[\"PNB (Erstwhile-United Bank of India)\",\"UNIB\"],[\"Punjab & Sind Bank\",\"PSBNB\"],[\"Punjab National Bank\",\"PNBB\"],[\"Punjab National Bank - Corporate Banking\",\"CPNB\"],[\"RBL Bank\",\"RBLNB\"],[\"RBL Corporate Netbanking\",\"RBLCNB\"],[\"Saraswat Bank\",\"SRSWT\"],[\"Shivalik Small Finance Bank\",\"SHIVANB\"],[\"South Indian Bank\",\"SOIB\"],[\"Standard Chartered Bank\",\"SCBNB\"],[\"State Bank of India\",\"SBIB\"],[\"State Bank of India (Corporate)\",\"SBNCORP\"],[\"SVC Co-operative Bank Ltd.\",\"SVCNB\"],[\"SVC Corporate Bank TPV\",\"SVCCBTPV\"],[\"Tamilnad Mercantile Bank\",\"TMBB\"],[\"UCO Bank\",\"UCOB\"],[\"UCO Corporate\",\"UCOCNB\"],[\"Yes Bank\",\"YESB\"]]",
      codeColumnsJson: "[1]",
    };
  }
  return null;
};
  // PRESET_LOOKUP_END

  const preset = useMemo(() => lookupPreset(tableKey), [tableKey]);

  const coerceRows = (input) => {
    if (input == null || input === '') return [];
    if (Array.isArray(input)) {
      return input.filter((row) => Array.isArray(row));
    }
    if (typeof input === 'string') {
      const text = input.trim();
      if (!text) return [];
      try {
        return coerceRows(JSON.parse(text));
      } catch (e) {
        return [];
      }
    }
    if (typeof input === 'object') {
      if (Array.isArray(input.rows)) return coerceRows(input.rows);
      if (typeof input.length === 'number' && input.length >= 0) {
        const list = [];
        for (let i = 0; i < input.length; i += 1) list.push(input[i]);
        return coerceRows(list);
      }
    }
    return [];
  };

  const parseJsonProp = (text, fallback) => {
    if (text == null || text === '') return fallback;
    if (typeof text !== 'string') return text;
    const trimmed = text.trim();
    if (!trimmed) return fallback;
    try {
      return JSON.parse(trimmed);
    } catch (e) {
      return fallback;
    }
  };

  const resolvedHeaders = useMemo(() => {
    const fromJson = parseJsonProp(headersJson, null);
    if (Array.isArray(fromJson) && fromJson.length > 0) {
      return fromJson.map((h) => String(h == null ? '' : h));
    }
    if (preset && preset.headersJson) {
      const fromPreset = parseJsonProp(preset.headersJson, []);
      if (fromPreset.length > 0) {
        return fromPreset.map((h) => String(h == null ? '' : h));
      }
    }
    if (Array.isArray(headers) && headers.length > 0) {
      return headers.map((h) => String(h == null ? '' : h));
    }
    return parseJsonProp(demoHeadersJson, []).map((h) =>
      String(h == null ? '' : h)
    );
  }, [headers, headersJson, preset, demoHeadersJson]);

  const resolvedCodeColumns = useMemo(() => {
    const fromJson = parseJsonProp(codeColumnsJson, null);
    if (Array.isArray(fromJson)) return fromJson;
    if (Array.isArray(codeColumns)) return codeColumns;
    if (preset && preset.codeColumnsJson) {
      const fromPreset = parseJsonProp(preset.codeColumnsJson, null);
      if (Array.isArray(fromPreset)) return fromPreset;
    }
    return [];
  }, [codeColumns, codeColumnsJson, preset]);

  const tableRows = useMemo(() => {
    const fromRows = coerceRows(rows);
    if (fromRows.length > 0) return fromRows;

    const fromJson = coerceRows(rowsJson);
    if (fromJson.length > 0) return fromJson;

    if (preset && preset.rowsJson) {
      const fromPreset = coerceRows(preset.rowsJson);
      if (fromPreset.length > 0) return fromPreset;
    }

    return coerceRows(demoRowsJson);
  }, [rows, rowsJson, preset, demoRowsJson]);

  const codeColIndexes = useMemo(() => {
    if (resolvedCodeColumns.length > 0) return resolvedCodeColumns;
    const idx = resolvedHeaders.findIndex((h) =>
      /error|code/i.test(String(h == null ? '' : h))
    );
    return idx >= 0 ? [idx] : resolvedHeaders.length > 0 ? [0] : [];
  }, [resolvedCodeColumns, resolvedHeaders]);

  const isCodeCol = (i) => codeColIndexes.indexOf(i) >= 0;

  const normalize = (str) =>
    str == null ? '' : String(str).toLowerCase().trim().replace(/\s+/g, ' ');

  const rowMatchesQuery = (row, query) => {
    const q = normalize(query);
    if (!q) return true;
    const words = q.split(/\s+/).filter(Boolean);
    const rowText = row.map((cell) => normalize(cell)).join(' ');
    return words.every((word) => rowText.includes(word));
  };

  const filteredRows = useMemo(
    () => tableRows.filter((row) => rowMatchesQuery(row, searchQuery)),
    [tableRows, searchQuery]
  );

  const hasSearchQuery = normalize(searchQuery).length > 0;
  const showEmptyState = hasSearchQuery && filteredRows.length === 0;
  const showNoRows = tableRows.length === 0 && !showEmptyState;
  const colCount = resolvedHeaders.length;

  const fontSans =
    '-apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, "Helvetica Neue", Arial, sans-serif';
  const fontMono =
    'ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, monospace';
  const borderColor = '#e5e7eb';

  const codeStyle = {
    padding: '2px 6px',
    background: '#f3f4f6',
    color: '#1f2937',
    borderRadius: '4px',
    fontFamily: fontMono,
    fontSize: '13px',
    fontWeight: '400',
  };

  const renderCellContent = (value, colIndex) => {
    if (value == null) return '';
    if (typeof value !== 'string') return value;
    const plain = value.indexOf('`') >= 0 ? value.replace(/`/g, '') : value;
    return isCodeCol(colIndex) ? (
      <code style={codeStyle}>{plain}</code>
    ) : (
      plain
    );
  };

  const headerCellStyle = (colIndex, colTotal) => ({
    padding: '12px 16px',
    textAlign: 'left',
    fontFamily: fontSans,
    fontSize: '12px',
    fontWeight: '700',
    color: '#374151',
    textTransform: 'uppercase',
    letterSpacing: '0.05em',
    backgroundColor: '#f9fafb',
    borderBottom: '1px solid ' + borderColor,
    borderRight:
      colIndex < colTotal - 1 ? '1px solid ' + borderColor : 'none',
    borderTop: 'none',
    borderLeft: 'none',
    verticalAlign: 'middle',
    position: 'sticky',
    top: 0,
    zIndex: 10,
    boxSizing: 'border-box',
    whiteSpace: 'nowrap',
  });

  const bodyCellStyle = (colIndex, colTotal, isLastRow) => ({
    padding: '12px 16px',
    fontFamily: fontSans,
    fontSize: '14px',
    fontWeight: '400',
    lineHeight: '1.5',
    color: '#111827',
    backgroundColor: '#ffffff',
    borderBottom: isLastRow ? 'none' : '1px solid ' + borderColor,
    borderRight:
      colIndex < colTotal - 1 ? '1px solid ' + borderColor : 'none',
    borderTop: 'none',
    borderLeft: 'none',
    verticalAlign: 'top',
    boxSizing: 'border-box',
    whiteSpace: 'nowrap',
  });

  const messageCellStyle = {
    padding: '16px',
    fontFamily: fontSans,
    fontSize: '14px',
    color: '#6b7280',
    border: 'none',
    boxSizing: 'border-box',
  };

  return (
    <div
      className={className}
      style={{
        display: 'inline-block',
        width: 'max-content',
        maxWidth: '100%',
        verticalAlign: 'top',
        fontFamily: fontSans,
        fontSize: '14px',
        color: '#111827',
      }}
    >
      <div
        style={{
          display: 'inline-block',
          width: 'max-content',
          maxWidth: '100%',
          border: '1px solid ' + borderColor,
          borderRadius: '6px',
          overflow: 'hidden',
          backgroundColor: '#ffffff',
          boxSizing: 'border-box',
        }}
      >
        <div
          style={{
            padding: '12px 16px',
            backgroundColor: '#f9fafb',
            borderBottom: '1px solid ' + borderColor,
            boxSizing: 'border-box',
            width: '100%',
          }}
        >
          <div
            style={{
              display: 'flex',
              alignItems: 'center',
              width: '100%',
              minWidth: '240px',
              border: '1px solid #d1d5db',
              borderRadius: '6px',
              backgroundColor: '#ffffff',
              boxShadow: '0 1px 2px 0 rgba(0,0,0,0.05)',
              boxSizing: 'border-box',
            }}
          >
            <span
              aria-hidden="true"
              style={{
                display: 'flex',
                alignItems: 'center',
                paddingLeft: '12px',
                color: '#9ca3af',
                flexShrink: 0,
              }}
            >
              <svg
                xmlns="http://www.w3.org/2000/svg"
                width="18"
                height="18"
                viewBox="0 0 24 24"
                fill="none"
                stroke="currentColor"
                strokeWidth="2"
                strokeLinecap="round"
                strokeLinejoin="round"
              >
                <circle cx="11" cy="11" r="8" />
                <path d="m21 21-4.35-4.35" />
              </svg>
            </span>
            <input
              type="search"
              placeholder={placeholder}
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              aria-label="Search table"
              style={{
                flex: 1,
                minWidth: 0,
                padding: '8px 12px 8px 8px',
                border: 0,
                backgroundColor: 'transparent',
                outline: 'none',
                fontSize: '14px',
                fontFamily: fontSans,
                color: '#111827',
              }}
            />
          </div>
        </div>

        <div
          style={{
            overflowY: 'auto',
            overflowX: 'auto',
            maxHeight: showEmptyState ? 'none' : maxHeight,
            maxWidth: '100%',
            boxSizing: 'border-box',
          }}
        >
          <table
            style={{
              width: 'auto',
              borderCollapse: 'collapse',
              borderSpacing: 0,
              margin: 0,
              tableLayout: 'auto',
              fontFamily: fontSans,
            }}
          >
            <thead>
              <tr>
                {resolvedHeaders.map((header, i) => (
                  <th
                    key={i}
                    scope="col"
                    style={headerCellStyle(i, resolvedHeaders.length)}
                  >
                    {renderCellContent(header, i)}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody>
              {showNoRows ? (
                <tr>
                  <td colSpan={colCount || 1} style={messageCellStyle}>
                    No rows loaded. Use tableKey on the page, or pass rowsJson.
                    Re-paste SearchableTableSmall.jsx after npm run build:searchable-table:small.
                  </td>
                </tr>
              ) : showEmptyState ? (
                <tr>
                  <td
                    colSpan={colCount || 1}
                    role="alert"
                    style={{
                      ...messageCellStyle,
                      backgroundColor: '#fef2f2',
                      color: '#b91c1c',
                    }}
                  >
                    {emptyMessage}
                  </td>
                </tr>
              ) : (
                filteredRows.map((row, rowIndex) => (
                  <tr key={rowIndex}>
                    {row.map((cell, cellIndex) => (
                      <td
                        key={cellIndex}
                        style={bodyCellStyle(
                          cellIndex,
                          row.length,
                          rowIndex === filteredRows.length - 1
                        )}
                      >
                        {renderCellContent(cell, cellIndex)}
                      </td>
                    ))}
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>

      {!showNoRows && filteredRows.length > 0 ? (
        <p
          style={{
            marginTop: '8px',
            fontSize: '12px',
            fontFamily: fontSans,
            color: '#6b7280',
          }}
        >
          Showing {filteredRows.length} of {tableRows.length} row
          {tableRows.length !== 1 ? 's' : ''}
          {hasSearchQuery ? ' (filtered by search)' : ''}.
        </p>
      ) : null}
    </div>
  );
};


<SearchableTableSmall tableKey="netBankingCodes" placeholder="Search" />
