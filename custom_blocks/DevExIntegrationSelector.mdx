// DevExIntegrationSelector - Accessible radio-group for choosing an integration path.
// Props:
//   optionsJson: JSON array of { id, label, description, recommended? }
//   value: controlled selected id
//   legend: fieldset legend text
//   onChangeId: optional callback (id) when used inside a parent React tree
// Usage:
//   <DevExIntegrationSelector value="hosted" />

import { useState } from 'react';

export const DevExIntegrationSelector = (props) => {
  var DEFAULT_OPTIONS = JSON.parse(
    '[{"id":"hosted","label":"Hosted Checkout","description":"Redirect to PayU-hosted payment page. Fastest go-live, no PCI on your side.","recommended":true},{"id":"seamless","label":"Seamless","description":"Merchant-hosted checkout UI on your site. Full control; PCI required for cards."},{"id":"commercepro","label":"CommercePro","description":"Conversion-optimized checkout. PayU handles sensitive card data."},{"id":"sdk","label":"Mobile SDK","description":"Native or cross-platform Checkout Pro SDKs for Android, iOS, RN, Flutter."},{"id":"plugins","label":"Plugins","description":"Shopify, WooCommerce, Magento, and other eCommerce platform plugins."},{"id":"other","label":"Other","description":"Server-to-Server, Payment Links, UPI QR, or specialized flows."}]'
  );

  function parseOptions(raw) {
    if (!raw) return DEFAULT_OPTIONS;
    try {
      var parsed = typeof raw === 'string' ? JSON.parse(raw) : raw;
      return Array.isArray(parsed) && parsed.length ? parsed : DEFAULT_OPTIONS;
    } catch (e) {
      return DEFAULT_OPTIONS;
    }
  }

  var options = parseOptions(props.optionsJson);
  var legend = props.legend != null ? String(props.legend) : 'How are you integrating?';
  var _value = useState(props.value || options[0].id);
  var value = props.value != null ? props.value : _value[0];
  var setValue = _value[1];

  function select(id) {
    if (props.value == null) setValue(id);
    if (typeof props.onChangeId === 'function') props.onChangeId(id);
  }

  function onKeyDown(e, index) {
    var next = index;
    if (e.key === 'ArrowDown' || e.key === 'ArrowRight') {
      e.preventDefault();
      next = (index + 1) % options.length;
    } else if (e.key === 'ArrowUp' || e.key === 'ArrowLeft') {
      e.preventDefault();
      next = (index - 1 + options.length) % options.length;
    } else if (e.key === ' ' || e.key === 'Enter') {
      e.preventDefault();
      select(options[index].id);
      return;
    } else {
      return;
    }
    select(options[next].id);
    var btn = e.currentTarget.parentNode.querySelector('[data-opt-index="' + next + '"]');
    if (btn) btn.focus();
  }

  return (
    <fieldset
      style={{
        border: '1px solid #e5e7eb',
        borderRadius: '12px',
        padding: '16px',
        margin: '16px 0',
        background: '#fff',
        fontFamily: 'system-ui, -apple-system, Segoe UI, Roboto, sans-serif',
      }}
    >
      <legend
        style={{
          padding: '0 8px',
          fontSize: '15px',
          fontWeight: 700,
          color: '#1e3a5f',
        }}
      >
        {legend}
      </legend>
      <div role="radiogroup" aria-label={legend} style={{ display: 'grid', gap: '8px' }}>
        {options.map(function (opt, index) {
          var selected = opt.id === value;
          return (
            <button
              key={opt.id}
              type="button"
              role="radio"
              aria-checked={selected}
              data-opt-index={index}
              tabIndex={selected ? 0 : -1}
              onClick={function () {
                select(opt.id);
              }}
              onKeyDown={function (e) {
                onKeyDown(e, index);
              }}
              style={{
                display: 'flex',
                gap: '12px',
                alignItems: 'flex-start',
                textAlign: 'left',
                width: '100%',
                padding: '12px 14px',
                borderRadius: '10px',
                border: '1px solid ' + (selected ? '#00A651' : '#e5e7eb'),
                background: selected ? 'rgba(0,166,81,.06)' : '#f9fafb',
                cursor: 'pointer',
                color: '#1e3a5f',
              }}
            >
              <span
                aria-hidden="true"
                style={{
                  width: '18px',
                  height: '18px',
                  marginTop: '2px',
                  borderRadius: '50%',
                  border: '2px solid ' + (selected ? '#00A651' : '#9ca3af'),
                  boxSizing: 'border-box',
                  display: 'inline-flex',
                  alignItems: 'center',
                  justifyContent: 'center',
                  flexShrink: 0,
                }}
              >
                {selected ? (
                  <span
                    style={{
                      width: '8px',
                      height: '8px',
                      borderRadius: '50%',
                      background: '#00A651',
                    }}
                  />
                ) : null}
              </span>
              <span style={{ flex: 1, minWidth: 0 }}>
                <span style={{ display: 'flex', flexWrap: 'wrap', gap: '8px', alignItems: 'center' }}>
                  <span style={{ fontWeight: 600, fontSize: '14px' }}>{opt.label}</span>
                  {opt.recommended ? (
                    <span
                      style={{
                        fontSize: '11px',
                        fontWeight: 700,
                        color: '#067a3d',
                        background: 'rgba(0,166,81,.12)',
                        padding: '2px 8px',
                        borderRadius: '6px',
                      }}
                    >
                      Recommended
                    </span>
                  ) : null}
                </span>
                <span style={{ display: 'block', marginTop: '4px', fontSize: '12px', color: '#6b7280' }}>
                  {opt.description}
                </span>
              </span>
            </button>
          );
        })}
      </div>
    </fieldset>
  );
};

<DevExIntegrationSelector />
