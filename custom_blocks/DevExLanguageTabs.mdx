// DevExLanguageTabs - Instant language switcher that preserves scroll position.
// Props:
//   languagesJson: JSON array of { id, label }
//   value: controlled language id
//   storageKey: optional localStorage key to remember preference
// Usage:
//   <DevExLanguageTabs value="nodejs" storageKey="payu-devex-lang" />

import { useEffect, useState } from 'react';

export const DevExLanguageTabs = (props) => {
  var DEFAULT_LANGS = JSON.parse(
    '[{"id":"nodejs","label":"Node.js"},{"id":"java","label":"Java"},{"id":"php","label":"PHP"},{"id":"python","label":"Python"},{"id":"go","label":"Go"},{"id":"dotnet","label":".NET"}]'
  );

  function parseLangs(raw) {
    if (!raw) return DEFAULT_LANGS;
    try {
      var parsed = typeof raw === 'string' ? JSON.parse(raw) : raw;
      return Array.isArray(parsed) && parsed.length ? parsed : DEFAULT_LANGS;
    } catch (e) {
      return DEFAULT_LANGS;
    }
  }

  var languages = parseLangs(props.languagesJson);
  var storageKey = props.storageKey ? String(props.storageKey) : '';

  function initialValue() {
    if (props.value != null) return props.value;
    if (storageKey && typeof window !== 'undefined') {
      try {
        var saved = window.localStorage.getItem(storageKey);
        if (saved) {
          for (var i = 0; i < languages.length; i += 1) {
            if (languages[i].id === saved) return saved;
          }
        }
      } catch (e) {
        // ignore
      }
    }
    return languages[0].id;
  }

  var _value = useState(initialValue);
  var value = props.value != null ? props.value : _value[0];
  var setValue = _value[1];

  useEffect(
    function () {
      if (props.value != null) return;
      if (!storageKey || typeof window === 'undefined') return;
      try {
        window.localStorage.setItem(storageKey, value);
      } catch (e) {
        // ignore
      }
    },
    [value, storageKey, props.value]
  );

  function select(id) {
    var y = typeof window !== 'undefined' ? window.scrollY : 0;
    if (props.value == null) setValue(id);
    if (typeof props.onChangeId === 'function') props.onChangeId(id);
    if (typeof window !== 'undefined') {
      requestAnimationFrame(function () {
        window.scrollTo(0, y);
      });
    }
  }

  function onKeyDown(e, index) {
    var next = index;
    if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
      e.preventDefault();
      next = (index + 1) % languages.length;
    } else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
      e.preventDefault();
      next = (index - 1 + languages.length) % languages.length;
    } else if (e.key === 'Home') {
      e.preventDefault();
      next = 0;
    } else if (e.key === 'End') {
      e.preventDefault();
      next = languages.length - 1;
    } else {
      return;
    }
    select(languages[next].id);
    var btn = e.currentTarget.parentNode.querySelector('[data-lang-index="' + next + '"]');
    if (btn) btn.focus();
  }

  return (
    <div
      role="tablist"
      aria-label="Programming language"
      style={{
        display: 'flex',
        flexWrap: 'wrap',
        gap: '6px',
        margin: '12px 0',
        fontFamily: 'system-ui, -apple-system, Segoe UI, Roboto, sans-serif',
      }}
    >
      {languages.map(function (lang, index) {
        var selected = lang.id === value;
        return (
          <button
            key={lang.id}
            type="button"
            role="tab"
            aria-selected={selected}
            id={'devex-lang-' + lang.id}
            data-lang-index={index}
            tabIndex={selected ? 0 : -1}
            onClick={function () {
              select(lang.id);
            }}
            onKeyDown={function (e) {
              onKeyDown(e, index);
            }}
            style={{
              padding: '8px 12px',
              borderRadius: '8px',
              border: '1px solid ' + (selected ? '#00A651' : '#e5e7eb'),
              background: selected ? '#00A651' : '#fff',
              color: selected ? '#fff' : '#374151',
              fontWeight: 600,
              fontSize: '13px',
              cursor: 'pointer',
            }}
          >
            {lang.label}
          </button>
        );
      })}
    </div>
  );
};

<DevExLanguageTabs storageKey="payu-devex-lang-preview" />
