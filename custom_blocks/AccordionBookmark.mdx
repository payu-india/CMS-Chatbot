// AccordionBookmark - ReadMe CMS Custom Component
// Accordion with a real heading anchor (ReadMe bookmark). Chevron is a separate button; title is h2-h6 with id.
// Usage: <AccordionBookmark title="1. What is an SI plan" bookmarkId="what-is-an-si-plan">content</AccordionBookmark>
// Link: #user-content-what-is-an-si-plan (same page) or full doc URL + that hash.
// Props: title, icon, emoji, bookmarkId, defaultOpen, headingLevel.

import { useState } from 'react';

export const AccordionBookmark = (props) => {
  const {
    title = 'Section',
    icon = '',
    emoji = '',
    bookmarkId,
    defaultOpen = false,
    headingLevel = 3,
    children,
  } = props;

  const [isOpen, setIsOpen] = useState(defaultOpen);
  const [contentId] = useState(() => `accordion-bookmark-${Math.random().toString(36).slice(2, 11)}`);

  const autoBookmarkId =
    String(title || 'section')
      .toLowerCase()
      .trim()
      .replace(/\s+/g, '-')
      .replace(/[^a-z0-9-]/g, '')
      .replace(/-+/g, '-')
      .replace(/^-|-$/g, '') || 'section';

  const slug = bookmarkId != null && bookmarkId !== '' ? String(bookmarkId) : autoBookmarkId;
  const anchorId = slug.indexOf('user-content-') === 0 ? slug : `user-content-${slug}`;

  const level = Math.min(6, Math.max(2, Number(headingLevel)));
  const headingClass =
    'flex-1 min-w-0 !my-0 !text-base font-semibold text-gray-800 scroll-mt-6';

  const titleHeading =
    level === 2 ? (
      <h2 id={anchorId} className={headingClass}>
        {title}
      </h2>
    ) : level === 3 ? (
      <h3 id={anchorId} className={headingClass}>
        {title}
      </h3>
    ) : level === 4 ? (
      <h4 id={anchorId} className={headingClass}>
        {title}
      </h4>
    ) : level === 5 ? (
      <h5 id={anchorId} className={headingClass}>
        {title}
      </h5>
    ) : (
      <h6 id={anchorId} className={headingClass}>
        {title}
      </h6>
    );

  return (
    <div className="border border-gray-200 rounded-md overflow-hidden my-4 bg-white">
      <div className="w-full flex items-center gap-3 px-4 py-3 text-left bg-white hover:bg-gray-50 border-b border-gray-200">
        <button
          type="button"
          onClick={() => setIsOpen(!isOpen)}
          className="flex-shrink-0 p-1 -m-1 text-gray-800 rounded focus:outline-none focus:ring-2 focus:ring-blue-500"
          aria-expanded={isOpen}
          aria-controls={contentId}
          aria-label={isOpen ? `Collapse ${title}` : `Expand ${title}`}
        >
          <span
            className={`block transition-transform duration-200 ${isOpen ? 'rotate-90' : ''}`}
            aria-hidden="true"
          >
            <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
              <polyline points="9 18 15 12 9 6" />
            </svg>
          </span>
        </button>
        {emoji ? (
          <span className="flex-shrink-0 text-lg leading-none" aria-hidden="true">
            {emoji}
          </span>
        ) : null}
        {icon ? (
          <span
            className="flex-shrink-0 w-6 h-6 rounded-full flex items-center justify-center text-white text-xs font-bold"
            style={{ backgroundColor: '#8edab6' }}
            aria-hidden="true"
          >
            <i className={`fa ${icon}`} />
          </span>
        ) : null}
        {titleHeading}
      </div>
      <div
        id={contentId}
        className={isOpen ? '' : 'hidden'}
        role="region"
        aria-labelledby={anchorId}
      >
        <div className="px-4 py-3 border-t border-gray-100 bg-white prose prose-sm max-w-none">
          {children}
        </div>
      </div>
    </div>
  );
};

// Default preview for ReadMe CMS (component must appear after exports)

<AccordionBookmark title="Request Parameters" icon="fa-info-circle" bookmarkId="request-parameters">
  <p>Link to this section with #user-content-request-parameters</p>
</AccordionBookmark>
