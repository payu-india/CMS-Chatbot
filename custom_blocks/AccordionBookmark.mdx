// AccordionBookmark - ReadMe CMS Custom Component
// Accordion whose title is a bookmark (heading with anchor id). Supports Font Awesome icons and emojis.
// Usage: <AccordionBookmark title="Request Parameters" icon="fa-info-circle">content</AccordionBookmark>
// Props: title, icon, emoji, bookmarkId, defaultOpen, headingLevel.

import { useState } from 'react';

export const AccordionBookmark = (props) => {
  const {
    title = 'Section',
    icon = '',
    emoji = '',
    bookmarkId,
    defaultOpen = false,
    headingLevel = 3,
    children,
  } = props;

  const [isOpen, setIsOpen] = useState(defaultOpen);
  const [contentId] = useState(() => `accordion-bookmark-${Math.random().toString(36).slice(2, 11)}`);

  const autoBookmarkId =
    String(title || 'section')
      .toLowerCase()
      .trim()
      .replace(/\s+/g, '-')
      .replace(/[^a-z0-9-]/g, '')
      .replace(/-+/g, '-')
      .replace(/^-|-$/g, '') || 'section';

  const resolvedBookmarkId =
    bookmarkId != null && bookmarkId !== '' ? String(bookmarkId) : autoBookmarkId;

  const HeadingTag = `h${Math.min(6, Math.max(2, headingLevel))}`;

  return (
    <div className="border border-gray-200 rounded-md overflow-hidden my-4 bg-white">
      <div
        role="button"
        tabIndex={0}
        onClick={() => setIsOpen(!isOpen)}
        onKeyDown={(event) => {
          if (event.key === 'Enter' || event.key === ' ') {
            event.preventDefault();
            setIsOpen(!isOpen);
          }
        }}
        className="w-full flex items-center gap-3 px-4 py-3 text-left bg-white hover:bg-gray-50 border-b border-gray-200 focus:outline-none focus:ring-2 focus:ring-inset focus:ring-blue-500 cursor-pointer"
        aria-expanded={isOpen}
        aria-controls={contentId}
      >
        <span
          className={`flex-shrink-0 text-gray-800 transition-transform duration-200 ${isOpen ? 'rotate-90' : ''}`}
          aria-hidden="true"
        >
          <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
            <polyline points="9 18 15 12 9 6" />
          </svg>
        </span>
        {emoji ? (
          <span className="flex-shrink-0 text-lg leading-none" aria-hidden="true">
            {emoji}
          </span>
        ) : null}
        {icon ? (
          <span
            className="flex-shrink-0 w-6 h-6 rounded-full flex items-center justify-center text-white text-xs font-bold"
            style={{ backgroundColor: '#8edab6' }}
            aria-hidden="true"
          >
            <i className={`fa ${icon}`} />
          </span>
        ) : null}
        <HeadingTag
          id={resolvedBookmarkId}
          className="flex-1 min-w-0 !my-0 !text-base font-semibold text-gray-800 scroll-mt-6"
        >
          {title}
        </HeadingTag>
      </div>
      <div
        id={contentId}
        className={isOpen ? '' : 'hidden'}
        role="region"
        aria-label={title}
      >
        <div className="px-4 py-3 border-t border-gray-100 bg-white prose prose-sm max-w-none">
          {children}
        </div>
      </div>
    </div>
  );
};

// Default preview for ReadMe CMS (component must appear after exports)

<AccordionBookmark title="Request Parameters" icon="fa-info-circle">
  <p>This accordion title is a bookmark. Link to it with #request-parameters.</p>
</AccordionBookmark>
