    setAnswers(newAnswers);

    if (questionKey === 'Q1') {
      if (answerKey === 'website') {
        setCurrentQuestion('Q2');
      } else if (answerKey === 'payment_link') {
        showResult('payment_links', newAnswers);
      } else if (answerKey === 'in_person') {
        showResult('upi_qr', newAnswers);
      } else if (answerKey === 'recurring') {
        showResult('recurring_payments', newAnswers);
      } else if (answerKey === 'platform') {
        showResult('split_settlement', newAnswers);
      } else if (answerKey === 'not_sure') {
        setCurrentQuestion('Q1_fallback');
      }
    } else if (questionKey === 'Q1_fallback') {
      if (answerKey === 'website') {
        setCurrentQuestion('Q2');
      } else if (answerKey === 'social' || answerKey === 'nothing') {
        showResult('payment_links', newAnswers);
      } else if (answerKey === 'in_person') {
        showResult('upi_qr', newAnswers);
      } else if (answerKey === 'still_unsure') {
        window.open('https://docs.payu.in/', '_blank');
      }
    } else if (questionKey === 'Q2') {
      if (answerKey === 'platform') {
        setCurrentQuestion('Q2_platform');
      } else if (answerKey === 'custom') {
        setCurrentQuestion('Q3');
      } else if (answerKey === 'no_website') {
        showResult('payment_links', newAnswers);
      }

      var pluginMap = {

  function copyToClipboard(text) {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text).then(function() {
        alert('Copied to clipboard!');
      }).catch(function() {
        fallbackCopy(text);
      });
    } else {
      fallbackCopy(text);
    }
  }

  function fallbackCopy(text) {
    var textArea = document.createElement('textarea');
    textArea.value = text;
    textArea.style.position = 'fixed';
    textArea.style.left = '-999999px';
    document.body.appendChild(textArea);
    textArea.focus();
    textArea.select();
    try {
      document.execCommand('copy');
      alert('Copied to clipboard!');
    } catch (err) {
      alert('Could not copy to clipboard');
    }
    document.body.removeChild(textArea);
  }

          </div>
        </section>

        <details style={{ marginTop: '24px', marginBottom: '16px' }}>
          <summary style={{ cursor: 'pointer', fontWeight: '600', fontSize: '14px' }}>

            {whyText} This recommendation is based on your answers about what you're trying to achieve and where you want to accept payments.
          </p>
        </details>

        <button onClick={changeAnswers} style={Object.assign({}, buttonStyle, { backgroundColor: '#f0f0f0', color: '#333', width: 'auto' })} type="button">
