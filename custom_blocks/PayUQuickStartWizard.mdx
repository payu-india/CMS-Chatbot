  function handleAnswer(questionKey, answerKey) {
    var newAnswers = Object.assign({}, answers);
    newAnswers[questionKey] = answerKey;
    setAnswers(newAnswers);

    if (questionKey === 'Q1') {
      if (answerKey === 'website') {
        setCurrentQuestion('Q2');
      } else if (answerKey === 'mobile') {
        showResult('mobile_sdks', newAnswers);
      } else if (answerKey === 'payment_link') {
        showResult('payment_links', newAnswers);
      } else if (answerKey === 'in_person') {
        showResult('upi_qr', newAnswers);
      } else if (answerKey === 'recurring') {
        showResult('recurring_payments', newAnswers);
      } else if (answerKey === 'platform') {
        showResult('split_settlement', newAnswers);
      } else if (answerKey === 'payout') {
        showResult('payouts', newAnswers);
      } else if (answerKey === 'not_sure') {
        setCurrentQuestion('Q1_fallback');
      }
    } else if (questionKey === 'Q1_fallback') {
      if (answerKey === 'website') {
        setCurrentQuestion('Q2');
      } else if (answerKey === 'social' || answerKey === 'nothing') {
        showResult('payment_links', newAnswers);
      } else if (answerKey === 'in_person') {
        showResult('upi_qr', newAnswers);
      } else if (answerKey === 'still_unsure') {
        window.open('https://docs.payu.in/docs/what-can-you-do-next', '_blank');
      }
    } else if (questionKey === 'Q2') {
      if (answerKey === 'platform') {
        setCurrentQuestion('Q2_platform');
      } else if (answerKey === 'custom') {
        setCurrentQuestion('Q3');
      } else if (answerKey === 'no_website') {
        showResult('payment_links', newAnswers);
      }
    } else if (questionKey === 'Q2_platform') {
      var pluginMap = {
        shopify: 'plugin_shopify',
        woocommerce: 'plugin_woocommerce',
        magento: 'plugin_magento',
        other: 'plugin_other'
      };
      showResult(pluginMap[answerKey] || 'plugin_other', newAnswers);
    } else if (questionKey === 'Q3') {
      if (answerKey === 'simple' || answerKey === 'not_sure') {
        showResult('hosted_checkout', newAnswers);
      } else if (answerKey === 'exact_match') {
        showResult('merchant_hosted', newAnswers);
      }
    }
  }

  function showResult(productKey, currentAnswers) {
    var product = products[productKey];
    if (product) {
      setResult({ product: product, answers: currentAnswers });
    }
  }

  function goBackFromResult() {
    if (!result) return;
    setResult(null);
    setShowStep2(false);
    setShowRelated(false);
    setActivePanel('myself');
  }

  function goBack() {
    if (result) {
      setResult(null);
      return;
    }
    var prevQuestion = null;
    if (currentQuestion === 'Q1_fallback') prevQuestion = 'Q1';
    else if (currentQuestion === 'Q2') prevQuestion = answers.Q1 === 'not_sure' ? 'Q1_fallback' : 'Q1';
    else if (currentQuestion === 'Q2_platform') prevQuestion = 'Q2';
    else if (currentQuestion === 'Q3') prevQuestion = 'Q2';
    if (prevQuestion) {
      setCurrentQuestion(prevQuestion);
      var newAnswers = Object.assign({}, answers);
      delete newAnswers[currentQuestion];
      setAnswers(newAnswers);
    }
  }

  function restart() {
    setCurrentQuestion('Q1');
    setAnswers({});
    setResult(null);
    setShowStep2(false);
    setShowRelated(false);
    setActivePanel('myself');
  }

  function openModal() {
    setModalOpen(true);
  }

  function closeModal() {
    setModalOpen(false);
    restart();
  }

  function copyToClipboard(text) {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text).then(function () {
        alert('Copied to clipboard!');
      }).catch(function () {
        fallbackCopy(text);
      });
    } else {
      fallbackCopy(text);
    }
  }

  function fallbackCopy(text) {
    var textArea = document.createElement('textarea');
    textArea.value = text;
    textArea.style.position = 'fixed';
    textArea.style.left = '-999999px';
    document.body.appendChild(textArea);
    textArea.focus();
    textArea.select();
    try {
      document.execCommand('copy');
      alert('Copied to clipboard!');
    } catch (err) {
      alert('Could not copy to clipboard');
    }
    document.body.removeChild(textArea);
  }

            <div style={{ display: 'flex', gap: '8px', marginBottom: '12px', flexWrap: 'wrap' }}>
              <button
                onClick={function () { setActivePanel('myself'); setShowStep2(false); }}
                style={{
                  padding: '10px 16px',
                  fontSize: '14px',

                  border: '1px solid ' + (activePanel === 'myself' ? '#00A651' : '#e5e7eb'),

                }}

              >
                I'll do it myself
              </button>
              <button
                onClick={function () { setActivePanel('developer'); setShowStep2(false); }}
                style={{
                  padding: '10px 16px',
                  fontSize: '14px',
                  backgroundColor: activePanel === 'developer' ? '#00A651' : '#fff',
                  color: activePanel === 'developer' ? '#fff' : '#374151',

                  backgroundColor: activePanel === 'ai' ? '#00A651' : '#fff',

                </div>
              )}

              {activePanel === 'ai' && (

                          {hint1}

                      )}
                      <div style={{ position: 'relative' }}>
                        <pre style={{ backgroundColor: '#f9fafb', padding: '12px', borderRadius: '8px', fontSize: '12px', lineHeight: '1.6', overflowX: 'auto', whiteSpace: 'pre-wrap', wordWrap: 'break-word', border: '1px solid #e5e7eb', margin: 0 }}>
                          {prompt1}
                        </pre>
                        <button
                          onClick={function () { copyToClipboard(prompt1); }}
                          style={{ position: 'absolute', top: '8px', right: '8px', padding: '6px 10px', fontSize: '11px', fontWeight: 600, borderRadius: '6px', border: '1px solid #e5e7eb', background: '#fff', color: '#374151', cursor: 'pointer' }}
                          type="button"
                        >
                          Copy
                        </button>
                      </div>
                    </div>
                  )}
                  {hasStep2 && showStep2 && prompt2 && (
                    <div style={{ marginTop: '16px' }}>
                      <p style={{ fontSize: '13px', fontWeight: 600, color: '#1e3a5f', marginBottom: '8px' }}>{step2Label.replace('Show ', '')}</p>
                      {hint2 && (
                        <p style={{ fontSize: '12px', color: '#6b7280', fontStyle: 'italic', margin: '0 0 6px 0' }}>
                          {hint2}
                        </p>
                      )}
                      <div style={{ position: 'relative' }}>
                        <pre style={{ backgroundColor: '#f9fafb', padding: '12px', borderRadius: '8px', fontSize: '12px', lineHeight: '1.6', overflowX: 'auto', whiteSpace: 'pre-wrap', wordWrap: 'break-word', border: '1px solid #e5e7eb', margin: 0 }}>
                          {prompt2}
                        </pre>
                        <button
                          onClick={function () { copyToClipboard(prompt2); }}
                          style={{ position: 'absolute', top: '8px', right: '8px', padding: '6px 10px', fontSize: '11px', fontWeight: 600, borderRadius: '6px', border: '1px solid #e5e7eb', background: '#fff', color: '#374151', cursor: 'pointer' }}
                          type="button"
                        >
                          Copy
                        </button>
                      </div>
                    </div>
                  )}
                  <div style={{ marginTop: '12px', display: 'flex', gap: '8px' }}>
                    {hasStep2 && !showStep2 && (
                      <button
                        onClick={function () { setShowStep2(true); }}
                        style={{ padding: '8px 14px', fontSize: '13px', fontWeight: 600, borderRadius: '8px', background: '#00A651', color: '#fff', border: 'none', cursor: 'pointer' }}
                        type="button"
                      >
                        {step2Label}
                      </button>
                    )}
                  </div>
                  <div style={{ backgroundColor: '#fff3cd', border: '1px solid #ffc107', padding: '10px 12px', borderRadius: '8px', marginTop: '16px', fontSize: '12px' }}>

                    <p style={{ lineHeight: '1.6', margin: 0, color: '#78350f' }}>

                    </p>
                  </div>
                </div>
              )}
            </div>
          </div>

          {relatedProducts.length > 0 && (
            <div style={{ marginTop: '20px', borderTop: '1px solid #e5e7eb', paddingTop: '16px' }}>
              <button
                type="button"
                onClick={function () { setShowRelated(!showRelated); }}
                style={{ display: 'flex', alignItems: 'center', gap: '8px', background: 'none', border: 'none', cursor: 'pointer', padding: 0, fontSize: '13px', fontWeight: 600, color: '#1e3a5f' }}

                <span>{showRelated ? '▼' : '▶'}</span>
                <span>Explore what to add next</span>
              </button>
              {showRelated && (
                <div style={{ marginTop: '12px', display: 'grid', gap: '8px' }}>
                  {relatedProducts.map(function (item, idx) {
                    return (
                      <a
                        key={idx}
                        href={item.url}
                        target="_blank"
                        rel="noopener noreferrer"
                        style={{ display: 'block', padding: '10px 12px', borderRadius: '8px', border: '1px solid #e5e7eb', background: '#f9fafb', textDecoration: 'none', color: 'inherit' }}

          )}

            <button type="button" onClick={goBackFromResult} style={{ padding: '8px 14px', borderRadius: '8px', fontSize: '13px', fontWeight: 600, cursor: 'pointer', background: '#fff', border: '1px solid #e5e7eb', color: '#374151' }}>
              Back
            </button>
            <button type="button" onClick={restart} style={{ padding: '8px 14px', borderRadius: '8px', fontSize: '13px', fontWeight: 600, cursor: 'pointer', background: '#00A651', border: 'none', color: '#fff' }}>
              Start over
            </button>
          </div>
        </div>
      );
    }
  }

  var wizardBody = computeWizardBody();

  return (
    <div style={shell}>
      {!modalOpen && (
        <div style={{ textAlign: 'center' }}>
          <button
            ref={launchBtnRef}
            type="button"
            onClick={openModal}
            onMouseEnter={function () { setHoveredLaunch(true); }}
            onMouseLeave={function () { setHoveredLaunch(false); }}
            style={launchBtnStyle}
          >
            <span style={{ fontSize: '20px' }}>{'\uD83D\uDE80'}</span>
            <span>Find Your Integration Path</span>
          </button>

      )}

      {modalOpen && (
        <div
          style={{
            position: 'fixed',
            top: 0,
            left: 0,
            right: 0,
            bottom: 0,
            zIndex: 99999,
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            background: 'rgba(15, 23, 42, 0.55)',
            backdropFilter: 'blur(4px)',
            padding: '16px'
          }}
          onClick={function (e) {
            if (e.target === e.currentTarget) closeModal();

          <div

              maxWidth: '680px',

                alignItems: 'center',
                borderBottom: '1px solid rgba(255,255,255,0.1)'
              }}
            >
              <h2 id="wizard-title" style={{ margin: 0, fontSize: '1.25rem', fontWeight: 700 }}>
                Find Your Integration Path
              </h2>
              <button
                onClick={closeModal}
                type="button"
                aria-label="Close dialog"
