    var newAnswers = Object.assign({}, answers);
    newAnswers[questionKey] = answerKey;
    setAnswers(newAnswers);

      if (answerKey === 'website') {

        setCurrentQuestion('Q2');

      } else if (answerKey === 'in_person') {
        showResult('upi_qr', newAnswers);

      if (answerKey === 'platform') {
        setCurrentQuestion('Q2_platform');
      } else if (answerKey === 'custom') {
        setCurrentQuestion('Q3');
      } else if (answerKey === 'no_website') {

    setActivePanel('myself');
    // Keep currentQuestion and answers as they were before showing result
  }

  function goBack() {
    if (result) {
      setResult(null);
      return;
    }
    var prevQuestion = null;
    if (currentQuestion === 'Q1_fallback') prevQuestion = 'Q1';
    else if (currentQuestion === 'Q2') prevQuestion = answers.Q1 === 'not_sure' ? 'Q1_fallback' : 'Q1';
    else if (currentQuestion === 'Q2_platform') prevQuestion = 'Q2';
    else if (currentQuestion === 'Q3') prevQuestion = 'Q2';
    if (prevQuestion) {
      setCurrentQuestion(prevQuestion);
      var newAnswers = Object.assign({}, answers);
      delete newAnswers[currentQuestion];
      setAnswers(newAnswers);
    }
  }

  function copyToClipboard(text) {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text).then(function () {
        alert('Copied to clipboard!');
      }).catch(function () {
        fallbackCopy(text);
      });
    } else {
      fallbackCopy(text);
    }
  }

  function fallbackCopy(text) {
    var textArea = document.createElement('textarea');
    textArea.value = text;
    textArea.style.position = 'fixed';
    textArea.style.left = '-999999px';
    document.body.appendChild(textArea);
    textArea.focus();
    textArea.select();
    try {
      document.execCommand('copy');
      alert('Copied to clipboard!');
    } catch (err) {
      alert('Could not copy to clipboard');
    }
    document.body.removeChild(textArea);
  }

                </button>
              );
            })}
          </div>
          {currentQuestion === 'Q3' && (
            <p style={{ marginTop: '12px', fontSize: '12px', color: '#6b7280', fontStyle: 'italic' }}>
              Most merchants start with a simple, secure PayU page — you can add full design control later.
            </p>
          )}
        </div>

        <div style={{ marginTop: '16px' }}>
          <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: '12px', color: '#6b7280', marginBottom: '6px' }}>
            <span>Step {getProgressValue()} of up to 3</span>
          </div>
          <div
            role="progressbar"
            aria-valuenow={getProgressValue()}
            aria-valuemin={0}
            aria-valuemax={3}
            aria-label={'Quick Start progress: step ' + getProgressValue() + ' of 3'}
            style={{ height: '6px', background: '#e5e7eb', borderRadius: '999px', overflow: 'hidden' }}
          >
            <div style={{ height: '100%', width: (getProgressValue() * 33.33) + '%', background: '#00A651', borderRadius: '999px', transition: 'width 0.25s ease' }} />
          </div>
        </div>

        <div style={{ display: 'flex', justifyContent: 'space-between', gap: '12px', marginTop: '16px', paddingTop: '12px', borderTop: '1px solid #e5e7eb' }}>
          {currentQuestion !== 'Q1' ? (
            <button type="button" onClick={goBack} style={{ padding: '8px 14px', borderRadius: '8px', fontSize: '13px', fontWeight: 600, cursor: 'pointer', background: '#fff', border: '1px solid #e5e7eb', color: '#374151' }}>
              Back
            </button>
          ) : (
            <span />
          )}
          <button type="button" onClick={restart} style={{ padding: '8px 14px', borderRadius: '8px', fontSize: '13px', fontWeight: 600, cursor: 'pointer', background: '#00A651', border: 'none', color: '#fff' }}>
            Start over
          </button>
        </div>
      </div>
    );
  } else {
    var product = result.product;
    var whyText = renderWhyText(product, result.answers);
    var tierInfo = getTierIndicator(product.tier, product.effortLabel);

    wizardBody = (
      <div>
        <div style={{ marginBottom: '16px', padding: '12px 14px', borderRadius: '10px', border: '1px solid rgba(0,166,81,.25)', background: 'rgba(0,166,81,.06)', fontSize: '13px', color: '#374151' }}>
          {whyText}
        </div>

        <div style={{ border: '2px solid #00A651', borderRadius: '12px', overflow: 'hidden', marginBottom: '16px' }}>
          <div style={{ background: '#00A651', color: '#fff', fontSize: '12px', fontWeight: 600, padding: '8px 14px' }}>
            Your recommended setup
          </div>
          <div style={{ padding: '16px' }}>
            <h3 style={{ margin: 0, fontSize: '1.1rem', color: '#1e3a5f' }}>{product.name}</h3>
            <p style={{ margin: '4px 0 0', fontSize: '12px', color: '#6b7280' }}>{product.category}</p>
            <p style={{ margin: '10px 0', fontSize: '13px' }}>{product.description}</p>
            <div style={{ display: 'inline-flex', alignItems: 'center', gap: '6px', padding: '6px 12px', borderRadius: '8px', background: '#f3f4f6', fontSize: '12px', fontWeight: 600, color: '#374151', marginTop: '8px' }}>
              <span>{tierInfo.emoji}</span>
              <span>{tierInfo.label}</span>
            </div>
            {product.requiresEnablement && (
              <div style={{ display: 'inline-flex', alignItems: 'center', gap: '6px', padding: '6px 12px', borderRadius: '8px', background: '#fef3c7', border: '1px solid #fbbf24', fontSize: '12px', fontWeight: 600, color: '#78350f', marginTop: '8px', marginLeft: '8px' }}>
                <span>⏳</span>
                <span>May require enabling by your account team first</span>
              </div>
            )}
          </div>
        </div>

        <div style={{ marginBottom: '16px' }}>
          <h4 style={{ fontSize: '14px', fontWeight: '600', color: '#1e3a5f', marginBottom: '8px' }}>What you'll need</h4>
          <ul style={{ lineHeight: '1.8', paddingLeft: '20px', fontSize: '13px', color: '#4b5563', margin: 0 }}>
            {product.prerequisites.map(function (prereq, index) {
              return <li key={index}>{prereq}</li>;
            })}
          </ul>
        </div>

        <div style={{ marginBottom: '16px' }}>
          <h4 style={{ fontSize: '14px', fontWeight: '600', color: '#1e3a5f', marginBottom: '12px' }}>How do you want to proceed?</h4>
          <div style={{ display: 'flex', gap: '8px', marginBottom: '12px', flexWrap: 'wrap' }}>
            <button
              onClick={function () { setActivePanel('myself'); setShowStep2(false); }}
              style={{
                padding: '10px 16px',
                fontSize: '14px',
                backgroundColor: activePanel === 'myself' ? '#00A651' : '#fff',
                color: activePanel === 'myself' ? '#fff' : '#374151',
                border: '1px solid ' + (activePanel === 'myself' ? '#00A651' : '#e5e7eb'),
                borderRadius: '8px',
                cursor: 'pointer',
                transition: 'all 0.2s',
                fontWeight: 600
              }}
              type="button"
            >
              I'll do it myself
            </button>
            <button
              onClick={function () { setActivePanel('developer'); setShowStep2(false); }}
              style={{
                padding: '10px 16px',
                fontSize: '14px',
                backgroundColor: activePanel === 'developer' ? '#00A651' : '#fff',
                color: activePanel === 'developer' ? '#fff' : '#374151',
                border: '1px solid ' + (activePanel === 'developer' ? '#00A651' : '#e5e7eb'),
                borderRadius: '8px',
                cursor: 'pointer',
                transition: 'all 0.2s',
                fontWeight: 600
              }}
              type="button"
            >
              I have a developer
            </button>
            <button
              onClick={function () { setActivePanel('ai'); setShowStep2(false); }}
              style={{
                padding: '10px 16px',
                fontSize: '14px',
                backgroundColor: activePanel === 'ai' ? '#00A651' : '#fff',
                color: activePanel === 'ai' ? '#fff' : '#374151',
                border: '1px solid ' + (activePanel === 'ai' ? '#00A651' : '#e5e7eb'),
                borderRadius: '8px',
                cursor: 'pointer',
                transition: 'all 0.2s',
                fontWeight: 600
              }}
              type="button"
            >
              I'll use AI to build it
            </button>
          </div>

          <div style={{ border: '1px solid #e5e7eb', borderRadius: '10px', padding: '16px', backgroundColor: '#f9fafb', minHeight: '200px' }}>
            {activePanel === 'myself' && (
              <div style={{ fontSize: '14px', lineHeight: '1.6' }}>
                <p style={{ margin: '0 0 12px', fontSize: '13px', color: '#4b5563' }}>{product.description}</p>
                <h5 style={{ fontSize: '13px', fontWeight: '600', color: '#1e3a5f', marginBottom: '8px', marginTop: '16px' }}>Next steps</h5>
                <ol style={{ lineHeight: '1.7', paddingLeft: '20px', marginBottom: '12px', fontSize: '13px', color: '#4b5563' }}>
                  {product.prerequisites.map(function (prereq, index) {
                    return <li key={index}>{prereq}</li>;
                  })}
                </ol>
                <a href={product.docsUrl} target="_blank" rel="noopener noreferrer" style={{ display: 'inline-block', marginTop: '8px', fontSize: '13px', fontWeight: 600, color: '#00A651' }}>
                  View PayU documentation →
                </a>
              </div>
            )}

            {activePanel === 'developer' && (
              <div style={{ fontSize: '14px', lineHeight: '1.6' }}>
                <p style={{ margin: '0 0 12px', fontSize: '13px', color: '#4b5563' }}>Share this setup summary with your developer or agency:</p>
                <div style={{ backgroundColor: '#f9fafb', padding: '12px 14px', borderRadius: '8px', fontSize: '13px', fontFamily: 'monospace', lineHeight: '1.7', border: '1px solid #e5e7eb' }}>
                  <p style={{ margin: '0 0 8px' }}><strong>What I want to achieve:</strong> {getQ1AnswerText(result.answers)}</p>
                  {(result.answers.Q2 || result.answers.Q3) && (
                    <p style={{ margin: '0 0 8px' }}><strong>Where I want to accept payments:</strong> {result.answers.Q2 === 'custom' || result.answers.Q2 === 'platform' ? 'On my website' : 'For my business'}</p>
                  )}
                  <p style={{ margin: '0 0 8px' }}><strong>Recommended PayU solution:</strong> {product.name}</p>
                  <p style={{ margin: '0 0 4px' }}><strong>What's needed before starting:</strong></p>
                  <ul style={{ paddingLeft: '20px', margin: '4px 0 8px 0', fontSize: '12px' }}>
                    {product.prerequisites.map(function (prereq, index) {
                      return <li key={index}>{prereq}</li>;
                    })}
                  </ul>
                  <p style={{ margin: 0 }}><strong>Technical guide:</strong> <a href={product.docsUrl} target="_blank" rel="noopener noreferrer" style={{ color: '#00A651', fontWeight: 600 }}>{product.docsUrl}</a></p>
                </div>
                <button
                  onClick={function () {
                    var summary = 'What I want to achieve: ' + getQ1AnswerText(result.answers) + '\n\nRecommended PayU solution: ' + product.name + '\n\nWhat\'s needed:\n' + product.prerequisites.map(function (p) { return '- ' + p; }).join('\n') + '\n\nTechnical guide: ' + product.docsUrl;
                    copyToClipboard(summary);
                  }}
                  style={{ marginTop: '12px', padding: '8px 14px', fontSize: '13px', fontWeight: 600, borderRadius: '8px', border: '1px solid #e5e7eb', background: '#fff', color: '#374151', cursor: 'pointer' }}
                  type="button"
                >
                  Copy summary
                </button>
                <div style={{ backgroundColor: '#fff3cd', border: '1px solid #ffc107', padding: '10px 12px', borderRadius: '8px', marginTop: '16px', fontSize: '12px' }}>
                  <p style={{ fontWeight: '600', marginBottom: '4px', color: '#92400e' }}>⚠️ Before you send this</p>
                  <p style={{ lineHeight: '1.6', margin: 0, color: '#78350f' }}>
                    This summary describes what your developer needs — it does not include your actual Merchant Key or Salt. Do not add your real credential values into this message before sending it. If your developer doesn't have PayU Dashboard access, check with your account team on the right way to share access — this isn't yet a documented process.
                  </p>
                </div>
              </div>
            )}

            {activePanel === 'ai' && (
              <div style={{ fontSize: '14px', lineHeight: '1.6' }}>
                <p style={{ lineHeight: '1.6', marginBottom: '12px', fontSize: '12px', color: '#6b7280' }}>
                  Use this with Claude, ChatGPT, or another AI coding assistant. Copy the entire prompt and paste it into your conversation.
                </p>
                <div style={{ position: 'relative' }}>
                  <pre style={{ backgroundColor: '#f9fafb', padding: '12px', borderRadius: '8px', fontSize: '12px', lineHeight: '1.6', overflowX: 'auto', whiteSpace: 'pre-wrap', wordWrap: 'break-word', border: '1px solid #e5e7eb', margin: 0 }}>
                    {buildAIPrompt(product, 1)}
                  </pre>
                  <button
                    onClick={function () { copyToClipboard(buildAIPrompt(product, 1)); }}
                    style={{ position: 'absolute', top: '8px', right: '8px', padding: '6px 10px', fontSize: '11px', fontWeight: 600, borderRadius: '6px', border: '1px solid #e5e7eb', background: '#fff', color: '#374151', cursor: 'pointer' }}
                    type="button"
                  >
                    Copy
                  </button>
                </div>
                {showStep2 && (
                  <div style={{ marginTop: '16px' }}>
                    <p style={{ fontSize: '13px', fontWeight: 600, color: '#1e3a5f', marginBottom: '8px' }}>Step 2: Handle webhook response</p>
                    <div style={{ position: 'relative' }}>
                      <pre style={{ backgroundColor: '#f9fafb', padding: '12px', borderRadius: '8px', fontSize: '12px', lineHeight: '1.6', overflowX: 'auto', whiteSpace: 'pre-wrap', wordWrap: 'break-word', border: '1px solid #e5e7eb', margin: 0 }}>
                        {buildAIPrompt(product, 2)}
                      </pre>
                      <button
                        onClick={function () { copyToClipboard(buildAIPrompt(product, 2)); }}
                        style={{ position: 'absolute', top: '8px', right: '8px', padding: '6px 10px', fontSize: '11px', fontWeight: 600, borderRadius: '6px', border: '1px solid #e5e7eb', background: '#fff', color: '#374151', cursor: 'pointer' }}
                        type="button"
                      >
                        Copy
                      </button>
                    </div>
                  </div>
                )}
                <div style={{ marginTop: '12px', display: 'flex', gap: '8px' }}>
                  {!showStep2 && (
                    <button
                      onClick={function () { setShowStep2(true); }}
                      style={{ padding: '8px 14px', fontSize: '13px', fontWeight: 600, borderRadius: '8px', background: '#00A651', color: '#fff', border: 'none', cursor: 'pointer' }}
                      type="button"
                    >
                      Show Step 2
                    </button>
                  )}
                </div>
                <div style={{ backgroundColor: '#fff3cd', border: '1px solid #ffc107', padding: '10px 12px', borderRadius: '8px', marginTop: '16px', fontSize: '12px' }}>
                  <p style={{ fontWeight: '600', marginBottom: '4px', color: '#92400e' }}>⚠️ Security reminder</p>
                  <p style={{ lineHeight: '1.6', margin: 0, color: '#78350f' }}>
                    Never share your Merchant Key or Salt in prompts to AI. Keep credentials in environment variables, not in code.
                  </p>
                </div>
              </div>
            )}
          </div>
        </div>

        <div style={{ display: 'flex', justifyContent: 'space-between', gap: '12px', marginTop: '16px', paddingTop: '12px', borderTop: '1px solid #e5e7eb' }}>
          <button type="button" onClick={goBackFromResult} style={{ padding: '8px 14px', borderRadius: '8px', fontSize: '13px', fontWeight: 600, cursor: 'pointer', background: '#fff', border: '1px solid #e5e7eb', color: '#374151' }}>
            Back
          </button>
          <button type="button" onClick={restart} style={{ padding: '8px 14px', borderRadius: '8px', fontSize: '13px', fontWeight: 600, cursor: 'pointer', background: '#00A651', border: 'none', color: '#fff' }}>
            Start over
          </button>
        </div>
      </div>
    );
  }

  return (
    <div style={shell}>
      {!modalOpen && (
        <div style={{ textAlign: 'center' }}>
          <button
            ref={launchBtnRef}
            type="button"
            onClick={openModal}
            onMouseEnter={function () { setHoveredLaunch(true); }}
            onMouseLeave={function () { setHoveredLaunch(false); }}
            style={launchBtnStyle}
          >
            <span style={{ fontSize: '20px' }}>🚀</span>
            <span>Find Your Integration Path</span>
          </button>
        </div>
      )}

      {modalOpen && (
        <div
          style={{
            position: 'fixed',
            top: 0,
            left: 0,
            right: 0,
            bottom: 0,
            zIndex: 99999,
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            background: 'rgba(15, 23, 42, 0.55)',
            backdropFilter: 'blur(4px)',
            padding: '16px'
          }}
          onClick={function (e) {
            if (e.target === e.currentTarget) closeModal();
          }}
        >
          <div
            ref={modalRef}
            role="dialog"
            aria-modal="true"
            aria-labelledby="wizard-title"
            style={{
              position: 'relative',
              width: '100%',
              maxWidth: '680px',
              maxHeight: '90vh',
              background: '#fff',
              borderRadius: '16px',
              boxShadow: '0 25px 50px -12px rgba(0, 0, 0, 0.25)',
              display: 'flex',
              flexDirection: 'column',
              overflow: 'hidden'
            }}
          >
            <div
              style={{
                position: 'sticky',
                top: 0,
                zIndex: 1,
                background: 'linear-gradient(135deg, #00A651 0%, #008f47 100%)',
                color: '#fff',
                padding: '20px 24px',
                display: 'flex',
                justifyContent: 'space-between',
                alignItems: 'center',
                borderBottom: '1px solid rgba(255,255,255,0.1)'
              }}
            >
              <h2 id="wizard-title" style={{ margin: 0, fontSize: '1.25rem', fontWeight: 700 }}>
                Find Your Integration Path
              </h2>
              <button
                onClick={closeModal}
                type="button"
                aria-label="Close dialog"
                style={{
                  background: 'transparent',
                  border: 'none',
                  color: '#fff',
                  fontSize: '24px',
                  cursor: 'pointer',
                  padding: '4px',
                  lineHeight: 1,
                  opacity: 0.9,
                  transition: 'opacity 0.2s'
                }}
                onMouseEnter={function (e) { e.currentTarget.style.opacity = '1'; }}
                onMouseLeave={function (e) { e.currentTarget.style.opacity = '0.9'; }}
              >
                ×
              </button>
            </div>

            <div
              style={{
                flex: 1,
                overflowY: 'auto',
                padding: '24px'
              }}
            >
              {wizardBody}
            </div>
          </div>
        </div>
      )}
    </div>
  );
};
