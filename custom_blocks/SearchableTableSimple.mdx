/**
 * SearchableTableSimple - Readme CMS Custom Component
 *
 * Uses inline styles (Readme preview does not load Tailwind).
 * Paste into Settings, Custom Components. Name: SearchableTableSimple
 */

import { useState, useMemo } from 'react';

export const SearchableTableSimple = (props) => {
  const {
    headers = [],
    rows = [],
    placeholder = 'Search in table...',
    maxHeight = '400px',
    emptyMessage = 'No results found. Try a different search.',
    className = '',
  } = props;

  const [searchQuery, setSearchQuery] = useState('');

  const normalize = (str) =>
    str == null ? '' : String(str).toLowerCase().trim().replace(/\s+/g, ' ');

  const rowMatchesQuery = (row, query) => {
    const q = normalize(query);
    if (!q) return true;
    const words = q.split(/\s+/).filter(Boolean);
    const rowText = row.map((cell) => normalize(cell)).join(' ');
    return words.every((word) => rowText.includes(word));
  };

  const filteredRows = useMemo(
    () => rows.filter((row) => rowMatchesQuery(row, searchQuery)),
    [rows, searchQuery]
  );

  const hasSearchQuery = normalize(searchQuery).length > 0;
  const showEmptyState = hasSearchQuery && filteredRows.length === 0;
  const colCount = headers.length;

  const fontSans =
    '-apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, "Helvetica Neue", Arial, sans-serif';
  const borderColor = '#e5e7eb';

  const headerCellStyle = {
    padding: '12px 16px',
    textAlign: 'left',
    fontFamily: fontSans,
    fontSize: '12px',
    fontWeight: '700',
    color: '#374151',
    textTransform: 'uppercase',
    letterSpacing: '0.05em',
    backgroundColor: '#f9fafb',
    border: '1px solid ' + borderColor,
    verticalAlign: 'middle',
    position: 'sticky',
    top: 0,
    zIndex: 10,
    boxSizing: 'border-box',
  };

  const bodyCellStyle = {
    padding: '12px 16px',
    fontFamily: fontSans,
    fontSize: '14px',
    fontWeight: '400',
    lineHeight: '1.5',
    color: '#111827',
    backgroundColor: '#ffffff',
    border: '1px solid ' + borderColor,
    verticalAlign: 'top',
    boxSizing: 'border-box',
  };

  return (
    <div
      className={className}
      style={{
        width: '100%',
        display: 'block',
        fontFamily: fontSans,
        fontSize: '14px',
        color: '#111827',
      }}
    >
      <div
        style={{
          width: '100%',
          border: '1px solid ' + borderColor,
          borderRadius: '6px',
          overflow: 'hidden',
          backgroundColor: '#ffffff',
          boxSizing: 'border-box',
        }}
      >
        <div
          style={{
            padding: '12px 16px',
            backgroundColor: '#f9fafb',
            borderBottom: '1px solid ' + borderColor,
          }}
        >
          <div
            style={{
              display: 'flex',
              alignItems: 'center',
              width: '100%',
              maxWidth: '420px',
              border: '1px solid #d1d5db',
              borderRadius: '6px',
              backgroundColor: '#ffffff',
              boxShadow: '0 1px 2px 0 rgba(0,0,0,0.05)',
            }}
          >
            <span
              aria-hidden="true"
              style={{
                display: 'flex',
                alignItems: 'center',
                paddingLeft: '12px',
                color: '#9ca3af',
                flexShrink: 0,
              }}
            >
              <svg
                xmlns="http://www.w3.org/2000/svg"
                width="18"
                height="18"
                viewBox="0 0 24 24"
                fill="none"
                stroke="currentColor"
                strokeWidth="2"
                strokeLinecap="round"
                strokeLinejoin="round"
              >
                <circle cx="11" cy="11" r="8" />
                <path d="m21 21-4.35-4.35" />
              </svg>
            </span>
            <input
              type="search"
              placeholder={placeholder}
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              aria-label="Search table"
              style={{
                flex: 1,
                minWidth: 0,
                padding: '8px 12px 8px 8px',
                border: 0,
                backgroundColor: 'transparent',
                outline: 'none',
                fontSize: '14px',
                fontFamily: fontSans,
                color: '#111827',
              }}
            />
          </div>
        </div>

        <div
          style={{
            width: '100%',
            overflowY: 'auto',
            overflowX: 'auto',
            maxHeight: showEmptyState ? 'none' : maxHeight,
            borderTop: '1px solid ' + borderColor,
            boxSizing: 'border-box',
          }}
        >
          <table
            style={{
              width: '100%',
              minWidth: '100%',
              borderCollapse: 'collapse',
              borderSpacing: 0,
              margin: 0,
              fontFamily: fontSans,
            }}
          >
            <thead>
              <tr>
                {headers.map((header, i) => (
                  <th key={i} scope="col" style={headerCellStyle}>
                    {header}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody>
              {showEmptyState ? (
                <tr>
                  <td
                    colSpan={colCount}
                    role="alert"
                    style={{
                      padding: '16px',
                      fontFamily: fontSans,
                      fontSize: '14px',
                      backgroundColor: '#fef2f2',
                      color: '#b91c1c',
                      border: '1px solid ' + borderColor,
                    }}
                  >
                    {emptyMessage}
                  </td>
                </tr>
              ) : (
                filteredRows.map((row, rowIndex) => (
                  <tr key={rowIndex}>
                    {row.map((cell, cellIndex) => (
                      <td key={cellIndex} style={bodyCellStyle}>
                        {cell == null ? '' : cell}
                      </td>
                    ))}
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>

      {filteredRows.length > 0 ? (
        <p
          style={{
            marginTop: '8px',
            fontSize: '12px',
            fontFamily: fontSans,
            color: '#6b7280',
          }}
        >
          Showing {filteredRows.length} of {rows.length} row
          {rows.length !== 1 ? 's' : ''}
          {hasSearchQuery ? ' (filtered by search)' : ''}.
        </p>
      ) : null}
    </div>
  );
};


<SearchableTableSimple
  headers={['Bank', 'Type', 'Status']}
  rows={[
    ['HDFC Bank', 'Netbanking / Cards', 'Active'],
    ['ICICI Bank', 'Netbanking / Cards', 'Active'],
    ['State Bank of India', 'Netbanking', 'Active'],
    ['Axis Bank', 'Cards', 'Active'],
  ]}
  placeholder="Search"
/>
