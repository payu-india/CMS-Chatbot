/**
 * SearchableTableSimple - Readme CMS Custom Component
 *
 * A table with search, no-results error state, and a scrollable body.
 * Paste this entire file into Readme Settings, Custom Components.
 * Register the component name as: SearchableTableSimple
 *
 * Reference: Razorpay-style supported banks/cards table
 */

import { useState, useMemo } from 'react';

export const SearchableTableSimple = (props) => {
  const {
    headers = [],
    rows = [],
    placeholder = 'Search in table...',
    maxHeight = '400px',
    emptyMessage = 'No results found. Try a different search.',
    className = '',
  } = props;

  const [searchQuery, setSearchQuery] = useState('');

  const normalize = (str) =>
    str == null ? '' : String(str).toLowerCase().trim().replace(/\s+/g, ' ');

  const rowMatchesQuery = (row, query) => {
    const q = normalize(query);
    if (!q) return true;
    const words = q.split(/\s+/).filter(Boolean);
    const rowText = row.map((cell) => normalize(cell)).join(' ');
    return words.every((word) => rowText.includes(word));
  };

  const filteredRows = useMemo(
    () => rows.filter((row) => rowMatchesQuery(row, searchQuery)),
    [rows, searchQuery]
  );

  const hasSearchQuery = normalize(searchQuery).length > 0;
  const showEmptyState = hasSearchQuery && filteredRows.length === 0;

  const colCount = headers.length;

  return (
    <div className={'w-fit max-w-full ' + className}>
      <div className="border border-gray-200 rounded-md overflow-hidden">
        <div className="px-4 py-3 bg-gray-50 border-b border-gray-200">
          <div className="flex items-center w-full max-w-md border border-gray-300 rounded-md bg-white shadow-sm focus-within:ring-2 focus-within:ring-blue-500 focus-within:border-blue-500">
            <span
              className="flex items-center justify-center pl-3 text-gray-400 flex-shrink-0"
              aria-hidden="true"
            >
              <svg
                xmlns="http://www.w3.org/2000/svg"
                width="18"
                height="18"
                viewBox="0 0 24 24"
                fill="none"
                stroke="currentColor"
                strokeWidth="2"
                strokeLinecap="round"
                strokeLinejoin="round"
              >
                <circle cx="11" cy="11" r="8" />
                <path d="m21 21-4.35-4.35" />
              </svg>
            </span>
            <input
              type="search"
              placeholder={placeholder}
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className="flex-1 min-w-0 py-2 pr-3 border-0 bg-transparent focus:ring-0 focus:outline-none"
              aria-label="Search table"
            />
          </div>
        </div>

        <div
          className="overflow-auto border-t-0"
          style={showEmptyState ? {} : { maxHeight: maxHeight }}
        >
          <table className="min-w-full divide-y divide-gray-200 border-collapse">
            <thead className="bg-gray-50 sticky top-0 z-10 shadow-[0_2px_0_0_rgba(0,0,0,0.05)]">
              <tr>
                {headers.map((header, i) => (
                  <th
                    key={i}
                    scope="col"
                    className="px-4 py-3 text-left text-xs font-medium text-gray-700 uppercase tracking-wider bg-gray-50 border-b border-gray-200"
                  >
                    {header}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody className="bg-white divide-y divide-gray-200">
              {showEmptyState ? (
                <tr>
                  <td
                    colSpan={colCount}
                    className="px-4 py-4 bg-red-50 text-red-700 border-b border-red-200"
                    role="alert"
                  >
                    {emptyMessage}
                  </td>
                </tr>
              ) : (
                filteredRows.map((row, rowIndex) => (
                  <tr key={rowIndex} className="hover:bg-gray-50">
                    {row.map((cell, cellIndex) => (
                      <td
                        key={cellIndex}
                        className="px-4 py-2 text-sm text-gray-900"
                      >
                        {cell}
                      </td>
                    ))}
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>

      {filteredRows.length > 0 ? (
        <p className="mt-2 text-xs text-gray-500">
          Showing {filteredRows.length} of {rows.length} row
          {rows.length !== 1 ? 's' : ''}
          {hasSearchQuery ? ' (filtered by search)' : ''}.
        </p>
      ) : null}
    </div>
  );
};


<SearchableTableSimple
  headers={['Bank', 'Type', 'Status']}
  rows={[
    ['HDFC Bank', 'Netbanking / Cards', 'Active'],
    ['ICICI Bank', 'Netbanking / Cards', 'Active'],
    ['State Bank of India', 'Netbanking', 'Active'],
    ['Axis Bank', 'Cards', 'Active'],
  ]}
  placeholder="Search"
/>
