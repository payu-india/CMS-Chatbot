// DevExProgressIndicator - Lightweight progress bar for workflow guides.
// Props:
//   current (number, 0-based step index)
//   total (number of steps)
//   label (optional accessible label)
// Usage: <DevExProgressIndicator current={2} total={10} label="Onboarding progress" />

export const DevExProgressIndicator = (props) => {
  var current = Number(props.current);
  if (isNaN(current) || current < 0) current = 0;
  var total = Number(props.total);
  if (isNaN(total) || total < 1) total = 1;
  if (current > total - 1) current = total - 1;
  var label = props.label != null ? String(props.label) : 'Progress';
  var pct = Math.round(((current + 1) / total) * 100);

  return (
    <div
      role="progressbar"
      aria-valuemin={1}
      aria-valuemax={total}
      aria-valuenow={current + 1}
      aria-label={label + ': step ' + (current + 1) + ' of ' + total}
      style={{ margin: '12px 0 16px' }}
    >
      <div
        style={{
          display: 'flex',
          justifyContent: 'space-between',
          marginBottom: '6px',
          fontSize: '12px',
          color: '#6b7280',
          fontFamily: 'system-ui, -apple-system, Segoe UI, Roboto, sans-serif',
        }}
      >
        <span>
          Step {current + 1} of {total}
        </span>
        <span>{pct}%</span>
      </div>
      <div
        style={{
          height: '6px',
          background: '#e5e7eb',
          borderRadius: '999px',
          overflow: 'hidden',
        }}
      >
        <div
          style={{
            height: '100%',
            width: pct + '%',
            background: '#00A651',
            borderRadius: '999px',
            transition: 'width 0.25s ease',
          }}
        />
      </div>
    </div>
  );
};

<DevExProgressIndicator current={2} total={10} label="Accept first payment" />
