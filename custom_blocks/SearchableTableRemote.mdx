// SearchableTableRemote - Readme CMS Custom Component
// Register in Readme as component name: SearchableTableRemote
// Use for large tables (50+ rows). Host JSON on a public GitHub repo and pass dataUrl.
// JSON may include headers, codeColumns, and rows. Regenerate: npm run build:searchable-table:remote
// Do not put curly braces inside line comments in this file.

import { useState, useMemo, useEffect, useRef } from 'react';

export const SearchableTableRemote = (props) => {
  const {
    headers = [],
    headersJson = '',
    dataUrl = '',
    codeColumnsJson = '',
    placeholder = 'Search in table...',
    maxHeight = '400px',
    emptyMessage = 'No results found. Try a different search.',
    loadErrorMessage =
      'Could not load table data. Check the GitHub raw URL and try again.',
    className = '',
    columnWidths = null,
    maxLines = 0,
    codeColumns = null,
  } = props;

  const [searchQuery, setSearchQuery] = useState('');
  const [remoteRows, setRemoteRows] = useState(null);
  const [loadState, setLoadState] = useState(dataUrl ? 'loading' : 'idle');
  const [loadError, setLoadError] = useState('');
  const [remoteHeaders, setRemoteHeaders] = useState(null);
  const [remoteCodeColumns, setRemoteCodeColumns] = useState(null);
  const scrollRef = useRef(null);

  const coerceRows = (input) => {
    if (input == null || input === '') return [];
    if (Array.isArray(input)) {
      return input.filter((row) => Array.isArray(row));
    }
    if (typeof input === 'string') {
      const text = input.trim();
      if (!text) return [];
      try {
        return coerceRows(JSON.parse(text));
      } catch (e) {
        return [];
      }
    }
    if (typeof input === 'object') {
      if (Array.isArray(input.rows)) return coerceRows(input.rows);
      if (typeof input.length === 'number' && input.length >= 0) {
        const list = [];
        for (let i = 0; i < input.length; i += 1) {
          list.push(input[i]);
        }
        return coerceRows(list);
      }
      const vals = Object.keys(input)
        .sort((a, b) => Number(a) - Number(b))
        .map((k) => input[k]);
      if (vals.length > 0 && vals.every((v) => Array.isArray(v))) return vals;
    }
    return [];
  };

  const parseJsonProp = (text, fallback) => {
    if (text == null || text === '') return fallback;
    if (typeof text !== 'string') return text;
    const trimmed = text.trim();
    if (!trimmed) return fallback;
    try {
      return JSON.parse(trimmed);
    } catch (e) {
      return fallback;
    }
  };

  useEffect(() => {
    const url = dataUrl && String(dataUrl).trim();
    if (!url) {
      setRemoteRows(null);
      setRemoteHeaders(null);
      setRemoteCodeColumns(null);
      setLoadState('idle');
      setLoadError('');
      return undefined;
    }

    let cancelled = false;
    setLoadState('loading');
    setLoadError('');
    setRemoteRows(null);

    const primary = String(url).trim();
    const rawToJsDelivr = (rawUrl) => {
      const m = rawUrl.match(
        /^https:\/\/raw\.githubusercontent\.com\/([^/]+)\/([^/]+)\/([^/]+)\/(.+)$/
      );
      if (!m) return null;
      return (
        'https://cdn.jsdelivr.net/gh/' +
        m[1] +
        '/' +
        m[2] +
        '@' +
        m[3] +
        '/' +
        m[4]
      );
    };

    const parsePayload = (json) => {
      const loaded =
        json && Array.isArray(json.rows)
          ? json.rows
          : Array.isArray(json)
            ? json
            : null;
      if (!loaded) {
        throw new Error('JSON must be rows array or object with rows key');
      }
      const hdrs =
        json && Array.isArray(json.headers) ? json.headers : null;
      const codes =
        json && Array.isArray(json.codeColumns) ? json.codeColumns : null;
      return { rows: loaded, headers: hdrs, codeColumns: codes };
    };

    const fetchOne = (fetchUrl) =>
      fetch(fetchUrl).then((res) =>
        res.ok ? res.json() : Promise.reject(new Error('HTTP ' + res.status))
      );

    const urls = [primary];
    const cdn = rawToJsDelivr(primary);
    if (cdn && cdn !== primary) urls.push(cdn);

    const loadSequential = (index, lastErr) => {
      if (index >= urls.length) {
        return Promise.reject(lastErr || new Error('Failed to fetch'));
      }
      return fetchOne(urls[index])
        .then(parsePayload)
        .catch((err) => loadSequential(index + 1, err));
    };

    loadSequential(0, null)
      .then((payload) => {
        if (cancelled) return;
        setRemoteRows(payload.rows);
        setRemoteHeaders(payload.headers);
        setRemoteCodeColumns(payload.codeColumns);
        setLoadState('ready');
      })
      .catch((err) => {
        if (cancelled) return;
        const msg = err && err.message ? err.message : 'Load failed';
        const hint =
          msg === 'Failed to fetch'
            ? ' (browser blocked the URL — use the jsDelivr link from GITHUB_HOSTING.md, or confirm the repo is public)'
            : msg.indexOf('HTTP 404') >= 0
              ? ' (file not found — check repo path: data/payment-error-codes.json)'
              : '';
        setLoadError(msg + hint);
        setLoadState('error');
      });

    return () => {
      cancelled = true;
    };
  }, [dataUrl]);

  const resolvedHeaders = useMemo(() => {
    const fromJson = parseJsonProp(headersJson, null);
    if (Array.isArray(fromJson) && fromJson.length > 0) {
      return fromJson.map((h) => String(h == null ? '' : h));
    }
    if (Array.isArray(headers) && headers.length > 0) {
      return headers.map((h) => String(h == null ? '' : h));
    }
    if (Array.isArray(remoteHeaders) && remoteHeaders.length > 0) {
      return remoteHeaders.map((h) => String(h == null ? '' : h));
    }
    return [];
  }, [headers, headersJson, remoteHeaders]);

  const resolvedCodeColumns = useMemo(() => {
    const fromJson = parseJsonProp(codeColumnsJson, null);
    if (Array.isArray(fromJson)) return fromJson;
    if (Array.isArray(codeColumns)) return codeColumns;
    if (Array.isArray(remoteCodeColumns)) return remoteCodeColumns;
    return null;
  }, [codeColumns, codeColumnsJson, remoteCodeColumns]);

  const tableRows = remoteRows != null ? remoteRows : [];

  useEffect(() => {
    if (scrollRef.current) {
      scrollRef.current.scrollLeft = 0;
    }
  }, [tableRows, loadState, resolvedHeaders.length]);

  const normalize = (str) =>
    str == null
      ? ''
      : String(str)
          .toLowerCase()
          .trim()
          .replace(/`/g, '')
          .replace(/\s+/g, ' ');

  const rowSearchText = useMemo(
    () => tableRows.map((row) => row.map((cell) => normalize(cell)).join(' ')),
    [tableRows]
  );

  const filteredRows = useMemo(() => {
    const q = normalize(searchQuery);
    if (!q) return tableRows;
    const words = q.split(/\s+/).filter(Boolean);
    const matched = [];
    for (let i = 0; i < tableRows.length; i += 1) {
      const text = rowSearchText[i];
      if (words.every((word) => text.includes(word))) {
        matched.push(tableRows[i]);
      }
    }
    return matched;
  }, [tableRows, rowSearchText, searchQuery]);

  const hasSearchQuery = normalize(searchQuery).length > 0;
  const isLoading = loadState === 'loading';
  const hasLoadError = loadState === 'error';
  const showEmptyState =
    !isLoading && !hasLoadError && hasSearchQuery && filteredRows.length === 0;
  const colCount = resolvedHeaders.length;

  const codeColIndexes = useMemo(() => {
    if (Array.isArray(resolvedCodeColumns)) return resolvedCodeColumns;
    const idx = resolvedHeaders.findIndex((h) =>
      /error|code/i.test(String(h == null ? '' : h))
    );
    return idx >= 0 ? [idx] : resolvedHeaders.length > 0 ? [0] : [];
  }, [resolvedCodeColumns, resolvedHeaders]);

  const isCodeCol = (i) => codeColIndexes.indexOf(i) >= 0;

  const isTitleCol = (colIndex) =>
    /^title$/i.test(String(resolvedHeaders[colIndex] == null ? '' : resolvedHeaders[colIndex]).trim());

  const isMessageCol = (colIndex) =>
    /message/i.test(String(resolvedHeaders[colIndex] == null ? '' : resolvedHeaders[colIndex]));

  const headerText = (colIndex) =>
    String(resolvedHeaders[colIndex] == null ? '' : resolvedHeaders[colIndex]).trim();

  const isExpandedDescriptionCol = (colIndex) =>
    /expanded\s*description/i.test(headerText(colIndex));

  const isShortDescriptionCol = (colIndex) =>
    /^description$/i.test(headerText(colIndex));

  const isErrorDescriptionCol = (colIndex) =>
    /error\s*description/i.test(headerText(colIndex));

  const isFieldCol = (colIndex) => /^field$/i.test(headerText(colIndex));

  const isLayerCol = (colIndex) => /layer$/i.test(headerText(colIndex));

  const isProseCol = (colIndex) => {
    if (isCodeCol(colIndex) || isFieldCol(colIndex)) return false;
    if (
      isExpandedDescriptionCol(colIndex) ||
      isShortDescriptionCol(colIndex) ||
      isErrorDescriptionCol(colIndex) ||
      isMessageCol(colIndex) ||
      isTitleCol(colIndex)
    ) {
      return true;
    }
    const h = headerText(colIndex).toLowerCase();
    return /description|reason|message|title|details/i.test(h);
  };

  const getColMaxWidth = (colIndex) => {
    if (isTitleCol(colIndex)) return 300;
    if (isExpandedDescriptionCol(colIndex) || isErrorDescriptionCol(colIndex)) {
      return 520;
    }
    if (isShortDescriptionCol(colIndex)) return 320;
    if (isMessageCol(colIndex)) return 300;
    if (isFieldCol(colIndex) || isCodeCol(colIndex)) return 220;
    if (isLayerCol(colIndex)) return 240;
    return 360;
  };

  const measureCellLen = (colIndex, cell) => {
    if (cell == null) return 0;
    const text = String(cell).trim();
    if (!text) return 0;
    return isCodeCol(colIndex) ? text.length + 2 : text.length;
  };

  const formatHeaderLabel = (text) => {
    const raw = String(text == null ? '' : text).trim();
    if (!raw) return '';
    return raw
      .split(/\s+/)
      .map((word) => {
        if (!word) return word;
        return word.charAt(0).toUpperCase() + word.slice(1).toLowerCase();
      })
      .join(' ');
  };

  const autoColWidths = useMemo(() => {
    const widths = {};
    const rowsForMeasure =
      filteredRows.length > 0 ? filteredRows : tableRows;
    const sampleSize = Math.min(rowsForMeasure.length, 150);
    const pxPerChar = 8;
    const pad = 36;

    resolvedHeaders.forEach((header, colIdx) => {
      let maxLen = formatHeaderLabel(header).length;
      for (let r = 0; r < sampleSize; r += 1) {
        const cell = rowsForMeasure[r] && rowsForMeasure[r][colIdx];
        maxLen = Math.max(maxLen, measureCellLen(colIdx, cell));
      }

      const minW = isCodeCol(colIdx) || isFieldCol(colIdx) ? 110 : 100;
      const maxW = getColMaxWidth(colIdx);

      if (isExpandedDescriptionCol(colIdx) || isErrorDescriptionCol(colIdx)) {
        widths[colIdx] = 420;
        return;
      }

      if (isShortDescriptionCol(colIdx)) {
        widths[colIdx] = Math.min(Math.max(240, maxLen * 7 + pad), maxW);
        return;
      }

      if (isLayerCol(colIdx)) {
        widths[colIdx] = Math.min(
          Math.max(maxLen * pxPerChar + pad, 160),
          maxW
        );
        return;
      }

      if (isProseCol(colIdx)) {
        widths[colIdx] = Math.min(
          Math.max(260, Math.min(maxLen * 6 + pad, maxW)),
          maxW
        );
        return;
      }

      const fitsOneLine =
        isCodeCol(colIdx) || isFieldCol(colIdx) || maxLen <= 26;

      if (fitsOneLine) {
        widths[colIdx] = Math.min(
          Math.max(maxLen * pxPerChar + pad, minW),
          maxW
        );
        return;
      }

      widths[colIdx] = Math.min(Math.max(220, maxLen * 5 + pad), maxW);
    });

    return widths;
  }, [resolvedHeaders, filteredRows, tableRows, codeColIndexes]);

  const tablePixelWidth = useMemo(() => {
    const vals = Object.keys(autoColWidths).map((k) => autoColWidths[k]);
    if (vals.length === 0) return null;
    return vals.reduce((sum, w) => sum + w, 0);
  }, [autoColWidths]);

  const fontSans =
    '-apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, "Helvetica Neue", Arial, sans-serif';
  const fontMono =
    'ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, monospace';
  const borderColor = '#e5e7eb';

  const codeStyle = {
    padding: '2px 6px',
    background: '#f3f4f6',
    color: '#1f2937',
    borderRadius: '4px',
    fontFamily: fontMono,
    fontSize: '13px',
    fontWeight: '400',
  };

  const renderCellContent = (value, colIndex) => {
    if (value == null) return '';
    if (typeof value !== 'string') return value;
    const plain = value.indexOf('`') >= 0 ? value.replace(/`/g, '') : value;
    return isCodeCol(colIndex) ? (
      <code style={codeStyle}>{plain}</code>
    ) : (
      plain
    );
  };

  const colWidthStyle = (colIndex) => {
    const w = autoColWidths[colIndex];
    if (!w) return { minWidth: 0 };
    return {
      minWidth: 0,
      width: w + 'px',
      maxWidth: w + 'px',
    };
  };

  const headerCellStyle = (colIndex, colTotal) => ({
    padding: '12px 16px',
    textAlign: 'left',
    fontFamily: fontSans,
    fontSize: '14px',
    fontWeight: '700',
    lineHeight: '1.25',
    color: '#374151',
    backgroundColor: '#f9fafb',
    borderBottom: '1px solid ' + borderColor,
    borderRight:
      colIndex < colTotal - 1 ? '1px solid ' + borderColor : 'none',
    borderTop: 'none',
    borderLeft: 'none',
    verticalAlign: 'middle',
    boxSizing: 'border-box',
    whiteSpace: 'normal',
    wordBreak: 'normal',
    overflowWrap: 'break-word',
    ...colWidthStyle(colIndex),
  });

  const bodyCellStyle = (colIndex, colTotal, isLastRow) => ({
    padding: '12px 16px',
    fontFamily: fontSans,
    fontSize: '14px',
    fontWeight: '400',
    lineHeight: '1.5',
    color: '#111827',
    backgroundColor: '#ffffff',
    borderBottom: isLastRow ? 'none' : '1px solid ' + borderColor,
    borderRight:
      colIndex < colTotal - 1 ? '1px solid ' + borderColor : 'none',
    borderTop: 'none',
    borderLeft: 'none',
    verticalAlign: 'top',
    boxSizing: 'border-box',
    minWidth: 0,
    wordBreak: 'normal',
    overflowWrap: 'break-word',
    ...colWidthStyle(colIndex),
  });

  const bodyTextStyle = {
    display: 'block',
    margin: 0,
    padding: 0,
    fontFamily: fontSans,
    fontSize: '14px',
    fontWeight: '400',
    lineHeight: '1.5',
    color: '#111827',
    whiteSpace: 'normal',
    overflowWrap: 'break-word',
    wordBreak: 'normal',
  };

  const renderBodyCellInner = (cell, cellIndex) => {
    const content = renderCellContent(cell, cellIndex);
    const plain = cellPlainText(cell);

    if (clampStyle) {
      return (
        <div style={clampStyle} title={plain}>
          {content}
        </div>
      );
    }

    if (isCodeCol(cellIndex) || isFieldCol(cellIndex)) {
      return content;
    }

    return <span style={bodyTextStyle}>{content}</span>;
  };

  const messageCellStyle = {
    padding: '16px',
    fontFamily: fontSans,
    fontSize: '14px',
    color: '#6b7280',
    border: 'none',
    boxSizing: 'border-box',
  };

  const clampStyle =
    typeof maxLines === 'number' && maxLines > 0
      ? {
          display: '-webkit-box',
          WebkitBoxOrient: 'vertical',
          WebkitLineClamp: maxLines,
          overflow: 'hidden',
          textOverflow: 'ellipsis',
        }
      : null;

  const renderColgroup = () => {
    if (resolvedHeaders.length === 0) return null;
    if (Object.keys(autoColWidths).length === 0) return null;
    return (
      <colgroup>
        {resolvedHeaders.map((_, i) => (
          <col
            key={i}
            style={
              autoColWidths[i] != null
                ? {
                    width: autoColWidths[i] + 'px',
                    minWidth: autoColWidths[i] + 'px',
                    maxWidth: autoColWidths[i] + 'px',
                  }
                : undefined
            }
          />
        ))}
      </colgroup>
    );
  };

  const cellPlainText = (value) =>
    value == null ? '' : String(value).replace(/`/g, '');

  const showNoRows =
    !isLoading && !hasLoadError && tableRows.length === 0;

  return (
    <div
      className={className}
      style={{
        width: '100%',
        display: 'block',
        verticalAlign: 'top',
        fontFamily: fontSans,
        fontSize: '14px',
        lineHeight: '1.5',
        color: '#111827',
      }}
    >
      <div
        style={{
          width: '100%',
          border: '1px solid ' + borderColor,
          borderRadius: '6px',
          overflow: 'hidden',
          boxSizing: 'border-box',
          backgroundColor: '#ffffff',
        }}
      >
        <div
          style={{
            padding: '12px 16px',
            backgroundColor: '#f9fafb',
            borderBottom: '1px solid ' + borderColor,
          }}
        >
          <div
            style={{
              display: 'flex',
              alignItems: 'center',
              width: '100%',
              maxWidth: '420px',
              border: '1px solid #d1d5db',
              borderRadius: '6px',
              backgroundColor: '#ffffff',
              boxShadow: '0 1px 2px 0 rgba(0,0,0,0.05)',
            }}
          >
            <span
              aria-hidden="true"
              style={{
                display: 'flex',
                alignItems: 'center',
                paddingLeft: '12px',
                color: '#9ca3af',
                flexShrink: 0,
              }}
            >
              <svg
                xmlns="http://www.w3.org/2000/svg"
                width="18"
                height="18"
                viewBox="0 0 24 24"
                fill="none"
                stroke="currentColor"
                strokeWidth="2"
                strokeLinecap="round"
                strokeLinejoin="round"
              >
                <circle cx="11" cy="11" r="8" />
                <path d="m21 21-4.35-4.35" />
              </svg>
            </span>
            <input
              type="search"
              placeholder={placeholder}
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              aria-label="Search table"
              style={{
                flex: 1,
                minWidth: 0,
                padding: '8px 12px 8px 8px',
                border: 0,
                backgroundColor: 'transparent',
                outline: 'none',
                fontSize: '14px',
                fontFamily: fontSans,
                fontWeight: '400',
                color: '#111827',
              }}
            />
          </div>
        </div>

        <div
          ref={scrollRef}
          style={{
            width: '100%',
            minWidth: 0,
            overflowY: 'auto',
            overflowX: 'auto',
            position: 'relative',
            maxHeight: showEmptyState ? 'none' : maxHeight,
            boxSizing: 'border-box',
          }}
        >
          <table
            style={{
              width: tablePixelWidth ? tablePixelWidth + 'px' : '100%',
              minWidth: '100%',
              borderCollapse: 'collapse',
              borderSpacing: 0,
              margin: 0,
              tableLayout: tablePixelWidth ? 'fixed' : 'auto',
              fontFamily: fontSans,
            }}
          >
            {renderColgroup()}
            {resolvedHeaders.length > 0 ? (
              <thead>
                <tr>
                  {resolvedHeaders.map((header, i) => (
                    <th
                      key={i}
                      scope="col"
                      style={headerCellStyle(i, resolvedHeaders.length)}
                    >
                      <span
                        style={{
                          display: 'block',
                          margin: 0,
                          padding: 0,
                          lineHeight: '1.25',
                        }}
                      >
                        {formatHeaderLabel(header)}
                      </span>
                    </th>
                  ))}
                </tr>
              </thead>
            ) : null}
            <tbody style={{ backgroundColor: '#ffffff' }}>
              {showNoRows ? (
                <tr>
                  <td colSpan={colCount || 1} style={messageCellStyle}>
                    No rows loaded. Set dataUrl to a public JSON file on GitHub (jsDelivr URL). See GITHUB_HOSTING.md.
                  </td>
                </tr>
              ) : isLoading ? (
                <tr>
                  <td colSpan={colCount || 1} style={messageCellStyle}>
                    Loading table data...
                  </td>
                </tr>
              ) : hasLoadError ? (
                <tr>
                  <td
                    colSpan={colCount || 1}
                    role="alert"
                    style={{
                      ...messageCellStyle,
                      backgroundColor: '#fef2f2',
                      color: '#b91c1c',
                    }}
                  >
                    {loadErrorMessage}
                    {loadError ? ' (' + loadError + ')' : ''}
                  </td>
                </tr>
              ) : showEmptyState ? (
                <tr>
                  <td
                    colSpan={colCount || 1}
                    role="alert"
                    style={{
                      ...messageCellStyle,
                      backgroundColor: '#fef2f2',
                      color: '#b91c1c',
                    }}
                  >
                    {emptyMessage}
                  </td>
                </tr>
              ) : (
                filteredRows.map((row, rowIndex) => (
                  <tr key={rowIndex}>
                    {row.map((cell, cellIndex) => (
                      <td
                        key={cellIndex}
                        style={bodyCellStyle(
                          cellIndex,
                          row.length,
                          rowIndex === filteredRows.length - 1
                        )}
                      >
                        {renderBodyCellInner(cell, cellIndex)}
                      </td>
                    ))}
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>

      {!isLoading && !hasLoadError && !showNoRows && filteredRows.length > 0 ? (
        <p
          style={{
            marginTop: '8px',
            fontSize: '12px',
            fontFamily: fontSans,
            color: '#6b7280',
          }}
        >
          Showing {filteredRows.length} of {tableRows.length} row
          {tableRows.length !== 1 ? 's' : ''}
          {hasSearchQuery ? ' (filtered by search)' : ''}.
        </p>
      ) : null}
    </div>
  );
};
