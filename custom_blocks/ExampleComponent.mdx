      }
    }
    if (empty(f.email) === false) {
      if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(empty(f.email))) {
        return 'Enter a valid email address.';
      }
    }
    for (var j = 0; j < p.fields.length; j += 1) {
      var field = p.fields[j];
      if (field.type !== 'json') continue;
      var raw = empty(f[field.key]);
      if (!raw) continue;
      try {
        JSON.parse(raw);
      } catch (e2) {
        return field.label + ' must be valid JSON.';
      }
    }
    return '';
  }

  function resetForm() {
    if (!profile) return;
    setForm(mergeDefaults(profile.defaults, props.defaultsJson));
    setHashHex('');
    setPipeString('');
    setErr('');
    setCopyMsg('');
  }

  function generateHash() {
    if (!profile) {
      setErr('Unknown hash profile.');
      return;
    }
    var validationErr = validateForm(profile, form);
    if (validationErr) {
      setErr(validationErr);
      return;
    }
    var chain = buildHashString(profile.buildMode || 'forward-udf', form);
    if (!chain) {
      setErr('Could not build hash string for this profile.');
      return;
    }
    setBusy(true);
    setErr('');
    sha512Hex(chain)
      .then(function (h) {
        setHashHex(h);
        setPipeString(chain);
        setBusy(false);
      })
      .catch(function () {
        setErr('SHA-512 unavailable in this browser context.');
        setBusy(false);
      });
  }

  function copyText(text, label) {
    var value = String(text || '');
    if (!value) return;
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(value).then(function () {
        setCopyMsg(label + ' copied.');
        setTimeout(function () { setCopyMsg(''); }, 2000);
      });
      return;
    }
    setCopyMsg('Copy not supported in this browser.');
  }

  if (!profile) {
    return (
      <div style={shell} role="alert">
        <strong>PayU Forward Hash Generator</strong>
        <p style={{ margin: '8px 0 0', color: danger }}>Unknown profile: {profileId}</p>
      </div>
    );
  }

  var sortedFields = (profile.fields || []).slice().sort(function (a, b) {
    return (a.order || 0) - (b.order || 0);
  });

  var basicFields = [];
  var advancedFields = [];
  for (var fi = 0; fi < sortedFields.length; fi += 1) {
    if (sortedFields[fi].advanced) advancedFields.push(sortedFields[fi]);
    else basicFields.push(sortedFields[fi]);
  }

  var primaryLabel = (profile.ux && profile.ux.primaryAction) || 'Generate Forward Hash';
  var bannerText = (profile.security && profile.security.bannerText) || 'Test credentials only. Production hashes must be generated on your server.';
  var showFormula = !profile.ux || profile.ux.showFormula !== false;
  var showPipeString = !profile.ux || profile.ux.showPipeString !== false;
  var relatedLinks = (profile.ux && profile.ux.relatedLinks) || [];

  function renderField(field) {
    var val = form[field.key] != null ? form[field.key] : '';
    var isJson = field.type === 'json';
    var inputType = field.type === 'password' ? 'password' : field.type === 'email' ? 'email' : 'text';
    return (
      <div key={field.key} style={{ marginBottom: '12px' }}>
        <label style={{ display: 'block', fontSize: '12px', fontWeight: 600, marginBottom: '4px', color: '#CFE0DA' }}>
          {field.label}
          {field.required ? <span style={{ color: danger }}> *</span> : null}
        </label>
        {isJson ? (
          <textarea
            rows={3}
            value={val}
            onChange={function (ev) { patchField(field.key, ev.target.value); }}
            placeholder={field.placeholder || ''}
            style={Object.assign({}, inputStyle, { minHeight: '72px', resize: 'vertical' })}
          />
        ) : (
          <input
            type={inputType}
            value={val}
            onChange={function (ev) { patchField(field.key, ev.target.value); }}
            placeholder={field.placeholder || ''}
            style={inputStyle}
            autoComplete={field.type === 'password' ? 'off' : 'on'}
          />
        )}
        {field.help ? <p style={{ margin: '4px 0 0', fontSize: '11px', color: muted }}>{field.help}</p> : null}
      </div>
    );
  }

  return (
    <div style={shell} role="region" aria-label="PayU Forward Hash Generator">
      <div style={{ fontSize: '11px', fontWeight: 600, textTransform: 'uppercase', letterSpacing: '0.12em', color: mint, marginBottom: '8px' }}>
        PayU · {profile.product} · Forward hash lab
      </div>
      <h2 style={{ margin: '0 0 6px', fontSize: '1.25rem', fontWeight: 700, color: '#EAF2EE' }}>{profile.title}</h2>
      <p style={{ margin: '0 0 12px', fontSize: '13px', color: muted }}>{profile.description}</p>

      <div style={{ padding: '10px 12px', borderRadius: '10px', border: '1px solid #3d2f1a', background: '#1a1408', marginBottom: '16px', fontSize: '12px', color: '#FFD59A' }}>
        {bannerText}
      </div>

      {showFormula ? (
        <div style={{ marginBottom: '16px', padding: '10px 12px', borderRadius: '10px', background: '#121817', border: '1px solid ' + line }}>
          <div style={{ fontSize: '11px', fontWeight: 600, color: muted, marginBottom: '4px' }}>Formula</div>
          <code style={{ fontSize: '11px', wordBreak: 'break-all', color: mint }}>{profile.formulaDisplay}</code>
        </div>
      ) : null}

      <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fill, minmax(220px, 1fr))', gap: '0 16px' }}>
        {basicFields.map(renderField)}
      </div>

      {advancedFields.length > 0 ? (
        <div style={{ marginTop: '4px', marginBottom: '12px' }}>
          <button type="button" onClick={function () { setShowAdvanced(!showAdvanced); }} style={btnGhost}>
            {showAdvanced ? 'Hide advanced fields' : 'Show advanced fields (' + advancedFields.length + ')'}
          </button>
          {showAdvanced ? (
            <div style={{ marginTop: '12px', display: 'grid', gridTemplateColumns: 'repeat(auto-fill, minmax(220px, 1fr))', gap: '0 16px' }}>
              {advancedFields.map(renderField)}
            </div>
          ) : null}
        </div>
      ) : null}

      {err ? <p style={{ color: danger, fontSize: '13px', margin: '0 0 12px' }} role="alert">{err}</p> : null}
      {loadNote ? <p style={{ color: muted, fontSize: '11px', margin: '0 0 12px' }}>{loadNote}</p> : null}
      {copyMsg ? <p style={{ color: mint, fontSize: '12px', margin: '0 0 12px' }}>{copyMsg}</p> : null}

      <div style={{ display: 'flex', flexWrap: 'wrap', gap: '10px', marginBottom: '16px' }}>
        <button type="button" onClick={generateHash} disabled={busy} style={btnPrimary}>
          {busy ? 'Generating…' : primaryLabel}
        </button>
        <button type="button" onClick={resetForm} style={btnGhost}>Reset to examples</button>
        {profile.docsUrl ? (
          <a href={profile.docsUrl} target="_blank" rel="noopener noreferrer" style={{ alignSelf: 'center', fontSize: '12px', color: mint }}>
            View docs
          </a>
        ) : null}
      </div>

      {hashHex ? (
        <div style={{ marginBottom: '12px' }}>
          <div style={{ fontSize: '11px', fontWeight: 600, color: muted, marginBottom: '6px' }}>Generated hash (SHA-512)</div>
          <div style={{ display: 'flex', flexWrap: 'wrap', gap: '8px', alignItems: 'flex-start' }}>
            <code style={{ flex: '1 1 280px', fontSize: '11px', wordBreak: 'break-all', padding: '10px', borderRadius: '8px', background: '#121817', border: '1px solid ' + line, color: '#EAF2EE' }}>
              {hashHex}
            </code>
            <button type="button" onClick={function () { copyText(hashHex, 'Hash'); }} style={btnGhost}>Copy hash</button>
          </div>
        </div>
      ) : null}

      {showPipeString && pipeString ? (
        <div style={{ marginBottom: '12px' }}>
          <button type="button" onClick={function () { setShowPipe(!showPipe); }} style={Object.assign({}, btnGhost, { marginBottom: '8px', padding: '6px 12px', fontSize: '12px' })}>
            {showPipe ? 'Hide pipe-separated string' : 'Show pipe-separated string'}
          </button>
          {showPipe ? (
            <div style={{ display: 'flex', flexWrap: 'wrap', gap: '8px', alignItems: 'flex-start' }}>
              <code style={{ flex: '1 1 280px', fontSize: '11px', wordBreak: 'break-all', padding: '10px', borderRadius: '8px', background: '#121817', border: '1px solid ' + line, color: muted }}>
                {pipeString}
              </code>
              <button type="button" onClick={function () { copyText(pipeString, 'Pipe string'); }} style={btnGhost}>Copy string</button>
            </div>
          ) : null}
        </div>
      ) : null}

      {relatedLinks.length > 0 ? (
        <div style={{ marginTop: '8px', fontSize: '12px', color: muted }}>
          Related:{' '}
          {relatedLinks.map(function (link, idx) {
            var href = link.url || '';
            return (
              <span key={idx}>
                {idx > 0 ? ' · ' : ''}
                {href ? (
                  <a href={href} target="_blank" rel="noopener noreferrer" style={{ color: mint }}>{link.label}</a>
                ) : (
                  link.label
                )}
              </span>
            );
          })}
        </div>
      ) : null}
    </div>
  );
};

  );
