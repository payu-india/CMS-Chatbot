  function sha512Hex(msg) {
    var enc = new TextEncoder();
    return crypto.subtle.digest('SHA-512', enc.encode(msg)).then(function (buf) {
      return Array.from(new Uint8Array(buf))
        .map(function (b) {
          return b.toString(16).padStart(2, '0');
        })
        .join('');
    });
  }

  function validateForm(p, f) {

      return 'Missing required field: ' + req[i];

      if (!/^\d+(\.\d{1,2})?$/.test(empty(f.amount))) {
        return 'Amount must be a decimal string (e.g. 1499.00).';
      }

    if (empty(f.email) === false) {
      if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(empty(f.email))) {
        return 'Enter a valid email address.';

        return field.label + ' must be valid JSON.';

    }
    return '';
  }

  function generateHash() {
    if (!profile) {
      setErr('Unknown hash profile.');
      return;

    var validationErr = validateForm(profile, form);
    if (validationErr) {
      setErr(validationErr);
      return;

        setPipeString(chain);
        setBusy(false);

      .catch(function () {
        setErr('SHA-512 unavailable in this browser context.');
        setBusy(false);
      });

    if (!value) return;
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(value).then(function () {
        setCopyMsg(label + ' copied.');
        setTimeout(function () { setCopyMsg(''); }, 2000);
      });
      return;
    }
    setCopyMsg('Copy not supported in this browser.');
  }

  if (!profile) {
    return (
      <div className="my-5 rounded-lg border border-red-200 bg-red-50 p-4 dark:border-red-800 dark:bg-red-950" role="alert">
        <strong className="text-gray-900 dark:text-gray-100">PayU Forward Hash Generator</strong>
        <p className="mt-2 text-sm text-red-600 dark:text-red-400">Unknown profile: {profileId}</p>
      </div>
    );
  }

  var sortedFields = (profile.fields || []).slice().sort(function (a, b) {
    return (a.order || 0) - (b.order || 0);
  });

  var basicFields = [];
  var advancedFields = [];
  for (var fi = 0; fi < sortedFields.length; fi += 1) {
    if (sortedFields[fi].advanced) advancedFields.push(sortedFields[fi]);
    else basicFields.push(sortedFields[fi]);
  }

  var primaryLabel = (profile.ux && profile.ux.primaryAction) || 'Generate Forward Hash';
  var bannerText = (profile.security && profile.security.bannerText) || 'Test credentials only. Production hashes must be generated on your server.';
  var showFormula = !profile.ux || profile.ux.showFormula !== false;
  var showPipeString = !profile.ux || profile.ux.showPipeString !== false;
  var relatedLinks = (profile.ux && profile.ux.relatedLinks) || [];

  var inputClass =
    'w-full rounded-md border border-gray-300 bg-white px-3 py-2 text-sm font-mono text-gray-900 shadow-sm focus:border-teal-600 focus:outline-none focus:ring-1 focus:ring-teal-600 dark:border-gray-600 dark:bg-gray-800 dark:text-gray-100';

  function renderField(field) {
    var val = form[field.key] != null ? form[field.key] : '';
    var isJson = field.type === 'json';
    var inputType = field.type === 'password' ? 'password' : field.type === 'email' ? 'email' : 'text';
    return (
      <div key={field.key} className="mb-3">
        <label className="mb-1 block text-sm font-medium text-gray-700 dark:text-gray-200">
          {field.label}
          {field.required ? <span className="text-red-500"> *</span> : null}
        </label>
        {isJson ? (
          <textarea
            rows={3}
            value={val}
            onChange={function (ev) { patchField(field.key, ev.target.value); }}
            placeholder={field.placeholder || ''}
            className={inputClass + ' min-h-[72px] resize-y'}
          />
        ) : (
          <input
            type={inputType}
            value={val}
            onChange={function (ev) { patchField(field.key, ev.target.value); }}
            placeholder={field.placeholder || ''}
            className={inputClass}
            autoComplete={field.type === 'password' ? 'off' : 'on'}
          />
        )}
        {field.help ? <p className="mt-1 text-xs text-gray-500 dark:text-gray-400">{field.help}</p> : null}
      </div>
    );
  }

  return (
    <div className="my-5 rounded-lg border border-gray-200 bg-gray-50 p-5 shadow-sm dark:border-gray-700 dark:bg-gray-900" role="region" aria-label="PayU Forward Hash Generator">
      <div className="mb-2 text-xs font-semibold uppercase tracking-wider text-teal-700 dark:text-teal-400">
        PayU · {profile.product} · Forward hash lab
      </div>
      <h2 className="!mt-0 mb-1 text-xl font-semibold text-gray-900 dark:text-gray-100">{profile.title}</h2>
      <p className="mb-4 text-sm text-gray-600 dark:text-gray-300">{profile.description}</p>

      <div className="mb-4 rounded-md border border-amber-200 bg-amber-50 px-3 py-2 text-sm text-amber-900 dark:border-amber-800 dark:bg-amber-950 dark:text-amber-200">
        {bannerText}
      </div>

      {showFormula ? (
        <div className="mb-4 rounded-md border border-gray-200 bg-white px-3 py-2 dark:border-gray-700 dark:bg-gray-800">
          <div className="mb-1 text-xs font-semibold text-gray-500 dark:text-gray-400">Formula</div>
          <code className="break-all text-xs font-mono text-gray-800 dark:text-gray-200">{profile.formulaDisplay}</code>
        </div>
      ) : null}

      <div className="grid grid-cols-1 gap-x-4 sm:grid-cols-2">
        {basicFields.map(renderField)}
      </div>

      {advancedFields.length > 0 ? (
        <div className="mb-3 mt-1">
          <button
            type="button"
            onClick={function () { setShowAdvanced(!showAdvanced); }}
            className="cursor-pointer rounded-md border border-gray-300 bg-white px-4 py-2 text-sm font-semibold text-gray-700 hover:bg-gray-100 dark:border-gray-600 dark:bg-gray-800 dark:text-gray-200 dark:hover:bg-gray-700"
          >
            {showAdvanced ? 'Hide advanced fields' : 'Show advanced fields (' + advancedFields.length + ')'}
          </button>
          {showAdvanced ? (
            <div className="mt-3 grid grid-cols-1 gap-x-4 sm:grid-cols-2">
              {advancedFields.map(renderField)}
            </div>
          ) : null}
        </div>
      ) : null}

      {err ? <p className="mb-3 text-sm text-red-600 dark:text-red-400" role="alert">{err}</p> : null}
      {loadNote ? <p className="mb-3 text-xs text-gray-500 dark:text-gray-400">{loadNote}</p> : null}
      {copyMsg ? <p className="mb-3 text-sm text-teal-700 dark:text-teal-400">{copyMsg}</p> : null}

      <div className="mb-4 flex flex-wrap items-center gap-2">
        <button
          type="button"
          onClick={generateHash}
          disabled={busy}
          className="cursor-pointer rounded-md border-0 bg-teal-700 px-4 py-2 text-sm font-semibold text-white hover:bg-teal-800 disabled:opacity-50 dark:bg-teal-600 dark:hover:bg-teal-500"
        >
          {busy ? 'Generating…' : primaryLabel}
        </button>
        <button
          type="button"
          onClick={resetForm}
          className="cursor-pointer rounded-md border border-gray-300 bg-white px-4 py-2 text-sm font-semibold text-gray-700 hover:bg-gray-100 dark:border-gray-600 dark:bg-gray-800 dark:text-gray-200 dark:hover:bg-gray-700"
        >
          Reset to examples
        </button>
        {profile.docsUrl ? (
          <a href={profile.docsUrl} target="_blank" rel="noopener noreferrer" className="self-center text-sm text-teal-700 underline hover:text-teal-800 dark:text-teal-400">
            View docs
          </a>
        ) : null}
      </div>

      {hashHex ? (
        <div className="mb-3">
          <div className="mb-2 text-xs font-semibold text-gray-500 dark:text-gray-400">Generated hash (SHA-512)</div>
          <div className="flex flex-wrap items-start gap-2">
            <code className="min-w-[280px] flex-1 break-all rounded-md border border-gray-200 bg-white p-3 text-xs font-mono text-gray-800 dark:border-gray-700 dark:bg-gray-800 dark:text-gray-100">
              {hashHex}
            </code>
            <button
              type="button"
              onClick={function () { copyText(hashHex, 'Hash'); }}
              className="cursor-pointer rounded-md border border-gray-300 bg-white px-4 py-2 text-sm font-semibold text-gray-700 hover:bg-gray-100 dark:border-gray-600 dark:bg-gray-800 dark:text-gray-200"
            >
              Copy hash
            </button>
          </div>
        </div>
      ) : null}

      {showPipeString && pipeString ? (
        <div className="mb-3">
          <button
            type="button"
            onClick={function () { setShowPipe(!showPipe); }}
            className="mb-2 cursor-pointer rounded-md border border-gray-300 bg-white px-3 py-1.5 text-xs font-semibold text-gray-700 hover:bg-gray-100 dark:border-gray-600 dark:bg-gray-800 dark:text-gray-200"
          >
            {showPipe ? 'Hide pipe-separated string' : 'Show pipe-separated string'}
          </button>
          {showPipe ? (
            <div className="flex flex-wrap items-start gap-2">
              <code className="min-w-[280px] flex-1 break-all rounded-md border border-gray-200 bg-white p-3 text-xs font-mono text-gray-600 dark:border-gray-700 dark:bg-gray-800 dark:text-gray-300">
                {pipeString}
              </code>
              <button
                type="button"
                onClick={function () { copyText(pipeString, 'Pipe string'); }}
                className="cursor-pointer rounded-md border border-gray-300 bg-white px-4 py-2 text-sm font-semibold text-gray-700 hover:bg-gray-100 dark:border-gray-600 dark:bg-gray-800 dark:text-gray-200"
              >
                Copy string
              </button>
            </div>
          ) : null}
        </div>
      ) : null}

      {relatedLinks.length > 0 ? (
        <div className="mt-2 text-sm text-gray-600 dark:text-gray-400">
          Related:{' '}
          {relatedLinks.map(function (link, idx) {
            var href = link.url || '';
            return (
              <span key={idx}>
                {idx > 0 ? ' · ' : ''}
                {href ? (
                  <a href={href} target="_blank" rel="noopener noreferrer" className="text-teal-700 underline hover:text-teal-800 dark:text-teal-400">{link.label}</a>
                ) : (
                  link.label
                )}
              </span>
            );
          })}
        </div>
      ) : null}
    </div>
  );
};
