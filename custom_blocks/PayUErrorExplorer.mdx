export const FILTERS = [
  'All',
  'Webhooks',
  'HTTP',
  'Duplicate',
  'Network',
  'Bank Declines',
  'Validation',
  'Auth',
  'Eligibility',
  'System',
];

export const severityStyles = {
  Low: 'bg-emerald-50 text-emerald-700 border-emerald-200',
  Medium: 'bg-amber-50 text-amber-700 border-amber-200',
  High: 'bg-rose-50 text-rose-700 border-rose-200',
};

export const retryStyles = {
  yes: 'bg-emerald-50 text-emerald-700 border-emerald-200',
  no: 'bg-zinc-100 text-zinc-700 border-zinc-200',
};

export function classifyToFilter(error) {
  if (FILTERS.includes(error.category)) return error.category;
  if (error.commonIn.includes('Webhooks')) return 'Webhooks';
  return 'System';
}

export function normalizeSearchText(error) {
  return `${error.code} ${error.message} ${error.description}`.toLowerCase();
}

export function getCategoryCount(errors, filter) {
  if (filter === 'All') return errors.length;
  return errors.filter((item) => classifyToFilter(item) === filter).length;
}

export const PayUErrorExplorer = () => {
  const normalizedErrors = useMemo(
    () =>
      ERRORS.map((item) => ({
        ...item,
        fix: FIX_TEMPLATES[item.fixTemplate] || item.fixTemplate,
        searchBlob: normalizeSearchText(item),
        groupKey: item.category === 'Duplicate' ? 'Duplicate' : item.fixTemplate,
      })),
    []
  );

  const [query, setQuery] = useState('');
  const [activeFilter, setActiveFilter] = useState('All');

  const initialCode = (() => {
    if (typeof window === 'undefined') return null;
    const value = new URLSearchParams(window.location.search).get('error');
    return value ? value.toUpperCase() : null;
  })();

  const initialMatch = normalizedErrors.find(
    (item) => item.code.toUpperCase() === initialCode
  );
  const initialSelectedCode = (initialMatch && initialMatch.code) || normalizedErrors[0].code;

  const [selectedCode, setSelectedCode] = useState(initialSelectedCode);

  const filteredErrors = useMemo(() => {
    const q = query.trim().toLowerCase();
    return normalizedErrors.filter((item) => {
      const matchesFilter =
        activeFilter === 'All' || classifyToFilter(item) === activeFilter;
      const matchesQuery = q.length === 0 || item.searchBlob.includes(q);
      return matchesFilter && matchesQuery;
    });
  }, [activeFilter, normalizedErrors, query]);

  useEffect(() => {
    if (!filteredErrors.length) return;
    if (!filteredErrors.some((item) => item.code === selectedCode)) {
      setSelectedCode(filteredErrors[0].code);
    }
  }, [filteredErrors, selectedCode]);

  const selectedError = useMemo(
    () => normalizedErrors.find((item) => item.code === selectedCode) || filteredErrors[0] || null,
    [filteredErrors, normalizedErrors, selectedCode]
  );

  useEffect(() => {
    const handleKeyDown = (event) => {
      if (!filteredErrors.length) return;
      if (event.key !== 'ArrowUp' && event.key !== 'ArrowDown') return;
      event.preventDefault();
      const currentIndex = filteredErrors.findIndex((item) => item.code === selectedCode);
      const safeIndex = currentIndex >= 0 ? currentIndex : 0;
      const delta = event.key === 'ArrowDown' ? 1 : -1;
      const nextIndex = (safeIndex + delta + filteredErrors.length) % filteredErrors.length;
      setSelectedCode(filteredErrors[nextIndex].code);
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [filteredErrors, selectedCode]);

  const relatedErrors = useMemo(() => {
    if (!selectedError) return [];
    return normalizedErrors
      .filter((item) => item.code !== selectedError.code)
      .filter(
        (item) =>
          item.category === selectedError.category ||
          item.fixTemplate === selectedError.fixTemplate ||
          item.groupKey === selectedError.groupKey
      )
      .slice(0, 6);
  }, [normalizedErrors, selectedError]);

  const copyText = async (text) => {
    if (!text) return;
    try {
      await navigator.clipboard.writeText(text);
    } catch (_error) {
      // Best-effort copy for docs embeds where clipboard API may be blocked.
    }
  };

  return (
    <div className="w-full bg-zinc-50 text-zinc-900">
      <div className="mx-auto w-full max-w-none px-2 py-3 sm:px-3">
        <div className="rounded-2xl border border-zinc-200 bg-white shadow-sm">
          <div className="grid grid-cols-1 xl:grid-cols-10">
            <aside className="border-b border-zinc-200 p-4 xl:col-span-3 xl:max-h-[82vh] xl:overflow-y-auto xl:border-b-0 xl:border-r">
              <div className="sticky top-0 z-10 -mx-4 -mt-4 border-b border-zinc-200 bg-white px-4 pb-4 pt-4">
                <h1 className="text-lg font-semibold tracking-tight">PayU Error Explorer</h1>
                <p className="mt-1 text-sm text-zinc-600">
                  Searchable error intelligence for faster integration debugging.
                </p>
                <input
                  type="text"
                  value={query}
                  onChange={(event) => setQuery(event.target.value)}
                  placeholder="Search code, message, description..."
                  className="mt-3 w-full rounded-lg border border-zinc-300 bg-white px-3 py-2 text-sm outline-none ring-indigo-500 focus:ring-2"
                />
                <div className="mt-3 flex flex-wrap gap-2">
                  {FILTERS.map((filter) => {
                    const active = activeFilter === filter;
                    return (
                      <button
                        key={filter}
                        type="button"
                        onClick={() => setActiveFilter(filter)}
                        className={`rounded-full border px-3 py-1 text-xs font-medium transition ${
                          active
                            ? 'border-indigo-200 bg-indigo-50 text-indigo-700'
                            : 'border-zinc-200 bg-white text-zinc-600 hover:bg-zinc-100'
                        }`}
                      >
                        {filter} ({getCategoryCount(normalizedErrors, filter)})
                      </button>
                    );
                  })}
                </div>
              </div>

              <div className="mt-4 space-y-2">
                {filteredErrors.length === 0 && (
                  <div className="rounded-xl border border-dashed border-zinc-300 bg-zinc-50 p-4 text-sm text-zinc-600">
                    No errors found. Try a different keyword or category.
                  </div>
                )}

                {filteredErrors.map((error) => {
                  const selected = selectedError && selectedError.code === error.code;
                  return (
                    <button
                      key={error.code}
                      type="button"
                      onClick={() => setSelectedCode(error.code)}
                      className={`w-full rounded-xl border p-3 text-left transition ${
                        selected
                          ? 'border-indigo-200 bg-indigo-50 shadow-sm'
                          : 'border-zinc-200 bg-white hover:bg-zinc-50'
                      }`}
                    >
                      <div className="flex items-center justify-between gap-2">
                        <span className="font-mono text-sm font-semibold text-zinc-900">{error.code}</span>
                        <span className="rounded-full border border-zinc-200 bg-zinc-100 px-2 py-0.5 text-[11px] text-zinc-700">
                          {error.category}
                        </span>
                      </div>
                      <p className="mt-1 text-sm text-zinc-700">{error.message}</p>
                    </button>
                  );
                })}
              </div>
            </aside>

            <section className="p-4 sm:p-5 xl:col-span-7">
              {!selectedError ? (
                <div className="rounded-xl border border-dashed border-zinc-300 bg-zinc-50 p-6 text-sm text-zinc-600">
                  Select an error to view debugging details.
                </div>
              ) : (
                <div className="space-y-6">
                  <header className="flex flex-wrap items-start justify-between gap-3">
                    <div>
                      <div className="flex items-center gap-2">
                        <span className="rounded-md border border-zinc-200 bg-zinc-100 px-2.5 py-1 font-mono text-sm font-semibold">
                          {selectedError.code}
                        </span>
                        <span className="rounded-full border border-indigo-200 bg-indigo-50 px-2.5 py-1 text-xs font-medium text-indigo-700">
                          {selectedError.category}
                        </span>
                      </div>
                      <h2 className="mt-2 text-xl font-semibold tracking-tight">{selectedError.message}</h2>
                    </div>
                    <div className="flex gap-2">
                      <button
                        type="button"
                        onClick={() => copyText(selectedError.code)}
                        className="rounded-lg border border-zinc-200 bg-white px-3 py-2 text-xs font-medium text-zinc-700 hover:bg-zinc-100"
                      >
                        Copy error code
                      </button>
                      <button
                        type="button"
                        onClick={() => copyText(selectedError.fix)}
                        className="rounded-lg border border-zinc-200 bg-white px-3 py-2 text-xs font-medium text-zinc-700 hover:bg-zinc-100"
                      >
                        Copy fix
                      </button>
                    </div>
                  </header>

                  <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
                    <div className="rounded-xl border border-zinc-200 bg-zinc-50 p-3">
                      <p className="text-xs text-zinc-500">Retry safe?</p>
                      <span
                        className={`mt-1 inline-flex rounded-full border px-2 py-0.5 text-xs font-medium ${
                          selectedError.retrySafe ? retryStyles.yes : retryStyles.no
                        }`}
                      >
                        {selectedError.retrySafe ? 'Yes' : 'No'}
                      </span>
                    </div>
                    <div className="rounded-xl border border-zinc-200 bg-zinc-50 p-3">
                      <p className="text-xs text-zinc-500">Severity</p>
                      <span
                        className={`mt-1 inline-flex rounded-full border px-2 py-0.5 text-xs font-medium ${
                          severityStyles[selectedError.severity]
                        }`}
                      >
                        {selectedError.severity}
                      </span>
                    </div>
                    <div className="rounded-xl border border-zinc-200 bg-zinc-50 p-3">
                      <p className="text-xs text-zinc-500">Common in</p>
                      <p className="mt-1 text-sm font-medium text-zinc-800">
                        {selectedError.commonIn.join(', ')}
                      </p>
                    </div>
                  </div>

                  <article className="space-y-4">
                    <div>
                      <h3 className="text-sm font-semibold text-zinc-800">Error Message</h3>
                      <p className="mt-1 text-sm text-zinc-700">{selectedError.message}</p>
                    </div>
                    <div>
                      <h3 className="text-sm font-semibold text-zinc-800">Description</h3>
                      <p className="mt-1 text-sm text-zinc-700">{selectedError.description}</p>
                    </div>
                    <div>
                      <h3 className="text-sm font-semibold text-zinc-800">Possible Causes</h3>
                      <ul className="mt-1 list-disc space-y-1 pl-5 text-sm text-zinc-700">
                        {selectedError.causes.map((cause) => (
                          <li key={cause}>{cause}</li>
                        ))}
                      </ul>
                    </div>
                    <div className="rounded-xl border border-emerald-200 bg-emerald-50 p-4">
                      <h3 className="text-sm font-semibold text-emerald-800">Recommended Fix</h3>
                      <p className="mt-1 text-sm text-emerald-900">{selectedError.fix}</p>
                    </div>
                  </article>

                  <section>
                    <h3 className="text-sm font-semibold text-zinc-800">Related Errors</h3>
                    <div className="mt-2 flex flex-wrap gap-2">
                      {relatedErrors.length === 0 && (
                        <span className="text-sm text-zinc-500">No related errors available.</span>
                      )}
                      {relatedErrors.map((error) => (
                        <button
                          key={error.code}
                          type="button"
                          onClick={() => setSelectedCode(error.code)}
                          className="rounded-lg border border-zinc-200 bg-white px-3 py-1.5 text-xs font-medium text-zinc-700 hover:bg-zinc-100"
                        >
                          <span className="font-mono">{error.code}</span> - {error.message}
                        </button>
                      ))}
                    </div>
                  </section>
                </div>
              )}
            </section>
          </div>
        </div>
      </div>
    </div>
  );
};
