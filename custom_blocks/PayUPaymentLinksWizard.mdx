  var customerEmail = _customerEmail[0];
  var setCustomerEmail = _customerEmail[1];

  var link = _link[0];
  var setLink = _link[1];

  var _showPreview = useState(false);

    setCopied(false);

          <div style={{ width: '100%', maxWidth: '720px', maxHeight: '92vh', background: 'white', borderRadius: '14px', display: 'flex', flexDirection: 'column', overflow: 'hidden' }}>

                      </div>
                    </div>

                                {rating}
                              </button>
                            );
                          })}

          <div style={{ width: '100%', maxWidth: '420px', background: 'white', borderRadius: '12px', overflow: 'hidden' }}>
