// PayUPaymentLinksWizard - Absolute Minimal Working Version

import { useState } from 'react';

export const PayUPaymentLinksWizard = function PayUPaymentLinksWizard() {
  var _step = useState(0);
  var step = _step[0];
  var setStep = _step[1];

  var _amount = useState('100.00');
  var amount = _amount[0];
  var setAmount = _amount[1];

  var _link = useState('');
  var link = _link[0];
  var setLink = _link[1];

  function makeLink() {
    var id = Date.now().toString(36);
    setLink('https://demo.payu.in/link/' + id);
  }

  var GREEN = '#0f766e';

  return (
    <div style={{ padding: '20px', border: '1px solid #ccc', borderRadius: '8px', margin: '24px 0', fontFamily: 'system-ui' }}>
      <h3 style={{ margin: '0 0 8px' }}>Payment Links Lab - Step {step + 1} of 4</h3>

      {step === 0 ? (
        <div>
          <p>Payment links let you collect money without a website.</p>
          <ul>
            <li>Quick - Create in seconds</li>
            <li>Flexible - Share anywhere</li>
            <li>Safe - Secure checkout</li>
          </ul>
        </div>
      ) : null}

      {step === 1 ? (
        <div>
          <label style={{ display: 'block', marginBottom: '6px', fontWeight: 600 }}>Amount (₹)</label>
          <input
            type="text"
            value={amount}
            onChange={function (e) { setAmount(e.target.value); }}
            style={{ width: '100%', padding: '8px', marginBottom: '10px', boxSizing: 'border-box' }}
          />
          <button
            onClick={makeLink}
            style={{ padding: '10px 16px', background: GREEN, color: 'white', border: 'none', borderRadius: '6px', cursor: 'pointer' }}
          >
            Create Link
          </button>
          {link ? (
            <div style={{ marginTop: '10px', padding: '10px', background: '#ecfdf5', borderRadius: '6px' }}>
              <strong>Your link:</strong>
              <div style={{ fontSize: '12px', wordBreak: 'break-all' }}>{link}</div>
            </div>
          ) : null}
        </div>
      ) : null}

      {step === 2 ? (
        <div>
          <p>Test your payment link. Pick an outcome:</p>
          <button style={{ display: 'block', width: '100%', padding: '10px', marginBottom: '8px', background: GREEN, color: 'white', border: 'none', borderRadius: '6px', cursor: 'pointer' }}>
            ✓ Payment successful
          </button>
        </div>
      ) : null}

      {step === 3 ? (
        <div style={{ textAlign: 'center' }}>
          <div style={{ fontSize: '48px' }}>🎉</div>
          <h3>You are all set!</h3>
          <p>You created and tested your first payment link</p>
        </div>
      ) : null}

      <div style={{ marginTop: '20px', display: 'flex', gap: '8px' }}>
        {step > 0 ? (
          <button onClick={function() { setStep(step - 1); }} style={{ padding: '10px 16px', background: 'white', border: '1px solid ' + GREEN, color: GREEN, borderRadius: '6px', cursor: 'pointer' }}>
            Back
          </button>
        ) : null}
        {step < 3 ? (
          <button onClick={function() { setStep(step + 1); }} style={{ padding: '10px 16px', background: GREEN, color: 'white', border: 'none', borderRadius: '6px', cursor: 'pointer' }}>
            Next
          </button>
        ) : null}
      </div>
    </div>
  );
};

<PayUPaymentLinksWizard />
