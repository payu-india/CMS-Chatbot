  function validateFields() {
    var errors = {};
    if (isBlank(amount)) errors.amount = 'Please enter an amount';
    else if (!/^\d+(\.\d{1,2})?$/.test(amount)) errors.amount = 'Enter a valid amount like 100.00';
    if (isBlank(purpose)) errors.purpose = 'Please tell us what this payment is for';
    if (!isBlank(customerEmail) && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(customerEmail)) {
      errors.customerEmail = 'Please enter a valid email address';
    }
    if (!isBlank(customerPhone) && !/^\d{10}$/.test(customerPhone.replace(/\s/g, ''))) {
      errors.customerPhone = 'Please enter a 10-digit phone number';
    }
    return errors;
  }

  function copyText(text) {
    var value = String(text || '');
    if (!value) return;
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(value).then(function () {
        setCopyMsg('✓ Link copied to clipboard!');
        setTimeout(function () { setCopyMsg(''); }, 3000);
      });
    } else {
      // Fallback for older browsers
      try {
        var textArea = document.createElement('textarea');
        textArea.value = value;
        textArea.style.position = 'fixed';
        textArea.style.left = '-999999px';
        document.body.appendChild(textArea);
        textArea.select();
        document.execCommand('copy');
        document.body.removeChild(textArea);
        setCopyMsg('✓ Link copied to clipboard!');
        setTimeout(function () { setCopyMsg(''); }, 3000);
      } catch (e) {
        setCopyMsg('Please copy the link manually');
        setTimeout(function () { setCopyMsg(''); }, 3000);
      }
    }
  }

    function lockScroll() {
      document.body.style.overflow = 'hidden';
    }
    function unlockScroll() {
      document.body.style.overflow = '';
    }

    function onKey(e) {
      if (e.key === 'Escape') {
        closeModal();
        return;
      }
    }

    lockScroll();
    document.addEventListener('keydown', onKey);
    focusRaf = requestAnimationFrame(function () {
      if (closeBtnRef.current) closeBtnRef.current.focus();
    });

    return function () {
      cancelAnimationFrame(focusRaf);
      document.removeEventListener('keydown', onKey);
      unlockScroll();
      if (launcherBtnRef.current) launcherBtnRef.current.focus();
    };
  }, [modalOpen]);

  return (
    <div className="min-w-0" style={{ fontFamily: 'system-ui, -apple-system, Segoe UI, Roboto, sans-serif' }}>
      <div style={launcherCardStyle} role="region" aria-label="Payment Links walkthrough">
        <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'flex-start', gap: '16px', justifyContent: 'space-between' }}>
          <div style={{ flex: '1 1 280px', minWidth: 0 }}>
            <div style={{ fontSize: '11px', fontWeight: 700, textTransform: 'uppercase', letterSpacing: '0.08em', color: PAYU_GREEN, marginBottom: '6px' }}>
              Hands-on Lab
            </div>
            <h3 style={{ margin: '0 0 8px', fontSize: '1.25rem', fontWeight: 700, color: '#111827' }}>
              Payment Links Lab
            </h3>
            <p style={{ margin: '0 0 12px', fontSize: '14px', lineHeight: 1.6, color: '#4b5563' }}>
              Practice creating a payment link - no account or login needed. Everything runs in your browser.
            </p>
            <ul style={{ margin: 0, paddingLeft: '18px', fontSize: '13px', lineHeight: 1.7, color: '#374151' }}>
              <li>Fill in 4 simple details</li>
              <li>Create your practice link</li>
              <li>See what customers see</li>
              <li>About 5 minutes</li>
            </ul>
          </div>
          <div style={{ flexShrink: 0, alignSelf: 'center' }}>
            <button type="button" ref={launcherBtnRef} onClick={labClick(openModal)} style={Object.assign({}, wizardBtnStyle('primary', false), { padding: '12px 24px', fontSize: '15px' })}>
              Open lab
            </button>
          </div>
        </div>
      </div>

      {modalOpen ? (
        <div style={modalOverlayStyle} onClick={onOverlayClick} role="dialog" aria-modal="true" aria-label="Payment Links walkthrough">
          <div ref={modalPanelRef} style={modalPanelStyle} onClick={function(e) { if (e && e.stopPropagation) e.stopPropagation(); }}>
            <div style={modalHeaderStyle}>
              <div style={{ display: 'flex', alignItems: 'flex-start', justifyContent: 'space-between', gap: '12px' }}>
                <div style={{ minWidth: 0 }}>
                  <div style={{ fontSize: '11px', fontWeight: 600, textTransform: 'uppercase', letterSpacing: '0.1em', opacity: 0.85, marginBottom: '4px' }}>
                    PayU Payment Links
                  </div>
                  <h2 style={{ margin: 0, fontSize: '1.2rem', fontWeight: 700, lineHeight: 1.3 }}>{current.title}</h2>
                  {step >= 0 ? (
                    <p style={{ margin: '6px 0 0', fontSize: '13px', opacity: 0.9 }}>Step {current.stepLabel} of 4</p>
                  ) : null}
                </div>
                <button
                  type="button"
                  ref={closeBtnRef}
                  onClick={labClick(closeModal)}
                  aria-label="Close walkthrough"
                  style={{ flexShrink: 0, cursor: 'pointer', border: 'none', background: 'rgba(255,255,255,0.2)', color: '#fff', borderRadius: '8px', width: '44px', height: '44px', fontSize: '18px', lineHeight: 1 }}
                >
                  ×
                </button>
              </div>
              {renderStepPills()}
              {renderModalTabs()}
              <div style={{ marginTop: '12px', height: '4px', background: 'rgba(255,255,255,0.25)', borderRadius: '4px', overflow: 'hidden' }}>
                <div style={{ height: '100%', width: pct + '%', background: '#fff', transition: 'width 0.25s ease' }} />
              </div>
            </div>
            <div style={modalBodyStyle}>
              {modalTab === 'help' ? renderHelpPanel() : renderWizardBody()}
            </div>
            <div style={modalFooterStyle}>
              {modalTab === 'lab' ? renderWizardFooter() : null}
              <p style={Object.assign({}, mutedTextStyle, { marginTop: modalTab === 'lab' ? '8px' : 0, marginBottom: 0 })}>
                {modalTab === 'help' ? 'Help runs locally - Esc to close' : 'Progress saved in this browser - Esc to close'}
              </p>
            </div>
          </div>
        </div>
      ) : null}

      {showCheckoutPreview ? renderCheckoutPreview() : null}
    </div>
  );
};
