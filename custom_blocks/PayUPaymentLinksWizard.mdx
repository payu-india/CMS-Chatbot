  function defaultProgress() {
    return {
      fieldsCompleted: false,
      linkCreated: false,
      linkShared: false,
      linkTested: false

  function wizardFresh() {
    return {
      progress: defaultProgress(),

    };

  var LAB_KEY = LAB_CONFIG.storageKey;

  function labRead() {
    try {
      if (typeof window === 'undefined') return null;
      if (typeof localStorage === 'undefined') return null;
      var raw = localStorage.getItem(LAB_KEY);
      if (!raw) return null;
      return JSON.parse(raw);
    } catch (e) { return null; }
  }

  function labWrite(data) {
    try {
      if (typeof window === 'undefined') return false;
      if (typeof localStorage === 'undefined') return false;
      var snapshot = Object.assign({}, data, { updatedAt: Date.now() });
      localStorage.setItem(LAB_KEY, JSON.stringify(snapshot));
      return true;
    } catch (e) { return false; }
  }

  function labEnsure() {
    var existing = labRead();
    if (existing && existing.progress) return existing;
    return wizardFresh();
  }

  function wizardProgress() {
    if (lab && lab.progress) return lab.progress;
    return defaultProgress();
  }

  function openModal() {
    setModalOpen(true);
    setModalTab('lab');
    setErr('');
    setMsg('');
    var fresh = wizardFresh();
    labWrite(fresh);
    setLab(fresh);
    setPrereq({});
    setGoDone({});
    setStep(0);
  }

  function closeModal() {
    setModalOpen(false);
    setModalTab('lab');
  }

  function onOverlayClick(e) {
    if (e.target === e.currentTarget) closeModal();
  }

  function labClick(fn) {
    return function (e) {
      if (e && e.stopPropagation) e.stopPropagation();
      fn(e);
    };
  }

  function goToStep(n) {
    if (n > step && !canAdvance()) return;
    setStep(n);
    setErr('');
    setMsg('');
  }

  function prereqDone(map) {
    for (var i = 0; i < PREREQ.length; i += 1) {
      if (!map[PREREQ[i].id]) return false;
    }
    return true;
  }

  function allChecked(map, list) {
    for (var i = 0; i < list.length; i += 1) {
      if (!map[i]) return false;
    }
    return true;
  }

  function canAdvance() {
    var p = wizardProgress();
    if (step === 0) return true;
    if (step === 1) return prereqDone(prereq);
    if (step === 2) return true;
    if (step === 3) return p.fieldsCompleted;
    if (step === 4) return p.linkCreated;
    if (step === 5) return p.linkShared;
    if (step === 6) return p.linkTested;
    if (step === 7) return true;
    return true;
  }

  function continueHint() {
    if (canAdvance()) return '';
    if (step === 1) return 'Tick all checkboxes above.';
    if (step === 3) return 'Mark fields as reviewed.';
    if (step === 4) return 'Create link in Dashboard and mark as created.';
    if (step === 5) return 'Mark link as shared.';
    if (step === 6) return 'Complete test payment.';
    return '';
  }

  function renderWizardBody() {
    var inner = null;
    if (step === 0) inner = renderWelcome();
    else if (step === 1) inner = renderPrereq();
    else if (step === 2) inner = renderNavigate();
    else if (step === 3) inner = renderFields();
    else if (step === 4) inner = renderCreate();
    else if (step === 5) inner = renderShare();
    else if (step === 6) inner = renderTest();
    else if (step === 7) inner = renderManage();
    else if (step === 8) inner = renderGolive();

    return (
      <div>
        {step > 0 && current.tagline ? <p style={taglineStyle}>{current.tagline}</p> : null}
        {step >= 2 && step <= 6 ? (
          <div className="mb-4 flex items-start gap-3 rounded-md border border-teal-200 bg-white/90 px-3 py-2.5" role="note">
            <span className="mt-0.5 shrink-0 rounded bg-teal-700 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wider text-white">Dashboard Guide</span>
            <p className="mb-0 min-w-0 text-sm leading-snug text-gray-700">
              <span className="font-semibold text-gray-900">Follow along in your PayU Dashboard</span> — no API calls or coding required.
            </p>
          </div>
        ) : null}
        {step > 1 ? (
          <LabStatus items={[
            { label: 'Fields reviewed', ok: prog.fieldsCompleted },
            { label: 'Link created', ok: prog.linkCreated },
            { label: 'Link shared', ok: prog.linkShared },
            { label: 'Payment tested', ok: prog.linkTested }
          ]} />
        ) : null}
        <div aria-live="polite" aria-atomic="true">
          {err ? <p style={errTextStyle}>{err}</p> : null}
          {msg ? <p style={okTextStyle}>{msg}</p> : null}
        </div>
        <div style={step === 0 ? { marginTop: '4px' } : contentCardStyle}>{inner}</div>
      </div>
    );
  }

  function renderWizardFooter() {
    var canGoForward = canAdvance();
    return (
      <div>
        <div style={navRowStyle}>
          {step > 0 ? (
            renderWizardButton('secondary', 'Back', labClick(function () { goToStep(step - 1); }), false)
          ) : (
            <span />
          )}
          {step < STEPS.length - 1 ? (
            renderWizardButton(
              'primary',
              step === 0 ? 'Next' : 'Continue',
              labClick(function () {
                if (!canAdvance()) return;
                goToStep(step + 1);
              }),
              !canGoForward
            )
          ) : null}
        </div>
        {!canGoForward && continueHint() ? (
          <p style={{ margin: '10px 0 0', fontSize: '14px', lineHeight: 1.5, color: '#b45309' }}>{continueHint()}</p>
        ) : null}
      </div>
    );
  }

  useEffect(function () {
    if (!modalOpen || typeof document === 'undefined') return;
    var focusRaf = 0;

    function lockScroll() {
      document.body.style.overflow = 'hidden';
    }
    function unlockScroll() {
      document.body.style.overflow = '';
    }

    function onKey(e) {
      if (e.key === 'Escape') {
        closeModal();
        return;
      }
    }

    lockScroll();
    document.addEventListener('keydown', onKey);
    focusRaf = requestAnimationFrame(function () {
      if (closeBtnRef.current) closeBtnRef.current.focus();
    });

    return function () {
      cancelAnimationFrame(focusRaf);
      document.removeEventListener('keydown', onKey);
      unlockScroll();
      if (launcherBtnRef.current) launcherBtnRef.current.focus();
    };
  }, [modalOpen]);

  return (
    <div className="min-w-0" style={{ fontFamily: 'system-ui, -apple-system, Segoe UI, Roboto, sans-serif' }}>
      <div style={launcherCardStyle} role="region" aria-label="Payment Links walkthrough">
        <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'flex-start', gap: '16px', justifyContent: 'space-between' }}>
          <div style={{ flex: '1 1 280px', minWidth: 0 }}>
            <div style={{ fontSize: '11px', fontWeight: 700, textTransform: 'uppercase', letterSpacing: '0.08em', color: PAYU_GREEN, marginBottom: '6px' }}>
              Dashboard Walkthrough
            </div>
            <h3 style={{ margin: '0 0 8px', fontSize: '1.25rem', fontWeight: 700, color: '#111827' }}>
              Payment Links Guide
            </h3>
            <p style={{ margin: '0 0 12px', fontSize: '14px', lineHeight: 1.6, color: '#4b5563' }}>
              Learn to create and manage payment links using PayU Dashboard — no coding required.
            </p>
            <ul style={{ margin: 0, paddingLeft: '18px', fontSize: '13px', lineHeight: 1.7, color: '#374151' }}>
              <li>Step-by-step Dashboard navigation</li>
              <li>Understand all form fields</li>
              <li>Create, share, and test links</li>
              <li>About 10 minutes</li>
            </ul>
          </div>
          <div style={{ flexShrink: 0, alignSelf: 'center' }}>
            <button type="button" ref={launcherBtnRef} onClick={labClick(openModal)} style={Object.assign({}, wizardBtnStyle('primary', false), { padding: '12px 24px', fontSize: '15px' })}>
              Open guide
            </button>
          </div>
        </div>
      </div>

      {modalOpen ? (
        <div style={modalOverlayStyle} onClick={onOverlayClick} role="dialog" aria-modal="true" aria-label="Payment Links walkthrough">
          <div ref={modalPanelRef} style={modalPanelStyle} onClick={function(e) { if (e && e.stopPropagation) e.stopPropagation(); }}>
            <div style={modalHeaderStyle}>
              <div style={{ display: 'flex', alignItems: 'flex-start', justifyContent: 'space-between', gap: '12px' }}>
                <div style={{ minWidth: 0 }}>
                  <div style={{ fontSize: '11px', fontWeight: 600, textTransform: 'uppercase', letterSpacing: '0.1em', opacity: 0.85, marginBottom: '4px' }}>
                    PayU Payment Links
                  </div>
                  <h2 style={{ margin: 0, fontSize: '1.2rem', fontWeight: 700, lineHeight: 1.3 }}>{current.title}</h2>
                  {step > 0 ? (
                    <p style={{ margin: '6px 0 0', fontSize: '13px', opacity: 0.9 }}>Step {current.stepLabel} of 8</p>
                  ) : null}
                </div>
                <button
                  type="button"
                  ref={closeBtnRef}
                  onClick={labClick(closeModal)}
                  aria-label="Close walkthrough"
                  style={{ flexShrink: 0, cursor: 'pointer', border: 'none', background: 'rgba(255,255,255,0.2)', color: '#fff', borderRadius: '8px', width: '44px', height: '44px', fontSize: '18px', lineHeight: 1 }}
                >
                  ×
                </button>
              </div>
              {renderStepPills()}
              {renderModalTabs()}
              <div style={{ marginTop: '12px', height: '4px', background: 'rgba(255,255,255,0.25)', borderRadius: '4px', overflow: 'hidden' }}>
                <div style={{ height: '100%', width: pct + '%', background: '#fff', transition: 'width 0.25s ease' }} />
              </div>
            </div>
            <div style={modalBodyStyle}>
              {modalTab === 'help' ? renderHelpPanel() : renderWizardBody()}
            </div>
            <div style={modalFooterStyle}>
              {modalTab === 'lab' ? renderWizardFooter() : null}
              <p style={Object.assign({}, mutedTextStyle, { marginTop: modalTab === 'lab' ? '8px' : 0, marginBottom: 0 })}>
                {modalTab === 'help' ? 'Help runs locally · Esc to close' : 'Progress saved in this browser · Esc to close'}
              </p>
            </div>
          </div>
        </div>
      ) : null}
    </div>
  );
};
