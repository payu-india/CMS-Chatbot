  function saveCredentials() {
    if (isBlank(clientId)) { setErr('Client ID is required.'); return; }

  function generateToken() {
    if (!lab.progress.credentialsSaved) { setErr('Save credentials first.'); return; }
    if (isBlank(clientId)) { setErr('Client ID missing.'); return; }
    if (isBlank(clientSecret)) { setErr('Client Secret missing.'); return; }

    setBusy(true);
    setErr('');
    setMsg('Generating token...');

    var scopeValue = selectedScope === 'all' ? 'create_payment_links update_payment_links read_payment_links' : selectedScope;

    var body = new URLSearchParams();
    body.append('client_id', clientId);
    body.append('client_secret', clientSecret);
    body.append('grant_type', 'client_credentials');
    body.append('scope', scopeValue);

    fetch(LAB_CONFIG.testTokenUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: body
    })
    .then(function (res) { return res.json(); })
    .then(function (data) {
      if (data.access_token) {
        var token = data.access_token;
        setAccessToken(token);
        var labData = labEnsure();
        labData.accessToken = token;
        labData.tokenExpiry = Date.now() + (data.expires_in || 7200) * 1000;
        labData.progress.tokenGenerated = true;
        labWrite(labData);
        setLab(labData);
        setMsg('Token generated successfully. Valid for ' + Math.floor((data.expires_in || 7200) / 60) + ' minutes.');
        setErr('');
      } else {
        setErr('Token generation failed: ' + (data.error_description || data.error || 'Unknown error'));
        setMsg('');
      }
      setBusy(false);
    })
    .catch(function (e) {
      setErr('Network error: ' + e.message);
      setMsg('');
      setBusy(false);
    });
  }

  function createPaymentLink() {
    if (!lab.progress.tokenGenerated) { setErr('Generate access token first.'); return; }
    if (isBlank(accessToken)) { setErr('Access token missing.'); return; }
    if (isBlank(merchantId)) { setErr('Merchant ID missing.'); return; }
    if (isBlank(linkAmount)) { setErr('Amount is required.'); return; }
    if (isBlank(linkDescription)) { setErr('Description is required.'); return; }

    setBusy(true);
    setErr('');
    setMsg('Creating payment link...');

    var payload = {
      subAmount: parseFloat(linkAmount),
      isPartialPaymentAllowed: false,
      description: linkDescription,
      source: 'API'
    };

    fetch(LAB_CONFIG.testApiUrl, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer ' + accessToken,
        'merchantId': merchantId
      },
      body: JSON.stringify(payload)
    })
    .then(function (res) { return res.json(); })
    .then(function (data) {
      if (data.status === 0 && data.result && data.result.paymentLink) {
        var link = data.result.paymentLink;
        setCreatedLinkUrl(link);
        var labData = labEnsure();
        labData.createdLinkUrl = link;
        labData.progress.linkCreated = true;
        labWrite(labData);
        setLab(labData);
        setMsg('Payment link created successfully!');
        setErr('');
      } else {
        setErr('Link creation failed: ' + (data.message || 'Unknown error'));
        setMsg('');
      }
      setBusy(false);
    })
    .catch(function (e) {
      setErr('Network error: ' + e.message);
      setMsg('');
      setBusy(false);
    });
  }

    var base = variant === 'primary' ? btnPrimaryStyle : btnSecondaryStyle;
    if (!disabled) return Object.assign({}, base);
    if (variant === 'primary') {
      return Object.assign({}, base, { background: '#94a3b8', color: '#f8fafc', cursor: 'not-allowed' });
    }
    return Object.assign({}, base, {
      borderColor: '#cbd5e1', background: '#f1f5f9', color: '#94a3b8', cursor: 'not-allowed'
    });
  }

  function renderWizardButton(variant, label, onClick, disabled) {
    var isDisabled = !!disabled;
    return (
      <button
        type="button"
        disabled={isDisabled}
        onClick={isDisabled ? undefined : onClick}
        style={wizardBtnStyle(variant, isDisabled)}
      >
        {label}
      </button>
    );
  }

  function LabStatus(props) {

  function renderWizardBody() {
    var inner = null;
    if (step === 0) inner = renderWelcome();
    else if (step === 1) inner = renderPrereq();
    else if (step === 2) inner = renderCredentials();
    else if (step === 3) inner = renderToken();
    else if (step === 4) inner = renderCreate();
    else if (step === 5) inner = renderShare();
    else if (step === 6) inner = renderTest();
    else if (step === 7) inner = renderManage();
    else if (step === 8) inner = renderGolive();

    return (
      <div>
        {step > 0 && current.tagline ? <p style={taglineStyle}>{current.tagline}</p> : null}
        {step >= 2 && step <= 6 ? (
          <div className="mb-4 flex items-start gap-3 rounded-md border border-teal-200 bg-white/90 px-3 py-2.5" role="note">
            <span className="mt-0.5 shrink-0 rounded bg-teal-700 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wider text-white">Try it</span>

              <span className="font-semibold text-gray-900">Hands-on lab — </span>
              Real API calls to PayU test environment. Everything stays in your browser.
            </p>
          </div>
        ) : null}
        {step > 1 ? (
          <LabStatus items={[
            { label: 'Credentials saved', ok: prog.credentialsSaved },
            { label: 'Token generated', ok: prog.tokenGenerated },
            { label: 'Link created', ok: prog.linkCreated },

              'primary',

              labClick(function () {

                goToStep(step + 1);
              }),
              !canGoForward

        ) : null}
      </div>
    );
  }

  useEffect(function () {
    if (!modalOpen || typeof document === 'undefined') return;
    var focusRaf = 0;

    function lockScroll() {
      document.body.style.overflow = 'hidden';
    }
    function unlockScroll() {
      document.body.style.overflow = '';
    }

    function onKey(e) {
      if (e.key === 'Escape') {
        closeModal();
        return;
      }
    }

    lockScroll();
    document.addEventListener('keydown', onKey);
    focusRaf = requestAnimationFrame(function () {
      if (closeBtnRef.current) closeBtnRef.current.focus();
    });

    return function () {
      cancelAnimationFrame(focusRaf);
      document.removeEventListener('keydown', onKey);
      unlockScroll();
      if (launcherBtnRef.current) launcherBtnRef.current.focus();
    };
  }, [modalOpen]);

            </div>

              Payment Links walkthrough
            </h3>
            <p style={{ margin: '0 0 12px', fontSize: '14px', lineHeight: 1.6, color: '#4b5563' }}>
              Create payment links via API, test in sandbox, and learn link management — all in one popup.
            </p>
            <ul style={{ margin: 0, paddingLeft: '18px', fontSize: '13px', lineHeight: 1.7, color: '#374151' }}>
              <li>OAuth 2.0 token generation</li>
              <li>Real API integration with test environment</li>
              <li>Test payment with demo credentials</li>
              <li>About 15 minutes</li>
            </ul>
          </div>
          <div style={{ flexShrink: 0, alignSelf: 'center' }}>
            <button type="button" ref={launcherBtnRef} onClick={labClick(openModal)} style={Object.assign({}, wizardBtnStyle('primary', false), { padding: '12px 24px', fontSize: '15px' })}>
              Open walkthrough
            </button>
          </div>
        </div>
      </div>

      {modalOpen ? (
        <div style={modalOverlayStyle} onClick={onOverlayClick} role="dialog" aria-modal="true" aria-label="Payment Links walkthrough">

              <div style={{ display: 'flex', alignItems: 'flex-start', justifyContent: 'space-between', gap: '12px' }}>
                <div style={{ minWidth: 0 }}>
                  <div style={{ fontSize: '11px', fontWeight: 600, textTransform: 'uppercase', letterSpacing: '0.1em', opacity: 0.85, marginBottom: '4px' }}>
                    PayU Payment Links
                  </div>

                  {step > 0 ? (
                    <p style={{ margin: '6px 0 0', fontSize: '13px', opacity: 0.9 }}>Step {current.stepLabel} of 8</p>
                  ) : null}
                </div>
                <button
                  type="button"
                  ref={closeBtnRef}
                  onClick={labClick(closeModal)}
                  aria-label="Close walkthrough"
                  style={{ flexShrink: 0, cursor: 'pointer', border: 'none', background: 'rgba(255,255,255,0.2)', color: '#fff', borderRadius: '8px', width: '44px', height: '44px', fontSize: '18px', lineHeight: 1 }}
                >
                  ×
                </button>
              </div>
              {renderStepPills()}
              {renderModalTabs()}
              <div style={{ marginTop: '12px', height: '4px', background: 'rgba(255,255,255,0.25)', borderRadius: '4px', overflow: 'hidden' }}>
                <div style={{ height: '100%', width: pct + '%', background: '#fff', transition: 'width 0.25s ease' }} />
              </div>
            </div>
            <div style={modalBodyStyle}>
              {modalTab === 'help' ? renderHelpPanel() : renderWizardBody()}
            </div>
            <div style={modalFooterStyle}>
              {modalTab === 'lab' ? renderWizardFooter() : null}
              <p style={Object.assign({}, mutedTextStyle, { marginTop: modalTab === 'lab' ? '8px' : 0, marginBottom: 0 })}>
                {modalTab === 'help' ? 'Help runs locally · Esc to close' : 'Progress saved in this browser · Esc to close'}
              </p>
            </div>
          </div>
        </div>
      ) : null}
    </div>
  );
};
