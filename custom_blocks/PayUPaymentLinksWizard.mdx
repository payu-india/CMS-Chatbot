  function validateFields() {
    var errors = {};
    if (isBlank(amount)) errors.amount = 'Amount is required';
    else if (!/^\d+(\.\d{1,2})?$/.test(amount)) errors.amount = 'Enter valid amount (e.g., 100.00)';
    if (isBlank(purpose)) errors.purpose = 'Purpose is required';
    if (!isBlank(customerEmail) && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(customerEmail)) {
      errors.customerEmail = 'Enter valid email';
    }
    if (!isBlank(customerPhone) && !/^\d{10}$/.test(customerPhone.replace(/\s/g, ''))) {
      errors.customerPhone = 'Enter 10-digit phone number';
    }
    return errors;
  }

  function canAdvance() {
    var p = wizardProgress();
    if (step === 0) return true;
    if (step === 1) return prereqDone(prereq);
    if (step === 2) return true;
    if (step === 3) return p.linkCreated && !isBlank(generatedLink);
    if (step === 4) return p.linkShared;
    if (step === 5) return p.linkTested;
    if (step === 6) return true;
    return true;
  }

    return '';
  }

  useEffect(function () {

    function lockScroll() {
      document.body.style.overflow = 'hidden';
    }
    function unlockScroll() {
      document.body.style.overflow = '';
    }

    function onKey(e) {
      if (e.key === 'Escape') {
        closeModal();
        return;
      }
    }

    lockScroll();
    document.addEventListener('keydown', onKey);
    focusRaf = requestAnimationFrame(function () {
      if (closeBtnRef.current) closeBtnRef.current.focus();
    });

    return function () {
      cancelAnimationFrame(focusRaf);
      document.removeEventListener('keydown', onKey);
      unlockScroll();
      if (launcherBtnRef.current) launcherBtnRef.current.focus();
    };
  }, [modalOpen]);
